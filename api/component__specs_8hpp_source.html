23inline const nlohmann::json& root_rule(const nlohmann::json& spec)
24{
25 for (const auto& rule : spec) {
26 if (rule.value("pointer", "") == "/") return rule;
27 }
28 static const nlohmann::json none = nlohmann::json::object();
29 return none;
30}

32inline std::string first_sentence(const std::string& text)
33{
34 for (size_t i = 0; i + 1 < text.size(); ++i) {
35 if ((text[i] == '.' || text[i] == '!' || text[i] == '?') && text[i + 1] == ' ') {
36 return text.substr(0, i + 1);
37 }
38 }
39 return text;
40}

42// The top-level options a spec has, in its order.
43inline std::vector<std::string> option_names(const nlohmann::json& spec)
44{
45 std::vector<std::string> names;
46 for (const auto& rule : spec) {
47 const std::string pointer = rule.value("pointer", "");
48 if (pointer.size() > 1 && pointer.find('/', 1) == std::string::npos &&
49 std::find(names.begin(), names.end(), pointer.substr(1)) == names.end()) {
50 names.push_back(pointer.substr(1));
51 }
52 }
53 return names;
54}

56// The keys of options that the spec does not have. A component ignores them, except
57// topological_offset, which refuses them. input_dir (injected by wmtk_app and the bindings) and
58// $schema (an editor's, removed before the component runs) are not reported.
59inline std::vector<std::string> unknown_options(
60 const nlohmann::json& spec,
61 const nlohmann::json& options)
62{
63 std::vector<std::string> unknown;
64 if (!options.is_object()) return unknown;
65 const auto names = option_names(spec);
66 const std::set<std::string> known(names.begin(), names.end());
67 for (const auto& [key, value] : options.items()) {
68 if (!known.count(key) && key != "input_dir" && key != "$schema") unknown.push_back(key);
69 }
70 return unknown;
71}

73// One line per option: name, type, default, and the first sentence of its doc.
74inline std::string format_options(const nlohmann::json& spec, const std::string& indent = " ")
75{
76 const auto& root = root_rule(spec);
77 std::set<std::string> required;
78 for (const auto& r : root.value("required", nlohmann::json::array())) {
79 required.insert(r.get<std::string>());
80 }
81 std::string out;
82 for (const auto& name : option_names(spec)) {
83 for (const auto& rule : spec) {
84 if (rule.value("pointer", "") != "/" + name) continue;
85 out += indent + name + " (" + rule.value("type", "");
86 if (required.count(name)) {
87 out += ", required";
88 } else if (rule.contains("default")) {
89 out += ", default " + rule["default"].dump();
90 }
91 out += ")";
92 const std::string doc = first_sentence(rule.value("doc", ""));
93 if (!doc.empty()) out += ": " + doc;
94 out += "\n";
95 break;
96 }
97 }
98 return out;
99}

101// What `help(wildmeshing.<name>)` and `wmtk_app --options <name>` show.
102inline std::string describe(const std::string& name, const nlohmann::json& spec)
103{
104 std::string out = root_rule(spec).value("doc", "");
105 if (!out.empty()) out += "\n\n";
106 out += "Options (every one, with its full description: " + reference_url + name + "/):\n";
107 out += format_options(spec);
108 return out;
109}
