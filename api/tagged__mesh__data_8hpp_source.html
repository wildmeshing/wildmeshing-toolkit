14inline MeshData geometry_data(const MatrixXd& V, const MatrixXi& C, CellType type)
15{
16 MeshData data;

18 data.V.leftCols(V.cols()) = V;
19 data.cells[type].connectivity = C.cast<int64_t>();

23template <typename Vertex, typename Cell>
24MeshData
25geometry_data(const std::vector<Vertex>& vertices, const std::vector<Cell>& cells, CellType type)
26{
27 MeshData data;
28 data.V = DataMatrix<double>::Zero(vertices.size(), 3);
29 for (size_t i = 0; i < vertices.size(); ++i)
30 for (int j = 0; j < vertices[i].size(); ++j) data.V(i, j) = vertices[i][j];

35 data.validate();

38// The envelope a component hands on -- outputs["envelope"] on the memory route,
39// <output>_envelope.msh on the file route -- is the REFERENCE it held its tags to: the geometry its
40// envelopes, tubes and anchors were built around (the input, or the simplified input, per
41// envelope_reference), labelled as its output mesh labels what it held there. In 3D a triangle
42// block (an input surface's face tag, a region's tet tag) and an edge block (the feature curves,
43// feature_<i>), in 2D an edge block, each with one `tags` column per label; the input points as a
44// point block without tags, since the mesh writes nothing for them. No eps: a receiver builds its
45// own envelopes around the reference at its own eps, label by label, so that every stage of a
46// chain holds its tags within one eps of the first component's input. Rebuilt from the mesh it
47// receives, the bound would grow by one eps per stage.
48//
49// append_envelope_block() adds one kind of it: the `type` cells of `block` (geometry_data() of
50// the reference's cells), on vertices of their own after the envelope's, row r labelled by the
51// bits of sets[r] (bit j: names[j]); without names, unlabelled. A block without cells adds
52// nothing. One block per type.
53inline void append_envelope_block(
54 MeshData& envelope,
55 const MeshData& block,
56 const CellType type,
57 const std::vector<InputSet>& sets,
58 const std::vector<std::string>& names)
59{
60 const auto& cells = block.cell(type).connectivity;
61 if (cells.rows() == 0) return;
62 if (envelope.cells.count(type))
63 throw std::runtime_error("An envelope has one block of each cell type");
64 if (!names.empty() && sets.size() != size_t(cells.rows()))
65 throw std::runtime_error("An envelope needs one label set per cell");
66 const Eigen::Index base = envelope.V.rows();
67 envelope.V.conservativeResize(base + block.V.rows(), 3);
68 envelope.V.bottomRows(block.V.rows()) = block.V;
69 auto& out = envelope.cells[type];
70 out.connectivity = (cells.array() + int64_t(base)).matrix();
71 if (names.empty()) return;
72 DataMatrix<uint8_t> member = DataMatrix<uint8_t>::Zero(cells.rows(), names.size());
73 for (Eigen::Index r = 0; r < cells.rows(); ++r)
74 for (size_t j = 0; j < names.size() && j < size_t(SurfaceTagAttributes::max_inputs); ++j)
75 member(r, j) = (sets[r] >> j) & 1;
76 out.attributes["tags"] = {std::move(member), names};
77}

141inline DataMatrix<int64_t> read_orientation(const CellData& cells, const std::string& what)
142{
143 if (!SurfaceTagAttributes::per_input_orientation) return {};
144 const auto it = cells.attributes.find("orientation");
145 if (it == cells.attributes.end()) return {};
146 const auto tags = cells.attributes.find("tags");
147 if (tags == cells.attributes.end() || it->second.labels != tags->second.labels)
148 throw std::runtime_error(what + ": orientation needs the tags' column labels");
149 if (!std::holds_alternative<DataMatrix<int64_t>>(it->second.values))
150 throw std::runtime_error(what + ": orientation must be an Int64 matrix");
151 DataMatrix<int64_t> counts = std::get<DataMatrix<int64_t>>(it->second.values);
152 if (((tags->second.integers().array() == 0) && (counts.array() != 0)).any())
153 throw std::runtime_error(what + ": an orientation count where its row lacks the tag");
154 return counts;
155}

201inline LabelledReference read_labelled_envelope(
202 const MeshData& envelope,
203 const int dim,
204 const std::function<InputSet(const std::string&)>& bits_named,
205 const std::vector<std::string>& feature_names,
206 const std::vector<std::string>& point_names = {})
207{
208 envelope.validate();
209 std::set<std::string> dropped = ComponentIO::dropped_fields(envelope.vertex_attributes, {});
210 const auto t = dim == 3 ? CellType::Triangle : CellType::Edge;
211 for (const auto& [ct, c] : envelope.cells) {
212 if (ct != t && ct != CellType::Point && !(dim == 3 && ct == CellType::Edge))
213 throw std::runtime_error("Incompatible envelope topology");
214 dropped.merge(ComponentIO::dropped_fields(c.attributes, {"tags"}));
215 }
216 ComponentIO::log_dropped("envelope", dropped);
217 const auto labelled = [](const CellData& cells) -> const MeshAttribute& {
218 const auto it = cells.attributes.find("tags");
219 if (it == cells.attributes.end())
220 throw std::runtime_error("A labelled envelope needs tags on each of its blocks");
221 const MeshAttribute& field = it->second;
222 if (!std::holds_alternative<DataMatrix<uint8_t>>(field.values) ||
223 field.labels.size() != size_t(field.cols()))
224 throw std::runtime_error("tags must be a UInt8 membership matrix with column labels");
225 if ((std::get<DataMatrix<uint8_t>>(field.values).array() > 1).any())
226 throw std::runtime_error("tags must be binary");
227 return field;
228 };
229 LabelledReference out;
230 // The rows of `cells` on vertices of their own, renumbered in order.
231 const auto own_vertices = [&](const DataMatrix<int64_t>& cells, MatrixXd& V, MatrixXi& C) {
232 std::vector<int> row_of(envelope.V.rows(), -1);
233 for (Eigen::Index i = 0; i < cells.size(); ++i) row_of[cells.data()[i]] = 0;
234 int n = 0;
235 for (auto& r : row_of)
236 if (r == 0) r = n++;
237 V.resize(n, dim);
238 for (Eigen::Index v = 0; v < envelope.V.rows(); ++v)
239 if (row_of[v] >= 0) V.row(row_of[v]) = envelope.V.row(v).head(dim);
240 C = ComponentIO::indices(cells);
241 for (Eigen::Index i = 0; i < C.size(); ++i) C.data()[i] = row_of[C.data()[i]];
242 };
243 const CellData& rows = envelope.cell(t);
244 const MeshAttribute& row_tags = labelled(rows);
245 const auto& member = std::get<DataMatrix<uint8_t>>(row_tags.values);
246 own_vertices(rows.connectivity, out.V, out.F);
247 out.F_labels.assign(out.F.rows(), 0);
248 for (size_t j = 0; j < row_tags.labels.size(); ++j) {
249 const InputSet bits = bits_named(row_tags.labels[j]);
250 for (Eigen::Index r = 0; r < member.rows(); ++r)
251 if (member(r, Eigen::Index(j))) out.F_labels[r] |= bits;
252 }
253 // The feature curves of the labels the receiver's feature curves have (feature_names), one
254 // input edge per edge and label, as a tagged mesh's own are read.
255 const auto curve_block = envelope.cells.find(CellType::Edge);
256 if (dim == 3 && curve_block != envelope.cells.end()) {
257 const MeshAttribute& curve_tags = labelled(curve_block->second);
258 const auto& on = std::get<DataMatrix<uint8_t>>(curve_tags.values);
259 const auto& E = curve_block->second.connectivity;
260 std::map<int64_t, size_t> vertex_of; // envelope row -> curves.edge_vertices
261 const auto vertex = [&](const int64_t v) {
262 const auto [it, added] = vertex_of.emplace(v, out.curves.edge_vertices.size());
263 if (added)
264 out.curves.edge_vertices.emplace_back(
265 envelope.V(v, 0),
266 envelope.V(v, 1),
267 envelope.V(v, 2));
268 return it->second;
269 };
270 for (size_t j = 0; j < curve_tags.labels.size(); ++j) {
271 const auto k =
272 std::find(feature_names.begin(), feature_names.end(), curve_tags.labels[j]);
273 if (k == feature_names.end()) continue;
274 for (Eigen::Index r = 0; r < E.rows(); ++r) {
275 if (!on(r, Eigen::Index(j))) continue;
276 out.curves.edges.push_back({{vertex(E(r, 0)), vertex(E(r, 1))}});
277 out.curves.edge_input.push_back(int(k - feature_names.begin()));
278 }
279 }
280 }
281 // The points (append_envelope_points()), each with the receiver's point labels it carries
282 // by name: none without tags, which an envelope built without point labels has.
283 if (const auto points = envelope.cells.find(CellType::Point); points != envelope.cells.end()) {
284 const auto& C = points->second.connectivity;
285 out.P.resize(C.rows(), dim);
286 out.P_labels.assign(size_t(C.rows()), 0);
287 for (Eigen::Index r = 0; r < C.rows(); ++r)
288 out.P.row(r) = envelope.V.row(C(r, 0)).head(dim);
289 if (points->second.attributes.count("tags")) {
290 const MeshAttribute& point_tags = labelled(points->second);
291 const auto& on = std::get<DataMatrix<uint8_t>>(point_tags.values);
292 for (size_t j = 0; j < point_tags.labels.size(); ++j) {
293 const auto k =
294 std::find(point_names.begin(), point_names.end(), point_tags.labels[j]);
295 const size_t bit = size_t(k - point_names.begin());
296 if (k == point_names.end() || bit >= size_t(SurfaceTagAttributes::max_inputs))
297 continue;
298 for (Eigen::Index r = 0; r < C.rows(); ++r)
299 if (on(r, Eigen::Index(j))) out.P_labels[size_t(r)] |= InputSet(1) << bit;
300 }
301 }
302 }
303 return out;
304}

321inline HeldRows held_rows(const MatrixXd& V, const MatrixXi& F, const std::vector<InputSet>& labels)
322{
323 HeldRows out;
324 std::vector<Eigen::Index> rows;
325 for (Eigen::Index r = 0; r < F.rows(); ++r) {
326 if (labels[r] != 0) rows.push_back(r);
327 }
328 if (rows.empty()) {
329 log_and_throw_error(
330 "envelope: none of its {} cells carries a label of this mesh",
331 F.rows());
332 }
333 std::vector<int> remap(V.rows(), -1);
334 for (const Eigen::Index r : rows)
335 for (Eigen::Index k = 0; k < F.cols(); ++k) remap[F(r, k)] = 0;
336 int n = 0;
337 for (int& v : remap)
338 if (v == 0) v = n++;
339 out.V.resize(n, V.cols());
340 for (Eigen::Index v = 0; v < V.rows(); ++v)
341 if (remap[v] >= 0) out.V.row(remap[v]) = V.row(v);
342 out.F.resize(rows.size(), F.cols());
343 for (size_t i = 0; i < rows.size(); ++i) {
344 for (Eigen::Index k = 0; k < F.cols(); ++k) out.F(i, k) = remap[F(rows[i], k)];
345 out.labels.push_back(labels[rows[i]]);
346 }
347 logger().info(
348 "envelope: a reference, {} of its {} cells of this mesh's labels held to",
349 rows.size(),
350 F.rows());
351 return out;
352}
353// The labels of tags 0 to count - 1, named by `names` (id -> name): a tag's name, or tag_<id>
354// without one, unless that name is taken, in which case '_' is appended as SimWildMesh::write_msh
355// does; labels must be unique.
356inline std::vector<std::string> tag_labels(
357 const std::map<int64_t, std::string>& names,
358 const size_t count)
359{
360 std::set<std::string> named, labelled;
361 for (const auto& [id, name] : names) named.insert(name);
362 std::vector<std::string> labels;
363 for (size_t j = 0; j < count; ++j) {
364 const auto it = names.find(int64_t(j));
365 const bool unnamed = it == names.end();
366 std::string label = unnamed ? "tag_" + std::to_string(j) : it->second;
367 while (labelled.count(label) || (unnamed && named.count(label))) label += "_";
368 labelled.insert(label);
369 labels.push_back(label);
370 }
371 return labels;
372}
373// The labels of an application's inputs, one per input: input_names, the rest tag_<i>, by the
374// rule of tag_labels(). What tetwild's face tags and triwild's edge tags name the inputs by, so
375// that an input's tags are named alike in every component.
376inline std::vector<std::string> input_labels(
377 const std::vector<std::string>& input_names,
378 const size_t n_inputs)
379{
380 std::map<int64_t, std::string> names;
381 for (size_t i = 0; i < input_names.size() && i < n_inputs; ++i)
382 names[int64_t(i)] = input_names[i];
383 return tag_labels(names, n_inputs);
384}
385template <typename Mesh, typename TupleRange, typename Tags>
386void collect_tags(Mesh& mesh, MeshMap& outputs, CellType type, const TupleRange& cells, Tags tags)
387{
388 // A result without cells has no block yet: the exporter only creates the ones it fills.
389 auto& block = outputs.at("mesh").cells[type];
390 if (block.connectivity.cols() != cell_size(type)) block.connectivity.resize(0, cell_size(type));
391 std::vector<std::string> labels = tag_labels(mesh.m_tag_id_to_name, mesh.m_tags_count);
392 DataMatrix<uint8_t> values = DataMatrix<uint8_t>::Zero(cells.size(), labels.size());
393 for (size_t i = 0; i < cells.size(); ++i)
394 for (size_t j = 0; j < labels.size(); ++j) values(i, j) = tags(cells[i]).count(j) ? 1 : 0;
395 // The fixed tags field replaces per-group named scalar arrays.
396 for (size_t j = 0; j < labels.size(); ++j) {
397 block.attributes.erase(labels[j]);
398 block.attributes.erase("tag_" + std::to_string(j));
399 }
400 // Cells without a tag are the background the component built. They are labelled
401 // explicitly: the IO infers nothing from an empty row.
402 if (std::find(labels.begin(), labels.end(), "ambient") == labels.end()) {
403 const auto empty = (values.template cast<int>().rowwise().sum().array() == 0).eval();
404 if (labels.empty() || empty.any()) {
405 values.conservativeResize(Eigen::NoChange, values.cols() + 1);
406 values.col(values.cols() - 1) = empty.template cast<uint8_t>();
407 labels.push_back("ambient");
408 }
409 }
410 block.attributes["tags"] = {std::move(values), std::move(labels)};
411}
412// The tracked simplices -- `type` Triangle: a tet mesh's faces; Edge: a triangle mesh's edges --
413// whose input set is not empty, as the `type` block of outputs["mesh"]: rows of its V (the
414// vertices in get_vertices() order, as compact_slot_outputs leaves them), one tags column per
415// name (bit j of the input set is names[j]), each row oriented as the simplex's input was. In a
416// build with WMTK_PER_INPUT_ORIENTATION, beside `tags` an `orientation` field with its columns:
417// per row and name, that input's count against the row's vertex order, 0 off the input
418// (read_orientation()). Nothing is written without names.
419template <CellType type, typename Mesh>
420void collect_input_set_tags(
421 const Mesh& mesh,
422 MeshMap& outputs,
423 const std::vector<std::string>& names)
424{
425 static_assert(type == CellType::Triangle || type == CellType::Edge);
426 constexpr size_t N = type == CellType::Triangle ? 3 : 2;
427 if (names.empty()) return;
428 std::vector<int64_t> row(mesh.vert_capacity(), -1);
429 int64_t n = 0;
430 for (const auto& v : mesh.get_vertices()) row[v.vid(mesh)] = n++;
431 std::vector<std::array<int64_t, N>> simplices;
432 std::vector<InputSet> sets;
433 std::vector<int64_t> counts; // per row, names.size() of them: with WMTK_PER_INPUT_ORIENTATION
434 const InputSet mask = names.size() >= size_t(SurfaceTagAttributes::max_inputs)
435 ? ~InputSet(0)
436 : (InputSet(1) << names.size()) - 1;
437 const auto add = [&](const SurfaceTagAttributes& attr, std::array<size_t, N> vids) {
438 // Bits past the names are not these tags -- simwild's region labels -- and are not
439 // written.
440 const InputSet set = attr.input_set() & mask;
441 if (set == 0) return;
442 std::sort(vids.begin(), vids.end());
443 if (attr.orientation_along(vids) < 0) std::swap(vids[N - 2], vids[N - 1]);
444#ifdef WMTK_PER_INPUT_ORIENTATION
445 for (size_t j = 0; j < names.size(); ++j)
446 counts.push_back(((set >> j) & 1) ? attr.input_orientation_along(int(j), vids) : 0);
447#endif
448 std::array<int64_t, N> r;
449 for (size_t k = 0; k < N; ++k) r[k] = row[vids[k]];
450 simplices.push_back(r);
451 sets.push_back(set);
452 };
453 if constexpr (type == CellType::Triangle) {
454 for (const auto& f : mesh.get_faces())
455 add(mesh.m_face_attribute[f.fid(mesh)], mesh.get_face_vids(f));
456 } else {
457 for (const auto& e : mesh.get_edges())
458 add(mesh.m_edge_attribute[e.eid(mesh)], mesh.get_edge_vids(e));
459 }
460 auto& block = outputs.at("mesh").cells[type];
461 block.connectivity.resize(simplices.size(), N);
462 DataMatrix<uint8_t> member = DataMatrix<uint8_t>::Zero(simplices.size(), names.size());
463 for (size_t i = 0; i < simplices.size(); ++i) {
464 for (size_t k = 0; k < N; ++k) block.connectivity(i, k) = simplices[i][k];
465 for (size_t j = 0; j < names.size(); ++j) member(i, j) = (sets[i] >> j) & 1;
466 }
467 block.attributes["tags"] = {std::move(member), names};
468 if (!SurfaceTagAttributes::per_input_orientation) return;
469 DataMatrix<int64_t> orientation(simplices.size(), names.size());
470 for (size_t i = 0; i < simplices.size(); ++i)
471 for (size_t j = 0; j < names.size(); ++j) orientation(i, j) = counts[i * names.size() + j];
472 block.attributes["orientation"] = {std::move(orientation), names};
473}
474// The tet edges tagged `feature` (TetOptimizerMesh::FeatureEdgeAttributes: the feature curves the
475// application was given) whose feature input set is not empty, as the Edge block of
476// outputs["mesh"], on the rows collect_input_set_tags() uses: one tags column per name (bit j of
477// the set is names[j], the input curve j). A curve carries no orientation, so a row is its
478// edge's vertices in ascending order. The order2 kind is not written: a reader derives it again
479// from the faces (tag_order2_edges()). Nothing is written without names or edge tracking.
480template <typename Mesh>
481void collect_feature_edge_tags(
482 const Mesh& mesh,
483 MeshMap& outputs,
484 const std::vector<std::string>& names)
485{
486 using EdgeTags = typename Mesh::FeatureEdgeAttributes;
487 if (names.empty() || !mesh.m_track_feature_edges) return;
488 std::vector<int64_t> row(mesh.vert_capacity(), -1);
489 int64_t n = 0;
490 for (const auto& v : mesh.get_vertices()) row[v.vid(mesh)] = n++;
491 const InputSet mask = names.size() >= size_t(SurfaceTagAttributes::max_inputs)
492 ? ~InputSet(0)
493 : (InputSet(1) << names.size()) - 1;
494 std::vector<std::array<int64_t, 2>> edges;
495 std::vector<InputSet> sets;
496 for (const auto& e : mesh.get_edges()) {
497 const auto& attr = mesh.m_feature_edge_attribute[e.eid(mesh)];
498 if (!attr.has(EdgeTags::feature)) continue;
499 const InputSet set = attr.inputs(EdgeTags::feature) & mask;
500 if (set == 0) continue;
501 const size_t a = e.vid(mesh), b = e.switch_vertex(mesh).vid(mesh);
502 edges.push_back({{row[std::min(a, b)], row[std::max(a, b)]}});
503 sets.push_back(set);
504 }
505 auto& block = outputs.at("mesh").cells[CellType::Edge];
506 block.connectivity.resize(edges.size(), 2);
507 DataMatrix<uint8_t> member = DataMatrix<uint8_t>::Zero(edges.size(), names.size());
508 for (size_t i = 0; i < edges.size(); ++i) {
509 for (size_t k = 0; k < 2; ++k) block.connectivity(i, k) = edges[i][k];
510 for (size_t j = 0; j < names.size(); ++j) member(i, j) = (sets[i] >> j) & 1;
511 }
512 block.attributes["tags"] = {std::move(member), names};
513}
514// The input points (TetOptimizerMesh / TriOptimizerMesh::m_feature_point_inputs, by their
515// anchors) as the Point block of outputs["mesh"], on the rows collect_input_set_tags() uses: one
516// tags column per name (bit j: names[j]), the input points' labels. Each at the vertex standing
517// for its anchor; one whose anchor a collapse merged into another's -- which the collapse allows
518// where each lies in the other's ball -- at the nearest vertex in its ball; one with neither, its
519// region filtered away, is not written. Several at one vertex are one row under all their labels.
520// Nothing is written without names or point tracking. Until 2026-11 no output marked the input
521// points: tetwild's and triwild's `features` listed them beside the mesh.
522template <typename Mesh>
523void collect_point_tags(const Mesh& mesh, MeshMap& outputs, const std::vector<std::string>& names)
524{
525 if (names.empty() || !mesh.m_track_feature_points) return;
526 const auto& inputs = mesh.m_feature_point_inputs;
527 std::vector<int64_t> row(mesh.vert_capacity(), -1);
528 std::vector<size_t> carrier(inputs.size(), size_t(-1)); // anchor -> its vertex
529 int64_t n = 0;
530 for (const auto& v : mesh.get_vertices()) {
531 const size_t vid = v.vid(mesh);
532 row[vid] = n++;
533 const size_t f = mesh.m_feature_point_attribute[vid].m_feature_point_id;
534 if (f < inputs.size()) carrier[f] = vid;
535 }
536 const InputSet mask = names.size() >= size_t(SurfaceTagAttributes::max_inputs)
537 ? ~InputSet(0)
538 : (InputSet(1) << names.size()) - 1;
539 std::map<int64_t, InputSet> sets; // output row -> labels
540 for (size_t f = 0; f < inputs.size(); ++f) {
541 if ((inputs[f] & mask) == 0) continue;
542 size_t vid = carrier[f];
543 if (vid == size_t(-1)) {
544 const double r = mesh.m_feature_point_radius[f];
545 double nearest = r * r;
546 for (const auto& v : mesh.get_vertices()) {
547 const double d =
548 (mesh.m_vertex_attribute[v.vid(mesh)].m_posf - mesh.m_feature_points[f])
549 .squaredNorm();
550 if (d <= nearest) {
551 nearest = d;
552 vid = v.vid(mesh);
553 }
554 }
555 if (vid == size_t(-1)) continue;
556 }
557 sets[row[vid]] |= inputs[f] & mask;
558 }
559 auto& block = outputs.at("mesh").cells[CellType::Point];
560 block.connectivity.resize(Eigen::Index(sets.size()), 1);
561 DataMatrix<uint8_t> member = DataMatrix<uint8_t>::Zero(sets.size(), names.size());
562 Eigen::Index i = 0;
563 for (const auto& [r, set] : sets) {
564 block.connectivity(i, 0) = r;
565 for (size_t j = 0; j < names.size(); ++j) member(i, Eigen::Index(j)) = (set >> j) & 1;
566 ++i;
567 }
568 block.attributes["tags"] = {std::move(member), names};
569}
570// A tagged mesh's point block, as collect_point_tags() writes it: its labels, and one point per
571// row and label -- as an edge is read once per row and label -- with its row of mesh.V and its
572// label's column. None without a point block. A point block needs tags, every row a label, as
573// every other lower block of a tagged mesh.
574struct PointTags
575{
576 std::vector<std::string> names;
577 std::vector<int64_t> rows;
578 std::vector<int> labels;
579};
580inline PointTags read_point_tags(const MeshData& mesh, const std::string& what)
581{
582 PointTags out;
583 const auto it = mesh.cells.find(CellType::Point);
584 if (it == mesh.cells.end() || it->second.connectivity.rows() == 0) return out;
585 const auto tags = it->second.attributes.find("tags");
586 if (tags == it->second.attributes.end())
587 throw std::runtime_error(what + ": its points need tags, one column per label");
588 const MeshAttribute& field = tags->second;
589 if (!std::holds_alternative<DataMatrix<uint8_t>>(field.values) ||
590 field.labels.size() != size_t(field.cols()))
591 throw std::runtime_error("tags must be a UInt8 membership matrix with column labels");
592 const auto& member = std::get<DataMatrix<uint8_t>>(field.values);
593 if ((member.array() > 1).any()) throw std::runtime_error("tags must be binary");
594 if ((member.cast<int>().rowwise().sum().array() == 0).any())
595 throw std::runtime_error(what + ": every tagged point needs a tag");
596 if (field.labels.size() > size_t(SurfaceTagAttributes::max_inputs))
597 throw std::runtime_error(what + ": more point labels than an input set holds");
598 out.names = field.labels;
599 const auto& C = it->second.connectivity;
600 for (Eigen::Index r = 0; r < C.rows(); ++r) {
601 for (Eigen::Index j = 0; j < member.cols(); ++j) {
602 if (!member(r, j)) continue;
603 out.rows.push_back(C(r, 0));
604 out.labels.push_back(int(j));
605 }
606 }
607 return out;
608}
