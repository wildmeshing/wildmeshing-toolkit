1126 double band_cell_term(const std::array<size_t, 4>& vids) const
1127 {
1128 if (m_n_regions > 1 && !m_offset_params.allow_mult_comp) {
1129 log_and_throw_error(
1130 "the energy's band term reads one field, and this input has {} regions",
1131 m_n_regions);
1132 }
1133 if (m_offset_params.band_volume_rule == "exact" ||
1134 m_offset_params.band_volume_rule == "front_flux" || swept_rule()) {
1135 // EXPERIMENTAL_band_volume_rule "exact" (and "front_flux", whose per-cell quantity
1136 // it is, see above): J_t / delta, J_t = int_t (d - delta) dV in closed form
1137 // (ExactBandIntegral), in place of the corner bound
1138 const std::array<Vector3d, 4> p = {
1139 {m_vertex_attribute[vids[0]].m_posf,
1140 m_vertex_attribute[vids[1]].m_posf,
1141 m_vertex_attribute[vids[2]].m_posf,
1142 m_vertex_attribute[vids[3]].m_posf}};
1143 const double delta = m_offset_params.target_distance;
1144 return m_band_exact->value(p, delta) / delta;
1145 }
1146 int64_t best = -1;
1147 return band_cell_corner_bound(vids, best);
1148 }
1150 double band_cell_corner_bound(const std::array<size_t, 4>& vids, int64_t& best) const
1151 {
1152 const double delta = m_offset_params.target_distance;
1153 std::array<Vector3d, 4> p;
1154 for (int j = 0; j < 4; ++j) p[size_t(j)] = m_vertex_attribute[vids[size_t(j)]].m_posf;
1155 const double vol = std::abs((p[1] - p[0]).dot((p[2] - p[0]).cross(p[3] - p[0]))) / 6.;
1156 const double md =
1157 min_mean_distance(*m_input_complex_envelope, m_input_complex_pad, p.data(), 4, best);
1158 return vol * ((md - delta) / delta);
1159 }
