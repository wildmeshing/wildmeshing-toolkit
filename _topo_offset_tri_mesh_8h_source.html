1168 std::shared_ptr<SampleEnvelope> surface_envelope_for_edge(
1169 const std::array<size_t, 2>& vids) const override
1170 {
1171 // Boundary geometry first, in both phases: a segment on any tag-region boundary may not
1172 // drift out of that boundary's tube, and one on several boundaries -- a junction -- is
1173 // held in their intersection. The mask carries the input complex too, since every complex
1174 // simplex lies on tag boundaries, and E_t is built from the input mesh before construction
1175 // touches it, so the per-tag tubes hold the as-loaded geometry.
1176 //
1177 // Keyed on the vertices, not the edge: every caller is an operation asking about a segment
1178 // it is about to create or has just created, whose own edge attributes are not written
1179 // yet. The endpoints' masks are, maintained by the operations themselves (AND at a split,
1180 // OR at a collapse).
1181 uint64_t mask = edge_mask(vids);
1182 bool all_offset = true;
1183 for (const size_t v : vids) {
1184 all_offset = all_offset && m_vertex_extra[v].m_is_on_offset;
1185 }
1186
1187 // The ambiguous case: both endpoints can be on region boundaries and on the offset front
1188 // at once -- a real state, and the point of tracking the two families separately. The
1189 // endpoint-mask AND is then necessary but not sufficient for the segment lying on a shared
1190 // boundary: two vertices on different junctions can share a tag bit by coincidence and be
1191 // joined by an offset chord nowhere near that tag's curve, which a container it was never
1192 // meant to satisfy then refuses to refine. Ask the edge's own class, the only record that
1193 // distinguishes a chord from a boundary.
1194 //
1195 // Reading the slot is safe here and would not be unconditionally: the base checks a
1196 // split's two child segments before writing their attributes, so their slots still hold a
1197 // recycled edge's class. A split child never reaches this branch -- its mask and its
1198 // offset flag are mutually exclusive by construction, so one of `mask` and `all_offset` is
1199 // always empty -- and the m_is_surface_fs guard leaves both constraints standing rather
1200 // than dropping one when a slot is illegible.
1201 if (mask != 0 && all_offset) {
1202 if (const auto found = try_tuple_from_edge(vids)) {
1203 const size_t eid = std::get<1>(*found);
1204 if (m_edge_attribute[eid].m_is_surface_fs) {
1205 if (edge_is_offset(eid)) {
1206 mask = 0; // an offset edge lies on no region boundary
1207 } else {
1208 all_offset = false; // a region edge is not the offset front
1209 }
1210 }
1211 }
1212 }
1213 // Both families compose: whatever holds this segment holds it at once. Phase A holds the
1214 // offset where Phase B left it; Phase B is what moves it, so it contributes nothing there
1215 // -- and null before the offset exists at all, which is the pre-pass.
1216 const std::shared_ptr<SampleEnvelope> base = containment_for(mask, all_offset);
1217 if (base || m_deform_tags.empty()) return base;
1218 // deform_others' ops-only tube: a released boundary is held by no mask -- its vertices
1219 // were freed so smoothing can carry the object -- which would leave the operations free
1220 // to decimate and reposition it. A segment the masks and the offset class do not claim,
1221 // but which lies on a released boundary by its incident faces' current tags, is held to
1222 // the tube around the boundary's current shape. A fresh split child can misread its
1223 // recycled face slots for this one check, and is at worst skipped or over-held once.
1224 if (const auto found = try_tuple_from_edge(vids)) {
1225 if (edge_borders_released_boundary(std::get<0>(*found))) return released_envelope();
1226 }
1227 return nullptr;
1228 }
