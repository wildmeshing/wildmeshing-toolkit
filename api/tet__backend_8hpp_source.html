122 // Same sizing-refinement implementation and shared helpers as TetWildMesh.cpp.
123 // The adapter adds fixed-boundary pins, a per-stage split budget and cancellation.
124 size_t refine_sizing_around_worst(double max_energy) override
125 {
126 const int n_rings = std::max(0, m_params.stuck_refine_rings);
127 const double filter = std::min(std::max(max_energy / 100, m_params.stop_energy), 100.);
128 const auto worst = wmtk::utils::select_worst_cells(
129 tet_capacity(),
130 [this](size_t t) { return tuple_from_tet(t).is_valid(*this); },
131 [this](size_t t) { return std::cbrt(cell_quality(t)); },
132 filter,
133 m_params.stuck_refine_num_worst);
134 if (worst.empty()) return 0;
135 m_force_split_edges.clear();
136 if (m_params.stuck_refine_force_split)
137 for (const auto& [q, t] : worst)
138 m_force_split_edges.insert(
139 wmtk::utils::longest_edge(
140 oriented_tet_vids(t),
141 [this](size_t v) -> const wmtk::Vector3d& {
142 return m_vertex_attribute[v].m_posf;
143 }));
144 std::vector<size_t> seeds;
145 seeds.reserve(4 * worst.size());
146 for (const auto& [q, t] : worst)
147 for (size_t v : oriented_tet_vids(t)) seeds.push_back(v);
148 const auto region = wmtk::utils::grow_vertex_region(seeds, n_rings, [this](size_t v) {
149 return get_one_ring_vids_for_vertex_adj(v);
150 });
151 const auto refined = wmtk::utils::apply_sizing_refinement(
152 region,
153 m_params.stuck_refine_factor,
154 m_params.stuck_refine_min_scalar,
155 [this](size_t v) -> double& { return m_vertex_attribute[v].m_sizing_scalar; });
156 gradation_smooth_sizing(m_params.stuck_refine_gradation, refined);
157 refinements.push_back(
158 {{"iteration", m_iterations_used},
159 {"worst_cells", worst.size()},
160 {"refined_vertices", refined.size()},
161 {"forced_edges", m_force_split_edges.size()},
162 {"maximum_AMIPS", max_energy},
163 {"seconds", seconds()}});
164 logger().debug("Tet sizing refinement: {}", refinements.back().dump());
165 write_progress();
166 return refined.size();
167 }

186 const auto vertices = get_vertices();

189 mx = std::max(mx, q);

198 {"pass", m_debug_pass_name},

242 void validate_fixed_boundary() const
243 {
244 if (boundary() != source_boundary)
245 throw std::runtime_error("Background boundary triangulation changed.");
246 std::set<int64_t> found;
247 for (const auto& v : get_vertices()) {
248 const auto& a = metadata[v.vid(*this)];
249 if (!a.pinned) continue;
250 const auto it = source_pins.find(a.original);
251 if (it == source_pins.end() || it->second != m_vertex_attribute[v.vid(*this)].m_posf)
252 throw std::runtime_error("Pinned background point moved.");
253 found.insert(a.original);
254 }
255 if (found.size() != source_pins.size())
256 throw std::runtime_error("Pinned background point removed.");
257 }

275 for (auto& v : c) v = numbering[v];
276 cells.push_back(c);

280 forced->push_back({numbering.at(e.vertices()[0]), numbering.at(e.vertices()[1])});
281 }
