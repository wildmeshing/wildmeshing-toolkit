489 void init(const VectorXd& min_, const VectorXd& max_)
490 {
491 box_min = min_;
492 box_max = max_;
493
494 // Not a user knob: a topological offset preserves the topology of the region it wraps, so
495 // the shared collapse must always apply the substructure link condition, or a collapse
496 // across a thin band pinches the two sides together and the region stops being manifold.
497 // Set here because tetwild and simwild leave the flag off.
498 preserve_topology = true;
499
500 // Fills diag_l, l/lr and splitting_l2 / collapsing_l2. It also derives eps from epsr,
501 // which the offset never reads: its envelope tolerance is m_envelope_eps, set on the mesh.
502 init_lengths_from_diagonal((max_ - min_).norm());
503
504 if (target_distance > 0) {
505 target_distance_rel = target_distance / diag_l;
506 } else {
507 target_distance = target_distance_rel * diag_l;
508 }
509
510 // An ordinary relative length: it bounds how far a region boundary may drift in space, so
511 // the bounding box is the right reference.
512 if (envelope_size > 0) {
513 envelope_size_rel = envelope_size / diag_l;
514 } else {
515 envelope_size = envelope_size_rel * diag_l;
516 }
517
518 // The convergence epsilon, absolute or relative; relative to target_distance (resolved
519 // above), the offset it measures. A growth stage changes target_distance later, never
520 // this.
521 if (front_conv > 0) {
522 front_conv_rel = front_conv / target_distance;
523 } else {
524 front_conv = front_conv_rel * target_distance;
525 }
526
527 // An epsilon coarser than the offset it measures decides nothing: every front vertex is
528 // "placed" and every face "resolved" from the first turn, whatever the offset looks like.
529 // Checked on the resolved ABSOLUTE values, so it catches the mistake whichever of the two
530 // forms the config used to state it.
531 if (front_conv > target_distance) {
532 log_and_throw_error(
533 "front_conv {} must be <= target_distance {}: the convergence epsilon cannot be "
534 "coarser than the offset distance it measures, or every front face reads as "
535 "resolved from the first turn",
536 front_conv,
537 target_distance);
538 }
539
540 // The offset surface's envelope is envelope_size's (no key of its own since 2026-10-20);
541 // the two fields only for the 2D code, which still reads them.
542 offset_envelope = envelope_size;
543 offset_envelope_rel = envelope_size_rel;
544
545 // l_min is relative to the offset distance rather than the bounding box: it is the offset
546 // that has to be resolved. See the declaration.
547 if (min_edge_length_rel < 0) {
548 // The envelope eps expressed as a multiple of target_distance, which is what this
549 // wants: envelope_size, the offset surface's envelope too since offset_envelope went
550 // (2026-10-20), resolved to an absolute length above.
551 min_edge_length_rel = std::max(envelope_size / std::max(target_distance, 1e-16), 1e-12);
552 }
553 if (min_edge_length < 0) {
554 min_edge_length = min_edge_length_rel * target_distance;
555 } else {
556 min_edge_length_rel = min_edge_length / std::max(target_distance, 1e-16);
557 }
558 }
