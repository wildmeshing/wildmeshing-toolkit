24MeshData
25geometry_data(const std::vector<Vertex>& vertices, const std::vector<Cell>& cells, CellType type)
26{
27 MeshData data;
28 data.V = DataMatrix<double>::Zero(vertices.size(), 3);
29 for (size_t i = 0; i < vertices.size(); ++i)

38template <typename Point>
39MeshData feature_data(
40 const std::vector<std::array<Point, 2>>& segments,
41 const std::vector<Point>& anchors)
42{
43 std::map<std::array<double, 3>, int64_t> rows;
44 MeshData data;
45 std::vector<std::array<double, 3>> vertices;
46 auto vertex = [&](const Point& p) {
47 std::array<double, 3> key{{0, 0, 0}};
48 for (int i = 0; i < p.size(); ++i) key[i] = p[i];
49 auto [it, added] = rows.emplace(key, rows.size());
50 if (added) vertices.push_back(key);
51 return it->second;
52 };
53 auto& E = data.cells[CellType::Edge].connectivity;
54 E.resize(segments.size(), 2);
55 for (size_t i = 0; i < segments.size(); ++i) {
56 E(i, 0) = vertex(segments[i][0]);
57 E(i, 1) = vertex(segments[i][1]);
58 }
59 auto& P = data.cells[CellType::Point].connectivity;
60 P.resize(anchors.size(), 1);
61 for (size_t i = 0; i < anchors.size(); ++i) P(i, 0) = vertex(anchors[i]);
62 data.V.resize(vertices.size(), 3);
63 for (size_t i = 0; i < vertices.size(); ++i)
64 for (int j = 0; j < 3; ++j) data.V(i, j) = vertices[i][j];
65 data.validate();
66 return data;
67}
68// The envelope a component hands on -- outputs["envelope"] on the memory route,
69// <output>_envelope.msh on the file route -- is the REFERENCE it held its tags to: the geometry its
70// envelopes, tubes and anchors were built around (the input, or the simplified input, per
71// envelope_reference), labelled as its output mesh labels what it held there. In 3D a triangle
72// block (an input surface's face tag, a region's tet tag) and an edge block (the feature curves,
73// feature_<i>), in 2D an edge block, each with one `tags` column per label; the input points as a
74// point block without tags, since the mesh writes nothing for them. No eps: a receiver builds its
75// own envelopes around the reference at its own eps, label by label, so that every stage of a
76// chain holds its tags within one eps of the first component's input. Rebuilt from the mesh it
77// receives, the bound would grow by one eps per stage.
78//
79// append_envelope_block() adds one kind of it: the `type` cells of `block` (geometry_data() of
80// the reference's cells), on vertices of their own after the envelope's, row r labelled by the
81// bits of sets[r] (bit j: names[j]); without names, unlabelled. A block without cells adds
82// nothing. One block per type.
83inline void append_envelope_block(
84 MeshData& envelope,
85 const MeshData& block,
86 const CellType type,
87 const std::vector<InputSet>& sets,
88 const std::vector<std::string>& names)
89{
90 const auto& cells = block.cell(type).connectivity;
91 if (cells.rows() == 0) return;
92 if (envelope.cells.count(type))
93 throw std::runtime_error("An envelope has one block of each cell type");
94 if (!names.empty() && sets.size() != size_t(cells.rows()))
95 throw std::runtime_error("An envelope needs one label set per cell");
96 const Eigen::Index base = envelope.V.rows();
97 envelope.V.conservativeResize(base + block.V.rows(), 3);
98 envelope.V.bottomRows(block.V.rows()) = block.V;
99 auto& out = envelope.cells[type];
100 out.connectivity = (cells.array() + int64_t(base)).matrix();
101 if (names.empty()) return;
102 DataMatrix<uint8_t> member = DataMatrix<uint8_t>::Zero(cells.rows(), names.size());
103 for (Eigen::Index r = 0; r < cells.rows(); ++r)
104 for (size_t j = 0; j < names.size() && j < size_t(SurfaceTagAttributes::max_inputs); ++j)
105 member(r, j) = (sets[r] >> j) & 1;
106 out.attributes["tags"] = {std::move(member), names};
107}

162inline LabelledReference read_labelled_envelope(
163 const MeshData& envelope,
164 const int dim,
165 const std::function<InputSet(const std::string&)>& bits_named,
166 const std::vector<std::string>& feature_names)
167{
168 envelope.validate();
169 std::set<std::string> dropped = ComponentIO::dropped_fields(envelope.vertex_attributes, {});
170 const auto t = dim == 3 ? CellType::Triangle : CellType::Edge;
171 for (const auto& [ct, c] : envelope.cells) {
172 if (ct != t && ct != CellType::Point && !(dim == 3 && ct == CellType::Edge))
173 throw std::runtime_error("Incompatible envelope topology");
174 dropped.merge(ComponentIO::dropped_fields(c.attributes, {"tags"}));
175 }
176 ComponentIO::log_dropped("envelope", dropped);
177 const auto labelled = [](const CellData& cells) -> const MeshAttribute& {
178 const auto it = cells.attributes.find("tags");
179 if (it == cells.attributes.end())
180 throw std::runtime_error("A labelled envelope needs tags on each of its blocks");
181 const MeshAttribute& field = it->second;
182 if (!std::holds_alternative<DataMatrix<uint8_t>>(field.values) ||
183 field.labels.size() != size_t(field.cols()))
184 throw std::runtime_error("tags must be a UInt8 membership matrix with column labels");
185 if ((std::get<DataMatrix<uint8_t>>(field.values).array() > 1).any())
186 throw std::runtime_error("tags must be binary");
187 return field;
188 };
189 LabelledReference out;
190 // The rows of `cells` on vertices of their own, renumbered in order.
191 const auto own_vertices = [&](const DataMatrix<int64_t>& cells, MatrixXd& V, MatrixXi& C) {
192 std::vector<int> row_of(envelope.V.rows(), -1);
193 for (Eigen::Index i = 0; i < cells.size(); ++i) row_of[cells.data()[i]] = 0;
194 int n = 0;
195 for (auto& r : row_of)
196 if (r == 0) r = n++;
197 V.resize(n, dim);
198 for (Eigen::Index v = 0; v < envelope.V.rows(); ++v)
199 if (row_of[v] >= 0) V.row(row_of[v]) = envelope.V.row(v).head(dim);
200 C = ComponentIO::indices(cells);
201 for (Eigen::Index i = 0; i < C.size(); ++i) C.data()[i] = row_of[C.data()[i]];
202 };
203 const CellData& rows = envelope.cell(t);
204 const MeshAttribute& row_tags = labelled(rows);
205 const auto& member = std::get<DataMatrix<uint8_t>>(row_tags.values);
206 own_vertices(rows.connectivity, out.V, out.F);
207 out.F_labels.assign(out.F.rows(), 0);
208 for (size_t j = 0; j < row_tags.labels.size(); ++j) {
209 const InputSet bits = bits_named(row_tags.labels[j]);
210 for (Eigen::Index r = 0; r < member.rows(); ++r)
211 if (member(r, Eigen::Index(j))) out.F_labels[r] |= bits;
212 }
213 // The feature curves of the labels the receiver's feature curves have (feature_names), one
214 // input edge per edge and label, as a tagged mesh's own are read.
215 const auto curve_block = envelope.cells.find(CellType::Edge);
216 if (dim == 3 && curve_block != envelope.cells.end()) {
217 const MeshAttribute& curve_tags = labelled(curve_block->second);
218 const auto& on = std::get<DataMatrix<uint8_t>>(curve_tags.values);
219 const auto& E = curve_block->second.connectivity;
220 std::map<int64_t, size_t> vertex_of; // envelope row -> curves.edge_vertices
221 const auto vertex = [&](const int64_t v) {
222 const auto [it, added] = vertex_of.emplace(v, out.curves.edge_vertices.size());
223 if (added)
224 out.curves.edge_vertices.emplace_back(
225 envelope.V(v, 0),
226 envelope.V(v, 1),
227 envelope.V(v, 2));
228 return it->second;
229 };
230 for (size_t j = 0; j < curve_tags.labels.size(); ++j) {
231 const auto k =
232 std::find(feature_names.begin(), feature_names.end(), curve_tags.labels[j]);
233 if (k == feature_names.end()) continue;
234 for (Eigen::Index r = 0; r < E.rows(); ++r) {
235 if (!on(r, Eigen::Index(j))) continue;
236 out.curves.edges.push_back({{vertex(E(r, 0)), vertex(E(r, 1))}});
237 out.curves.edge_input.push_back(int(k - feature_names.begin()));
238 }
239 }
240 }
241 return out;
242}

259inline HeldRows held_rows(const MatrixXd& V, const MatrixXi& F, const std::vector<InputSet>& labels)
260{
261 HeldRows out;
262 std::vector<Eigen::Index> rows;
263 for (Eigen::Index r = 0; r < F.rows(); ++r) {
264 if (labels[r] != 0) rows.push_back(r);
265 }
266 if (rows.empty()) {
267 log_and_throw_error(
268 "envelope: none of its {} cells carries a label of this mesh",
269 F.rows());
270 }
271 std::vector<int> remap(V.rows(), -1);
272 for (const Eigen::Index r : rows)
273 for (Eigen::Index k = 0; k < F.cols(); ++k) remap[F(r, k)] = 0;
274 int n = 0;
275 for (int& v : remap)
276 if (v == 0) v = n++;
277 out.V.resize(n, V.cols());
278 for (Eigen::Index v = 0; v < V.rows(); ++v)
279 if (remap[v] >= 0) out.V.row(remap[v]) = V.row(v);
280 out.F.resize(rows.size(), F.cols());
281 for (size_t i = 0; i < rows.size(); ++i) {
282 for (Eigen::Index k = 0; k < F.cols(); ++k) out.F(i, k) = remap[F(rows[i], k)];
283 out.labels.push_back(labels[rows[i]]);
284 }
285 logger().info(
286 "envelope: a reference, {} of its {} cells of this mesh's labels held to",
287 rows.size(),
288 F.rows());
289 return out;
290}
291// The labels of tags 0 to count - 1, named by `names` (id -> name): a tag's name, or tag_<id>
292// without one, unless that name is taken, in which case '_' is appended as SimWildMesh::write_msh
293// does; labels must be unique.
294inline std::vector<std::string> tag_labels(
295 const std::map<int64_t, std::string>& names,
296 const size_t count)
297{
298 std::set<std::string> named, labelled;
299 for (const auto& [id, name] : names) named.insert(name);
300 std::vector<std::string> labels;
301 for (size_t j = 0; j < count; ++j) {
302 const auto it = names.find(int64_t(j));
303 const bool unnamed = it == names.end();
304 std::string label = unnamed ? "tag_" + std::to_string(j) : it->second;
305 while (labelled.count(label) || (unnamed && named.count(label))) label += "_";
306 labelled.insert(label);
307 labels.push_back(label);
308 }
309 return labels;
310}
311// The labels of an application's inputs, one per input: input_names, the rest tag_<i>, by the
312// rule of tag_labels(). What tetwild's face tags and triwild's edge tags name the inputs by, so
313// that an input's tags are named alike in every component.
314inline std::vector<std::string> input_labels(
315 const std::vector<std::string>& input_names,
316 const size_t n_inputs)
317{
318 std::map<int64_t, std::string> names;
319 for (size_t i = 0; i < input_names.size() && i < n_inputs; ++i)
320 names[int64_t(i)] = input_names[i];
321 return tag_labels(names, n_inputs);
322}
323template <typename Mesh, typename TupleRange, typename Tags>
324void collect_tags(Mesh& mesh, MeshMap& outputs, CellType type, const TupleRange& cells, Tags tags)
325{
326 // A result without cells has no block yet: the exporter only creates the ones it fills.
327 auto& block = outputs.at("mesh").cells[type];
328 if (block.connectivity.cols() != cell_size(type)) block.connectivity.resize(0, cell_size(type));
329 std::vector<std::string> labels = tag_labels(mesh.m_tag_id_to_name, mesh.m_tags_count);
330 DataMatrix<uint8_t> values = DataMatrix<uint8_t>::Zero(cells.size(), labels.size());
331 for (size_t i = 0; i < cells.size(); ++i)
332 for (size_t j = 0; j < labels.size(); ++j) values(i, j) = tags(cells[i]).count(j) ? 1 : 0;
333 // The fixed tags field replaces per-group named scalar arrays.
334 for (size_t j = 0; j < labels.size(); ++j) {
335 block.attributes.erase(labels[j]);
336 block.attributes.erase("tag_" + std::to_string(j));
337 }
338 // Cells without a tag are the background the component built. They are labelled
339 // explicitly: the IO infers nothing from an empty row.
340 if (std::find(labels.begin(), labels.end(), "ambient") == labels.end()) {
341 const auto empty = (values.template cast<int>().rowwise().sum().array() == 0).eval();
342 if (labels.empty() || empty.any()) {
343 values.conservativeResize(Eigen::NoChange, values.cols() + 1);
344 values.col(values.cols() - 1) = empty.template cast<uint8_t>();
345 labels.push_back("ambient");
346 }
347 }
348 block.attributes["tags"] = {std::move(values), std::move(labels)};
349}
350// The tracked simplices -- `type` Triangle: a tet mesh's faces; Edge: a triangle mesh's edges --
351// whose input set is not empty, as the `type` block of outputs["mesh"]: rows of its V (the
352// vertices in get_vertices() order, as compact_slot_outputs leaves them), one tags column per
353// name (bit j of the input set is names[j]), each row oriented as the simplex's input was.
354// Nothing is written without names.
355template <CellType type, typename Mesh>
356void collect_input_set_tags(
357 const Mesh& mesh,
358 MeshMap& outputs,
359 const std::vector<std::string>& names)
360{
361 static_assert(type == CellType::Triangle || type == CellType::Edge);
362 constexpr size_t N = type == CellType::Triangle ? 3 : 2;
363 if (names.empty()) return;
364 std::vector<int64_t> row(mesh.vert_capacity(), -1);
365 int64_t n = 0;
366 for (const auto& v : mesh.get_vertices()) row[v.vid(mesh)] = n++;
367 std::vector<std::array<int64_t, N>> simplices;
368 std::vector<InputSet> sets;
369 const InputSet mask = names.size() >= size_t(SurfaceTagAttributes::max_inputs)
370 ? ~InputSet(0)
371 : (InputSet(1) << names.size()) - 1;
372 const auto add = [&](const SurfaceTagAttributes& attr, std::array<size_t, N> vids) {
373 // Bits past the names are not these tags -- simwild's region labels -- and are not
374 // written.
375 const InputSet set = attr.input_set() & mask;
376 if (set == 0) return;
377 std::sort(vids.begin(), vids.end());
378 if (attr.orientation_along(vids) < 0) std::swap(vids[N - 2], vids[N - 1]);
379 std::array<int64_t, N> r;
380 for (size_t k = 0; k < N; ++k) r[k] = row[vids[k]];
381 simplices.push_back(r);
382 sets.push_back(set);
383 };
384 if constexpr (type == CellType::Triangle) {
385 for (const auto& f : mesh.get_faces())
386 add(mesh.m_face_attribute[f.fid(mesh)], mesh.get_face_vids(f));
387 } else {
388 for (const auto& e : mesh.get_edges())
389 add(mesh.m_edge_attribute[e.eid(mesh)], mesh.get_edge_vids(e));
390 }
391 auto& block = outputs.at("mesh").cells[type];
392 block.connectivity.resize(simplices.size(), N);
393 DataMatrix<uint8_t> member = DataMatrix<uint8_t>::Zero(simplices.size(), names.size());
394 for (size_t i = 0; i < simplices.size(); ++i) {
395 for (size_t k = 0; k < N; ++k) block.connectivity(i, k) = simplices[i][k];
396 for (size_t j = 0; j < names.size(); ++j) member(i, j) = (sets[i] >> j) & 1;
397 }
398 block.attributes["tags"] = {std::move(member), names};
399}
400// The tet edges tagged `feature` (TetOptimizerMesh::FeatureEdgeAttributes: the feature curves the
401// application was given) whose feature input set is not empty, as the Edge block of
402// outputs["mesh"], on the rows collect_input_set_tags() uses: one tags column per name (bit j of
403// the set is names[j], the input_edges file j). A curve carries no orientation, so a row is its
404// edge's vertices in ascending order. The order2 kind is not written: a reader derives it again
405// from the faces (tag_order2_edges()). Nothing is written without names or edge tracking.
406template <typename Mesh>
407void collect_feature_edge_tags(
408 const Mesh& mesh,
409 MeshMap& outputs,
410 const std::vector<std::string>& names)
411{
412 using EdgeTags = typename Mesh::FeatureEdgeAttributes;
413 if (names.empty() || !mesh.m_track_feature_edges) return;
414 std::vector<int64_t> row(mesh.vert_capacity(), -1);
415 int64_t n = 0;
416 for (const auto& v : mesh.get_vertices()) row[v.vid(mesh)] = n++;
417 const InputSet mask = names.size() >= size_t(SurfaceTagAttributes::max_inputs)
418 ? ~InputSet(0)
419 : (InputSet(1) << names.size()) - 1;
420 std::vector<std::array<int64_t, 2>> edges;
421 std::vector<InputSet> sets;
422 for (const auto& e : mesh.get_edges()) {
423 const auto& attr = mesh.m_feature_edge_attribute[e.eid(mesh)];
424 if (!attr.has(EdgeTags::feature)) continue;
425 const InputSet set = attr.inputs(EdgeTags::feature) & mask;
426 if (set == 0) continue;
427 const size_t a = e.vid(mesh), b = e.switch_vertex(mesh).vid(mesh);
428 edges.push_back({{row[std::min(a, b)], row[std::max(a, b)]}});
429 sets.push_back(set);
430 }
431 auto& block = outputs.at("mesh").cells[CellType::Edge];
432 block.connectivity.resize(edges.size(), 2);
433 DataMatrix<uint8_t> member = DataMatrix<uint8_t>::Zero(edges.size(), names.size());
434 for (size_t i = 0; i < edges.size(); ++i) {
435 for (size_t k = 0; k < 2; ++k) block.connectivity(i, k) = edges[i][k];
436 for (size_t j = 0; j < names.size(); ++j) member(i, j) = (sets[i] >> j) & 1;
437 }
438 block.attributes["tags"] = {std::move(member), names};
439}
