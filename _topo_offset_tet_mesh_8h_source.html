1132 std::shared_ptr<SampleEnvelope> surface_envelope_for_face(
1133 const std::array<size_t, 3>& vids) const override
1134 {
1135 uint64_t mask = face_mask(vids);
1136 bool all_offset = true;
1137 for (const size_t v : vids) {
1138 all_offset = all_offset && m_vertex_extra[v].m_is_on_offset;

1140 // The ambiguous case: all corners can be on region boundaries and on the offset surface
1141 // at once. The corner-mask AND is then necessary but not sufficient for the face lying
1142 // on a shared boundary; ask the face's own class, the only record that distinguishes a
