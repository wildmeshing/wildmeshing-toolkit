318 std::array<int, 2> e;
319 for (int j = 0; j < 4; j++) {
320 if (v1_id == m_indices[j])
321 e[0] = j;
322 else if (v2_id == m_indices[j])
323 e[1] = j;
324 }
325 if (e[0] > e[1]) std::swap(e[0], e[1]);
326 int i =

328 if (i >= m_local_edges.size()) return -1;

335 for (int j = 0; j < 4; j++) {
336 if (v1_id == m_indices[j])
337 f[0] = j;
338 else if (v2_id == m_indices[j])
339 f[1] = j;
340 else if (v3_id == m_indices[j])
341 f[2] = j;
342 }
343 std::sort(f.begin(), f.end());
344 int i =
345 std::find(m_local_faces.begin(), m_local_faces.end(), f) - m_local_faces.begin();

854 // ---- OPERATION ACCOUNTING (counting only) ----
855 // Every call into split_edge, collapse_edge, swap_edge, swap_edge_44, swap_edge_56 and
856 // swap_face reports `attempt` on entry and exactly ONE other event on the way out: `committed`,
857 // or the place that refused it. The application's hooks are two of those places (before_hook,
858 // after_hook); why a hook said no is the hook's own to count. So for each kind, attempt equals
859 // the sum of the other events, and a subclass that counts can check both that identity and a
860 // hook's refusals against the reasons the hook counted. Added because the refusals made here,
861 // below every hook, were visible only as a difference: on the deliverable cube at
862 // target_distance_rel 1e-2 / front_conv_rel 1e-4, 521142 of the 542123 edge swaps that
863 // passed every before-hook in turn 5 were refused inside this class with no reason recorded.
864 //
865 // The default does nothing and no site reads a count: the mesh behaves identically with or
866 // without an override.
867 enum class OpKind : int { split, collapse, swap_32, swap_44, swap_56, swap_face, COUNT };
868 enum class OpEvent : int {
869 attempt,
870 before_hook, // the application's *_before said no
871 link_condition, // collapse: the classical link condition (m_collapse_check_link_condition)
872 no_return_tet, // collapse: both probes for the return tuple hit a boundary face
873 duplicate_tet, // a new tet repeats an existing one (collapse manifold check, 4-4, 5-6)
874 topology, // collapse: m_collapse_check_topology found a face missing
875 valence, // edge swap: the edge does not have this swap's number of tets around it
876 boundary, // edge swap: the edge's ring of tets is open; face swap: a boundary face
877 exists, // 3-2: the new face already exists; face swap: the new edge already exists
878 no_case_allowed, // 4-4 / 5-6: every candidate was vetoed by *_accept_case
879 no_better_case, // 4-4 / 5-6: no allowed candidate scored below the current cells
880 out_of_slots, // the preallocated vertex or tet slots ran out
881 after_hook, // the application's *_after said no, so the operation was rolled back
882 invariants, // invariants() said no after the after-hook passed; rolled back
883 committed,
884 COUNT
885 };
