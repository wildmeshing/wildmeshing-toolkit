476 {
477 const size_t floor = 64;
478 // A mesh that grows on demand only needs room for the passes not to grow at once.
479 const double factor = storage_grows_on_demand() ? 1.25 : m_preallocation_factor;
480 double c = std::ceil(factor * static_cast<double>(live_count));

482 if (capacity < live_count) capacity = live_count;
483 return capacity < floor ? floor : capacity;
