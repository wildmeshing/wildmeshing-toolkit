4// The displacement/reference vectors at a newly refined node are convex
5// combinations of its parent's nodal vectors. Enforcing the ratio over the
6// full convex hull therefore survives later conforming subdivision.
7inline bool thickness_vectors_feasible(
8 const std::vector<V>& reference,
9 const std::vector<V>& actual,
10 double margin = 1e-6,
11 double lower = .8,
12 double upper_bound = 1.2)
13{
14 using L = long double;
15 using LV = Eigen::Matrix<L, 3, 1>;
16 using LM = Eigen::Matrix<L, Eigen::Dynamic, Eigen::Dynamic>;
17 const size_t n = reference.size();
18 if (n == 0) return true;
19 std::vector<LV> r(n), u(n);
20 for (size_t j = 0; j < n; ++j) {
21 r[j] = reference[j].cast<L>();
22 u[j] = actual[j].cast<L>();
23 }
24 for (bool upper : {false, true}) {
25 const L limit = upper ? L(upper_bound - margin) : L(lower + margin), factor = limit * limit;
26 auto valid = [&](const LV& a, const LV& b) {
27 const L rr = a.squaredNorm(), uu = b.squaredNorm(),
28 q = upper ? factor * rr - uu : uu - factor * rr;
29 return q >= -1e-12L * rr;
30 };
31 LM q(n, n);
32 for (size_t i = 0; i < n; ++i)
33 for (size_t j = 0; j < n; ++j)
34 q(i, j) = upper ? factor * r[i].dot(r[j]) - u[i].dot(u[j])
35 : u[i].dot(u[j]) - factor * r[i].dot(r[j]);
36 for (size_t i = 0; i < n; ++i)
37 if (!valid(r[i], u[i])) return false;
38 // All two-node faces: the inequality is quadratic along an edge.
39 for (size_t i = 0; i < n; ++i)
40 for (size_t j = i + 1; j < n; ++j) {
41 const L a = q(i, i) + q(j, j) - 2 * q(i, j), b = 2 * (q(i, j) - q(i, i));
42 if (a > 0) {
43 const L t = -b / (2 * a);
44 if (t > 0 && t < 1 &&
45 !valid((1 - t) * r[i] + t * r[j], (1 - t) * u[i] + t * u[j]))
46 return false;
47 }
48 }
49 // A quadratic reaches a minimum at a vertex or a stationary point on
50 // some active simplex face. Enumerate those (at most 63 for a prism).
51 // Singular stationary families attain the same value on a smaller face.
52 for (size_t mask = 1; mask < (size_t(1) << n); ++mask) {
53 std::vector<size_t> ids;
54 for (size_t i = 0; i < n; ++i)
55 if (mask & (size_t(1) << i)) ids.push_back(i);
56 const size_t k = ids.size();
57 if (k < 3) continue;
58 L scale = 0;
59 for (size_t i : ids)
60 for (size_t j : ids) scale = std::max(scale, std::abs(q(i, j)));
61 if (scale == 0) continue;
62 LM a = LM::Zero(k + 1, k + 1);
63 Eigen::Matrix<L, Eigen::Dynamic, 1> b =
64 Eigen::Matrix<L, Eigen::Dynamic, 1>::Zero(k + 1);
65 b[k] = 1;
66 for (size_t i = 0; i < k; ++i) {
67 for (size_t j = 0; j < k; ++j) a(i, j) = 2 * q(ids[i], ids[j]) / scale;
68 a(i, k) = -1;
69 a(k, i) = 1;
70 }
71 Eigen::FullPivLU<LM> lu(a);
72 if (!lu.isInvertible()) continue;
73 const auto x = lu.solve(b).eval();
74 bool inside = true;
75 L sum = 0;
76 for (size_t i = 0; i < k; ++i) {
77 inside &= x[i] >= -1e-12L;
78 sum += std::max(L(0), x[i]);
79 }
80 if (!inside || sum == 0) continue;
81 LV rr = LV::Zero(), uu = LV::Zero();
82 for (size_t i = 0; i < k; ++i) {
83 const L w = std::max(L(0), x[i]) / sum;
84 rr += w * r[ids[i]];
85 uu += w * u[ids[i]];
86 }
87 if (!valid(rr, uu)) return false;
88 }
89 }
90 return true;
91}
92inline bool thickness_cell_feasible(
93 const Mesh& mesh,
94 const Cell& cell,
95 double margin = 1e-6,
96 double lower = .8,
97 double upper = 1.2)
98{
99 std::vector<V> r, u;
100 for (size_t v : cell.v) {
101 if (!std::isfinite(mesh.thickness[v])) return true;
102 const V a = mesh.initial[v] - mesh.base[v], b = mesh.p[v] - mesh.base[v];
103 if (a.squaredNorm() == 0 && b.squaredNorm() == 0) continue;
104 r.push_back(a);
105 u.push_back(b);
106 }
107 return thickness_vectors_feasible(r, u, margin, lower, upper);
108}
