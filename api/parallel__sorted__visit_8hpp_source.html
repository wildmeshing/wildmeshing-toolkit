32template <class T, class Less, class KeyLess, class MakeVisitor>
33void parallel_sorted_visit(
34 std::vector<T>& v,
35 int num_threads,
36 Less less,
37 KeyLess key_less,
38 MakeVisitor&& make_visitor)
39{
40 const size_t n = v.size();
41 constexpr size_t kMinSlice = size_t(1) << 14;
42 const size_t parts =
43 std::min<size_t>(size_t(std::max(1, num_threads)), std::max<size_t>(1, n / kMinSlice));
44 if (parts <= 1) {
45 std::sort(v.begin(), v.end(), less);
46 auto visit = make_visitor();
47 for (const T& x : v) visit(x);
48 return;
49 }
50
51 // 1. Sort the slices.
52 std::vector<size_t> bound(parts + 1);
53 for (size_t s = 0; s <= parts; ++s) bound[s] = n * s / parts;
54 dynamic_parallel_for(parts, int(parts), 1, [&](size_t b, size_t e) {
55 for (size_t s = b; s < e; ++s) {
56 std::sort(v.begin() + bound[s], v.begin() + bound[s + 1], less);
57 }
58 });
59
60 // 2. parts - 1 splitters, from `parts` evenly spaced elements of every slice.
61 std::vector<T> sample;
62 sample.reserve(parts * parts);
63 for (size_t s = 0; s < parts; ++s) {
64 const size_t len = bound[s + 1] - bound[s];
65 for (size_t j = 0; j < parts; ++j) sample.push_back(v[bound[s] + len * j / parts]);
66 }
67 std::sort(sample.begin(), sample.end(), less);
68 std::vector<T> split;
69 split.reserve(parts - 1);
70 for (size_t r = 1; r < parts; ++r) split.push_back(sample[r * sample.size() / parts]);
71 std::vector<T>().swap(sample);
72
73 // 3. Where each part starts in each slice: part r holds the keys in [split[r-1], split[r]),
74 // so c[s * stride + r] is the first element of slice s not key_less than split[r-1]. The
75 // splitters ascend, so the cuts do too.
76 const size_t stride = parts + 1;
77 std::vector<size_t> c(parts * stride);
78 for (size_t s = 0; s < parts; ++s) {
79 c[s * stride] = bound[s];
80 c[s * stride + parts] = bound[s + 1];
81 for (size_t r = 1; r < parts; ++r) {
82 c[s * stride + r] = size_t(
83 std::lower_bound(
84 v.begin() + c[s * stride + r - 1],
85 v.begin() + bound[s + 1],
86 split[r - 1],
87 key_less) -
88 v.begin());
89 }
90 }
91
92 // 4. Each part: a parts-way merge of its pieces.
93 dynamic_parallel_for(parts, int(parts), 1, [&](size_t b, size_t e) {
94 using Head = std::pair<const T*, const T*>; // next, end
95 std::vector<Head> heads;
96 heads.reserve(parts);
97 const auto later = [&less](const Head& x, const Head& y) {
98 return less(*y.first, *x.first);
99 };
100 for (size_t r = b; r < e; ++r) {
101 auto visit = make_visitor();
102 heads.clear();
103 for (size_t s = 0; s < parts; ++s) {
104 const size_t lo = c[s * stride + r], hi = c[s * stride + r + 1];
105 if (lo < hi) heads.emplace_back(v.data() + lo, v.data() + hi);
106 }
107 std::make_heap(heads.begin(), heads.end(), later);
108 while (!heads.empty()) {
109 std::pop_heap(heads.begin(), heads.end(), later);
110 Head& h = heads.back();
111 visit(*h.first);
112 if (++h.first == h.second) {
113 heads.pop_back();
114 } else {
115 std::push_heap(heads.begin(), heads.end(), later);
116 }
117 }
118 }
119 });
120}
