39inline uint64_t edge_queue_key(size_t a, size_t b)
40{
41 if (a > b) std::swap(a, b);
42 if (b >> 32) {
43 log_and_throw_error("edge_queue_key: vertex id {} does not fit in 32 bits", b);
44 }
45 return (uint64_t(a) << 32) | uint64_t(b);
46}

366 bool operator()(AppMesh& m, const std::vector<std::pair<Op, Tuple>>& operation_tuples)

368 // The queue holds an operation's INDEX rather than its name. edit_operation_maps is a
369 // std::map, so iterating it yields the names in lexicographic order and an index is
370 // exactly a name's rank among them -- comparing indices is therefore identical to
371 // comparing the strings, which is what keeps the queue order, and so the output,
372 // bit-for-bit unchanged. What it buys is that a queue element is now trivially
373 // copyable: a heap sift moves 8 bytes instead of a std::string, and a tie on the
374 // priority is an integer compare instead of a string compare. It also turns the
375 // per-operation dispatch from a string-keyed std::map lookup into an index.
376 using OpId = uint32_t;
377 // priority, op index, tuple, #retries, queue_key. The key compares last, after the tuple,
378 // and two elements with the same tuple have the same key, so the pop order is unchanged.
379 using Elem = std::tuple<double, OpId, Tuple, size_t, uint64_t>;
380 // Each task owns its queue outright -- it is seeded before any thread starts, and the
381 // task both pops from it and pushes its renewed operations back into it -- so those need
382 // no lock. `final_queue` is the one that genuinely crosses threads: tasks push retry
383 // overflow into it while running, and it is drained after the barrier. Both are
384 // std::priority_queue with the same comparator underneath, so pop order is unchanged.
385 using LocalQueue = wmtk::threading::serial_priority_queue<Elem>;
386 using SharedQueue = wmtk::threading::concurrent_priority_queue<Elem>;
387
388 std::vector<const Op*> op_name;
389 std::vector<std::function<std::optional<std::vector<Tuple>>(AppMesh&, const Tuple&)>*>
390 op_fn;
391 std::map<Op, OpId> op_id;
392 op_name.reserve(edit_operation_maps.size());
393 op_fn.reserve(edit_operation_maps.size());
394 for (auto& kv : edit_operation_maps) {
395 op_id.emplace(kv.first, OpId(op_name.size()));
396 op_name.push_back(&kv.first);
397 op_fn.push_back(&kv.second);
398 }
399 // An operation with no entry here was previously default-constructed into the map by
400 // operator[] and then called, which throws bad_function_call -- so it cannot occur in
401 // any working configuration. Say so plainly rather than ordering it arbitrarily.
402 const auto id_of = [&op_id](const Op& name) {
403 const auto it = op_id.find(name);
404 if (it == op_id.end()) {
405 log_and_throw_error("No operation registered under the name '{}'.", name);
406 }
407 return it->second;
408 };

410 std::atomic<bool> stop(false);
411 cnt_success = 0;
412 cnt_fail = 0;

414 // queue_key bookkeeping (see must_wait). `track` when an element is queued, `done` when
415 // it is tried or dropped; both no-ops for the untracked key 0.
416 m_queued.clear();
417 std::atomic<size_t> untracked_done(0);
418 const auto key_of = [&](const Op& op, const Tuple& e) -> uint64_t {
419 return queue_key ? queue_key(m, op, e) : 0;
420 };
421 const auto track = [&](const uint64_t key) {
422 if (key != 0) m_queued.add(key);
423 };
424 const auto done = [&](const uint64_t key) {
425 if (key != 0 && !m_queued.remove(key)) {
426 untracked_done.fetch_add(1, std::memory_order_relaxed);
427 }
428 };

430 // Whether anything actually watches the success count *while the pass runs*. When no
431 // stopping criterion is configured -- the case for every tetwild/triwild/simwild pass --
432 // nobody does, and the counters can be accumulated per task and folded in at the end
433 // instead of hammering one shared cache line from every thread on every operation.
434 const bool track_live_success =
435 stopping_criterion_checking_frequency != std::numeric_limits<size_t>::max();
436 std::atomic<size_t> live_success(0);

438 std::vector<LocalQueue> queues(num_threads);
439 SharedQueue final_queue;

441 // Contention accounting. Everything here is either a per-task local folded in once or a
442 // write to the task's own slot, so it adds nothing to the inner loop. It answers the
443 // two questions the pass could not previously be asked: how often ring acquisition
444 // loses a race, and how much of a "parallel" pass is really the serial drain.
445 m_stats = PassStats{};
446 std::atomic<size_t> lock_failures(0);
447 std::atomic<size_t> overflowed(0);
448 std::vector<double> task_seconds(queues.size(), 0.);

450 // Per-task tallies folded into the shared counters once, on the way out. The guard is
451 // RAII rather than a line at the bottom because run_single_queue has early returns.
452 //
453 // DECLARED HERE, NOT INSIDE THE LAMBDA, to work around an Apple clang 17 codegen bug.
454 // A local class declared inside a GENERIC lambda (one with an `auto` parameter, so its
455 // operator() is a template) gets its destructor emitted as an undefined reference and
456 // never defined, at every optimization level including -O0. The link then fails with
457 // Undefined symbols: ... ::'lambda'(auto&, int)::operator()<...>
458 // ::CountFlusher::~CountFlusher()
459 // referenced from every translation unit that instantiates a pass -- 26 references
460 // across 6 archives, defined nowhere. Homebrew clang 22 compiles the same code
461 // correctly, which is why CI does not see this.
462 //
463 // Hoisting the class one scope out is enough: it is still local to operator(), which is
464 // itself a member of a class template, and that instantiates fine. Nothing else changes
465 // -- `counts` is still constructed per task inside the lambda, so the RAII flush and its
466 // ordering are identical.
467 struct CountFlusher
468 {
469 std::atomic_int& success_total;
470 std::atomic_int& fail_total;
471 std::atomic<size_t>& lock_failure_total;
472 std::atomic<size_t>& overflow_total;
473 int success = 0;
474 int fail = 0;
475 size_t lock_failure = 0;
476 size_t overflow = 0;
477 ~CountFlusher()
478 {
479 success_total.fetch_add(success, std::memory_order_relaxed);
480 fail_total.fetch_add(fail, std::memory_order_relaxed);
481 lock_failure_total.fetch_add(lock_failure, std::memory_order_relaxed);
482 overflow_total.fetch_add(overflow, std::memory_order_relaxed);
483 }
484 };

486 // `serial`: Q is popped by this task alone and in priority order (kSeq, or the queue
487 // drained after the barrier), so a must_wait that says true is a defect there.
488 auto run_single_queue = [&](auto& Q, int task_id, const bool serial) {
489 CountFlusher counts{cnt_success, cnt_fail, lock_failures, overflowed};
490
491 Elem ele_in_queue;
492 // Operations that lost a race for their ring wait here rather than going straight
493 // back into the live queue. Pushing them back immediately is a spin: the element was
494 // just popped as the queue's maximum, `retry` is the last tie-break key in Elem, so
495 // the requeued copy compares strictly greater than what was popped and nothing else
496 // was added -- it comes right back off the top and is retried against a conflict that
497 // has had no time to clear. Deferring lets every other operation this task owns run
498 // first, which is both useful work and the delay the conflict needs.
499 std::vector<Elem> second_chance;
500 // Operations popped since the deferred list was last given back to the queue.
501 // Waiting for the queue to drain completely can mean thousands of operations, by
502 // which time the mesh around a deferred operation has moved and its work has to be
503 // rediscovered. A bounded window gives the conflict time to clear without letting
504 // the operation go stale. 0 = only when the queue empties.
505 size_t since_refill = 0;
506 const auto refill = [&] {
507 for (auto& e : second_chance) {
508 Q.emplace(std::move(e));
509 }
510 second_chance.clear();
511 since_refill = 0;
512 };
513 for (;;) {
514 if (!second_chance.empty() && deferral_window > 0 &&
515 since_refill >= deferral_window) {
516 refill();
517 }
518 if (!Q.try_pop(ele_in_queue)) {
519 if (second_chance.empty()) {
520 break;
521 }
522 // Queue exhausted: the deferred operations have now had everything else
523 // run ahead of them, so give them another go. `retry` still increments on
524 // each attempt and still overflows to final_queue at max_retry_limit, so
525 // this terminates after at most that many rounds.
526 refill();
527 std::this_thread::yield();
528 continue;
529 }
530 ++since_refill;
531 auto& [weight, op, tup, retry, key] = ele_in_queue;
532 if (!tup.is_valid(m)) {
533 done(key);
534 continue;
535 }
536
537 std::vector<Elem> renewed_elements;
538 {
539 auto locked_vid = lock_vertices(
540 m,
541 tup,
542 task_id); // Note that returning `Tuples` would be invalid.
543 if (!locked_vid) {
544 counts.lock_failure++;
545 retry++;
546 if (retry < max_retry_limit) {
547 second_chance.push_back(ele_in_queue);
548 } else {
549 retry = 0;
550 counts.overflow++;
551 final_queue.emplace(ele_in_queue);
552 }
553 continue;
554 }
555 if (tup.is_valid(m)) {
556 const Op& op_str = *op_name[op];
557 if (!is_weight_up_to_date(
558 m,
559 std::tuple<double, Op, Tuple>(weight, op_str, tup))) {
560 done(key);
561 operation_cleanup(m);
562 continue;
563 } // this can encode, in qslim, recompute(energy) == weight.
564 if (must_wait && must_wait(m, op_str, tup)) {
565 if (serial) {
566 m_wait_defects.fetch_add(1, std::memory_order_relaxed);
567 done(key);
568 on_fail(m, op_str, tup);
569 counts.fail++;
570 operation_cleanup(m);
571 continue;
572 }
573 // Set aside exactly like a lost lock race, above.
574 operation_cleanup(m);
575 m_waits.fetch_add(1, std::memory_order_relaxed);
576 retry++;
577 if (retry < max_retry_limit) {
578 second_chance.push_back(ele_in_queue);
579 } else {
580 retry = 0;
581 counts.overflow++;
582 final_queue.emplace(ele_in_queue);
583 }
584 continue;
585 }
586 auto newtup = (*op_fn[op])(m, tup);
587 done(key);
588 std::vector<std::pair<Op, Tuple>> renewed_tuples;
589 if (newtup) {
590 renewed_tuples = renew_neighbor_tuples(m, op_str, newtup.value());
591 counts.success++;
592 if (track_live_success) {
593 live_success.fetch_add(1, std::memory_order_relaxed);
594 }
595 } else {
596 on_fail(m, op_str, tup);
597 counts.fail++;
598 }
599 // Tracked here, with the ring still locked, although the elements are
600 // queued only after it is released: in between they count as queued.
601 for (const auto& [o, e] : renewed_tuples) {
602 auto val = priority(m, o, e);
603 if (should_renew(val)) {
604 const uint64_t k = key_of(o, e);
605 track(k);
606 renewed_elements.emplace_back(val, id_of(o), e, 0, k);
607 }
608 }
609 } else {
610 done(key);
611 }
612 operation_cleanup(m); // Maybe use RAII
613 }
614 for (auto& e : renewed_elements) {
615 Q.emplace(e);
616 }
617
618 if (stop.load(std::memory_order_acquire)) {
619 return;
620 }
621 if (track_live_success && live_success.load(std::memory_order_relaxed) >
622 stopping_criterion_checking_frequency) {
623 if (stopping_criterion(m)) {
624 stop.store(true);
625 return;
626 }
627 }
628 }
629 };
630
631 if (policy == ExecutionPolicy::kSeq) {
632 for (const auto& [op, e] : operation_tuples) {
633 if (!e.is_valid(m)) {
634 continue;
635 }
636 const uint64_t k = key_of(op, e);
637 track(k);
638 final_queue.emplace(priority(m, op, e), id_of(op), e, 0, k);
639 }
640 run_single_queue(final_queue, 0, true);
641 } else {
642 for (const auto& [op, e] : operation_tuples) {
643 if (!e.is_valid(m)) {
644 continue;
645 }
646 const uint64_t k = key_of(op, e);
647 track(k);
648 queues[get_partition_id(m, e)].emplace(priority(m, op, e), id_of(op), e, 0, k);
649 }
650 // Comment out parallel: work on serial first.
651 using clock = std::chrono::steady_clock;
652 const auto t_parallel = clock::now();
653 wmtk::threading::task_group tg;
654 for (int task_id = 0; task_id < queues.size(); task_id++) {
655 tg.run([&run_single_queue, &queues, &task_seconds, task_id] {
656 const auto t0 = clock::now();
657 run_single_queue(queues[task_id], task_id, false);
658 // Each task writes only its own slot.
659 task_seconds[task_id] =
660 std::chrono::duration<double>(clock::now() - t0).count();
661 });
662 }
663 tg.wait();
664 m_stats.parallel_seconds =
665 std::chrono::duration<double>(clock::now() - t_parallel).count();
666 m_stats.final_queue_size = final_queue.size();
667
668 logger().debug("Parallel Complete, remains element {}", final_queue.size());
669
670 const auto t_tail = clock::now();
671 run_single_queue(final_queue, 0, true);
672 m_stats.serial_tail_seconds =
673 std::chrono::duration<double>(clock::now() - t_tail).count();
674 }
675
676 m_stats.lock_failures = lock_failures.load(std::memory_order_relaxed);
677 m_stats.overflowed = overflowed.load(std::memory_order_relaxed);
678 if (!task_seconds.empty()) {
679 const auto mm = std::minmax_element(task_seconds.begin(), task_seconds.end());
680 m_stats.idlest_task_seconds = *mm.first;
681 m_stats.busiest_task_seconds = *mm.second;
682 }
683
684 logger().info(
685 "executed: {} | success / fail: {} / {}",
686 (int)cnt_success + (int)cnt_fail,
687 (int)cnt_success,
688 (int)cnt_fail);
689 log_contention();
690 // Every tracked element queued in this call was tried or dropped, unless the stopping
691 // criterion ended the call early. Anything else is a bookkeeping defect, and is_queued()
692 // may have answered wrongly during the call.
693 if (queue_key && !stop.load()) {
694 const size_t left = m_queued.size();
695 const size_t untracked = untracked_done.load();
696 if (left > 0 || untracked > 0) {
697 logger().warn(
698 "queue_key bookkeeping: {} keys still counted as queued after the pass, {} "
699 "finished elements were never counted",
700 left,
701 untracked);
702 }
703 }
704 m_queued.clear();
705 return true;
706 }

740 void log_contention() const
741 {
742 if (policy == ExecutionPolicy::kSeq || !logger().should_log(spdlog::level::debug)) {
743 return;
744 }
745 const int executed = (int)cnt_success + (int)cnt_fail;
746 const double total = m_stats.parallel_seconds + m_stats.serial_tail_seconds;
747 logger().debug(
748 " contention: {} ring-acquisition failures over {} executed ops ({:.2f} per op); "
749 "{} overflowed to the serial queue ({} queued at the barrier)",
750 m_stats.lock_failures,
751 executed,
752 executed > 0 ? double(m_stats.lock_failures) / executed : 0.,
753 m_stats.overflowed,
754 m_stats.final_queue_size);
755 logger().debug(
756 " time: {:.4}s parallel + {:.4}s serial tail ({:.1f}% of the pass); busiest task "
757 "{:.4}s, idlest {:.4}s",
758 m_stats.parallel_seconds,
759 m_stats.serial_tail_seconds,
760 total > 0. ? 100. * m_stats.serial_tail_seconds / total : 0.,
761 m_stats.busiest_task_seconds,
762 m_stats.idlest_task_seconds);
763 }
