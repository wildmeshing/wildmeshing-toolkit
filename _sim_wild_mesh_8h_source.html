176 // only used with unit tests
177 void create_mesh_attributes(
178 const std::vector<VertexAttributes>& _vertex_attribute,
179 const std::vector<TetAttributes>& _tet_attribute)
180 {
181 const size_t n_tet = _tet_attribute.size();
182 m_vertex_attribute.resize(_vertex_attribute.size());
183 m_face_attribute.resize(4 * n_tet);
184 m_tet_attribute.resize(n_tet);
185
186 for (size_t i = 0; i < _vertex_attribute.size(); i++)
187 m_vertex_attribute[i] = _vertex_attribute[i];
188
189 // Keep whatever init() reserved. AttributeCollection::resize only ever grows, so the
190 // calls above cannot shrink a collection -- but assigning m_attributes directly does,
191 // and it used to drop the tet attributes to exactly n_tet. The attributes have to stay
192 // at least as large as the connectivity: an operation that creates a new element
193 // indexes them by its id, and a 5->6 swap creates one. tetwild hit the same thing (see
194 // the note on its copy); there the overrun wrote a double past the end and libc++ let
195 // it pass, here it assigns a std::set and segfaults outright.
196 const size_t tcap = std::max(n_tet, m_tet_attribute.size());
197 m_tet_attribute.m_attributes = std::vector<TetAttributes>(tcap);
198 for (size_t i = 0; i < n_tet; i++) m_tet_attribute[i] = _tet_attribute[i];
199 for (size_t i = 0; i < n_tet; i++)
200 m_tet_attribute[i].m_quality = get_quality(tuple_from_tet(i));
201 }
