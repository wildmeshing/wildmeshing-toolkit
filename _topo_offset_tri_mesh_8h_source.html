1195 std::shared_ptr<SampleEnvelope> surface_envelope_for_edge(
1196 const std::array<size_t, 2>& vids) const override
1197 {
1198 // Boundary geometry first, in both phases: a segment on any tag-region boundary may not
1199 // drift out of that boundary's tube, and one on several boundaries -- a junction -- is
1200 // held in their intersection. The mask carries the input complex too, since every complex
1201 // simplex lies on tag boundaries, and E_t is built from the input mesh before construction
1202 // touches it, so the per-tag tubes hold the as-loaded geometry.
1203 //
1204 // Keyed on the vertices, not the edge: every caller is an operation asking about a segment
1205 // it is about to create or has just created, whose own edge attributes are not written
1206 // yet. The endpoints' masks are, maintained by the operations themselves (AND at a split,
1207 // OR at a collapse).
1208 uint64_t mask = edge_mask(vids);
1209 bool all_offset = true;
1210 for (const size_t v : vids) {
1211 all_offset = all_offset && m_vertex_extra[v].m_is_on_offset;
1212 }
1213
1214 // The ambiguous case: both endpoints can be on region boundaries and on the offset front
1215 // at once -- a real state, and the point of tracking the two families separately. The
1216 // endpoint-mask AND is then necessary but not sufficient for the segment lying on a shared
1217 // boundary: two vertices on different junctions can share a tag bit by coincidence and be
1218 // joined by an offset chord nowhere near that tag's curve, which a container it was never
1219 // meant to satisfy then refuses to refine. Ask the edge's own class, the only record that
1220 // distinguishes a chord from a boundary.
1221 //
1222 // Reading the slot is safe here and would not be unconditionally: the base checks a
1223 // split's two child segments before writing their attributes, so their slots still hold a
1224 // recycled edge's class. A split child never reaches this branch -- its mask and its
1225 // offset flag are mutually exclusive by construction, so one of `mask` and `all_offset` is
1226 // always empty -- and the m_is_surface_fs guard leaves both constraints standing rather
1227 // than dropping one when a slot is illegible.
1228 if (mask != 0 && all_offset) {
1229 if (const auto found = try_tuple_from_edge(vids)) {
1230 const size_t eid = std::get<1>(*found);
1231 if (m_edge_attribute[eid].m_is_surface_fs) {
1232 if (edge_is_offset(eid)) {
1233 mask = 0; // an offset edge lies on no region boundary
1234 } else {
1235 all_offset = false; // a region edge is not the offset front
1236 }
1237 }
1238 }
1239 }
1240 // Both families compose: whatever holds this segment holds it at once. Phase A holds the
1241 // offset where Phase B left it; Phase B is what moves it, so it contributes nothing there
1242 // -- and null before the offset exists at all.
1243 const std::shared_ptr<SampleEnvelope> base = containment_for(mask, all_offset);
1244 if (base || m_deform_tags.empty()) return base;
1245 // deform_others' ops-only tube: a released boundary is held by no mask -- its vertices
1246 // were freed so smoothing can carry the object -- which would leave the operations free
1247 // to decimate and reposition it. A segment the masks and the offset class do not claim,
1248 // but which lies on a released boundary by its incident faces' current tags, is held to
1249 // the tube around the boundary's current shape. A fresh split child can misread its
1250 // recycled face slots for this one check, and is at worst skipped or over-held once.
1251 if (const auto found = try_tuple_from_edge(vids)) {
1252 if (edge_borders_released_boundary(std::get<0>(*found))) return released_envelope();
1253 }
1254 return nullptr;
1255 }
