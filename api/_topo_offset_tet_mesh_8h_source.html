1104 double band_cell_term(const std::array<size_t, 4>& vids) const
1105 {
1106 if (m_n_regions > 1 && !m_offset_params.allow_mult_comp) {
1107 log_and_throw_error(
1108 "the energy's band term reads one field, and this input has {} regions",
1109 m_n_regions);
1110 }

1112 m_offset_params.band_volume_rule == "front_flux" || swept_rule()) {
1113 // EXPERIMENTAL_band_volume_rule "exact" (and "front_flux", whose per-cell quantity
1114 // it is, see above): J_t / delta, J_t = int_t (d - delta) dV in closed form
1115 // (ExactBandIntegral), in place of the corner bound
1116 const std::array<Vector3d, 4> p = {

1118 m_vertex_attribute[vids[1]].m_posf,

1122 return m_band_exact->value(p, delta) / delta;
1123 }

1126 }
1128 double band_cell_corner_bound(const std::array<size_t, 4>& vids, int64_t& best) const
1129 {

1135 min_mean_distance(*m_input_complex_envelope, m_input_complex_pad, p.data(), 4, best);
1136 return vol * ((md - delta) / delta);

2455 // raw sum of AMIPS over the one-ring (AMIPSEnergy3D), as TetWild's does.
2456 if (tetwild_rules()) return nullptr;

2471 // far off; the projected step then discards the normal motion anyway. MEASURED
