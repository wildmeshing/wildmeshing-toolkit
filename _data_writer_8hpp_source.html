58 bool write_mesh(
59 const std::string& path,
60 const Eigen::MatrixXd& V,
61 const std::vector<paraviewo::CellElement>& cells)
62 {
63 if (!output) return file.write_mesh(path, V, cells);
64 MeshData data;
65 data.V = DataMatrix<double>::Zero(V.rows(), 3);
66 data.V.leftCols(V.cols()) = V;
67 data.vertex_attributes = points;
68 std::map<CellType, std::vector<size_t>> rows;
69 for (size_t i = 0; i < cells.size(); ++i) {
70 const auto& c = cells[i];
71 const auto type = static_cast<CellType>(
72 paraviewo::paraview_tags::VTKTag(int(c.vertices.size()), c.ctype));
73 auto& block = data.cells[type].connectivity;
74 block.conservativeResize(block.rows() + 1, cell_size(type));
75 for (size_t j = 0; j < c.vertices.size(); ++j)
76 block(block.rows() - 1, j) = c.vertices[j];
77 rows[type].push_back(i);
78 }
79 for (const auto& [type, indices] : rows)
80 for (const auto& [name, a] : fields) {
81 auto values = a.doubles();
82 DataMatrix<double> sub(indices.size(), values.cols());
83 for (size_t r = 0; r < indices.size(); ++r) sub.row(r) = values.row(indices[r]);
84 data.cells[type].attributes[name].values = std::move(sub);
85 }
86 data.validate();
87 output->insert_or_assign(path, std::move(data));
88 // Like VTUWriter::write_mesh, which clears its fields after every write.
89 points.clear();
90 fields.clear();
91 return true;
92 }

94// Export arrays are sometimes indexed by reserved slots. Remove dead slots without
95// consolidating or otherwise mutating the algorithm's mesh, and keep every field aligned.
96inline void compact_vertices(MeshData& data, const std::vector<size_t>& rows)
97{
98 DataMatrix<double> V(rows.size(), 3);
99 std::vector<int64_t> remap(data.V.rows(), -1);
100 for (size_t i = 0; i < rows.size(); ++i) {
101 V.row(i) = data.V.row(rows[i]);
102 remap[rows[i]] = i;
103 }
104 data.V = std::move(V);
105 for (auto& [name, a] : data.vertex_attributes)
106 a.values = std::visit(
107 [&](const auto& values) -> AttributeValues {
108 using M = std::decay_t<decltype(values)>;
109 M compact(rows.size(), values.cols());
110 for (size_t i = 0; i < rows.size(); ++i) compact.row(i) = values.row(rows[i]);
111 return compact;
112 },
113 a.values);
114 for (auto& [type, block] : data.cells)
115 for (Eigen::Index i = 0; i < block.connectivity.size(); ++i) {
116 const auto v = remap.at(block.connectivity.data()[i]);
117 if (v < 0) throw std::runtime_error("Export references an inactive vertex");
118 block.connectivity.data()[i] = v;
119 }
120 data.validate();
121}
122template <typename Mesh>
123void compact_slot_outputs(const Mesh& mesh, MeshMap& outputs)
124{
125 std::vector<size_t> rows;
126 for (const auto& v : mesh.get_vertices()) rows.push_back(v.vid(mesh));
127 for (auto& [name, data] : outputs) compact_vertices(data, rows);
128}
129// The result key of each memory export. The exporters keep their legacy file naming,

133inline OutputKeys default_output_keys()

135 return {
136 {"mesh.vtu", "mesh"},
137 {"mesh_surf.vtu", "surface"},

143// Rename the exports into public result keys. An export missing from `keys`, or two exports
144// with the same key, is an error rather than a silently dropped output.
145inline MeshMap named_outputs(MeshMap raw, const OutputKeys& keys = default_output_keys())
146{
147 MeshMap result;
148 for (auto& [path, data] : raw) {
149 const auto key = keys.find(path);
150 if (key == keys.end()) throw std::runtime_error("Unexpected memory export: " + path);
151 if (key->second.empty()) continue;
152 if (!result.emplace(key->second, std::move(data)).second)
153 throw std::runtime_error("Two memory exports map to the result key " + key->second);
154 }
155 return result;
156}
