10inline double triangle_shape(const Mesh& m, const Tri& f)
11{
12 const V a = m.p[f[1]] - m.p[f[0]], b = m.p[f[2]] - m.p[f[0]];
13 const double s = a.squaredNorm() + b.squaredNorm() + (a - b).squaredNorm();

29 void expose(const Key& k, const Tri& f, bool add)
30 {
31 if (add)
32 triangles[k] = f;
33 else
34 triangles.erase(k);
35 for (int j = 0; j < 3; ++j) {
36 const auto e = native::edge(f[j], f[(j + 1) % 3]);
37 if (add) {
38 auto& fs = edges[e];
39 if (fs.empty()) {
40 neighbors[e[0]].insert(e[1]);
41 neighbors[e[1]].insert(e[0]);
42 }
43 fs.insert(k);
44 } else {
45 auto it = edges.find(e);
46 if (it == edges.end() || !it->second.erase(k))
47 throw std::runtime_error("boundary_index_edge_missing");
48 if (it->second.empty()) {
49 edges.erase(it);
50 neighbors[e[0]].erase(e[1]);
51 neighbors[e[1]].erase(e[0]);
52 }
53 }
54 }

56 void change(const Cell& c, bool add)
57 {
58 if (c.type != 4 || c.region != 2) return;
59 for (const auto& local : native::faces.at(4)) {
60 std::vector<size_t> v;
61 for (int j : local) v.push_back(c.v[j]);
62 const auto f = oriented_tri(v);
63 const auto k = native::facekey(v);
64 auto& all = contributions[k];
65 if (all.size() == 1) expose(k, *all.begin(), false);
66 if (add) {
67 if (!all.insert(f).second) throw std::runtime_error("duplicate_shell_tet_face");
68 } else if (!all.erase(f))
69 throw std::runtime_error("boundary_index_face_missing");
70 if (all.size() > 2) throw std::runtime_error("nonmanifold_shell_tet_face");
71 if (all.size() == 1) expose(k, *all.begin(), true);
72 if (all.empty()) contributions.erase(k);
73 }
74 }

128inline SurfaceDecision surface_decision(
129 const Mesh& m,
130 const BoundarySurface* surface,
131 Edge edge,
132 Edge newedge,
133 const std::vector<Tri>& old,
134 const std::vector<Tri>& next)
135{
136 SurfaceDecision d;
137 for (const auto& f : old) {
138 const auto q = triangle_shape(m, f);
139 d.before_min = std::min(d.before_min, q);
140 d.before_mean += q / old.size();
141 }
142 for (const auto& f : next) {
143 const auto q = triangle_shape(m, f);
144 d.after_min = std::min(d.after_min, q);
145 d.after_mean += q / next.size();
146 }
147 if (surface) {
148 for (size_t v : edge) {
149 const double x = double(surface->neighbors.at(v).size()) - 6;
150 d.before_degree += x * x;
151 d.after_degree += (x - 1) * (x - 1);
152 }
153 for (size_t v : newedge) {
154 const double x = double(surface->neighbors.at(v).size()) - 6;
155 d.before_degree += x * x;
156 d.after_degree += (x + 1) * (x + 1);
157 }
158 }
159 const double eps = 1e-10;
160 d.accepted = d.after_min > d.before_min + eps ||
161 (d.after_min >= d.before_min - eps &&
162 (d.after_mean > d.before_mean + eps || (d.after_mean >= d.before_mean - eps &&
163 d.after_degree < d.before_degree - eps)));
164 return d;
165}
