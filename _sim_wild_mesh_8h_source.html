221 void compute_vertex_partition_morton()
222 {
223 if (NUM_THREADS == 0) {
224 return;
225 }
226
227 logger().info("Number of parts: {} by morton", NUM_THREADS);
228
229 std::vector<Eigen::Vector3d> V_v(vert_capacity());
230
231 threading::parallel_for(
232 threading::range(0, V_v.size()),
233 [&](const threading::range& r) {
234 for (size_t i = r.begin(); i < r.end(); i++) {
235 V_v[i] = m_vertex_attribute[i].m_posf;
236 }
237 },
238 NUM_THREADS);
239
240
241 struct sortstruct
242 {
243 int order;
244 Resorting::MortonCode64 morton;
245 };
246
247 std::vector<sortstruct> list_v;
248 list_v.resize(V_v.size());
249 // since the morton code requires a correct scale of input vertices,
250 // we need to scale the vertices if their coordinates are out of range
251 std::vector<Eigen::Vector3d> V = V_v; // this is for rescaling vertices
252 Eigen::Vector3d vmin, vmax;
253 vmin = V.front();
254 vmax = V.front();
255
256 for (size_t j = 0; j < V.size(); j++) {
257 for (int i = 0; i < 3; i++) {
258 vmin(i) = std::min(vmin(i), V[j](i));
259 vmax(i) = std::max(vmax(i), V[j](i));
260 }
261 }
262
263 // get_bb_corners(V, vmin, vmax);
264 Eigen::Vector3d center = (vmin + vmax) / 2;
265
266 threading::parallel_for(
267 threading::range(0, V.size()),
268 [&](const threading::range& r) {
269 for (size_t i = r.begin(); i < r.end(); i++) {
270 V[i] = V[i] - center;
271 }
272 },
273 NUM_THREADS);
274
275 Eigen::Vector3d scale_point =
276 vmax - center; // after placing box at origin, vmax and vmin are symetric.
277
278 double xscale, yscale, zscale;
279 xscale = fabs(scale_point[0]);
280 yscale = fabs(scale_point[1]);
281 zscale = fabs(scale_point[2]);
282 double scale = std::max(std::max(xscale, yscale), zscale);
283 if (scale > 300) {
284 threading::parallel_for(
285 threading::range(0, V.size()),
286 [&](const threading::range& r) {
287 for (size_t i = r.begin(); i < r.end(); i++) {
288 V[i] = V[i] / scale;
289 }
290 },
291 NUM_THREADS);
292 }
293
294 constexpr int multi = 1000;
295 threading::parallel_for(
296 threading::range(0, V.size()),
297 [&](const threading::range& r) {
298 for (size_t i = r.begin(); i < r.end(); i++) {
299 list_v[i].morton = Resorting::MortonCode64(
300 int(V[i][0] * multi),
301 int(V[i][1] * multi),
302 int(V[i][2] * multi));
303 list_v[i].order = i;
304 }
305 },
306 NUM_THREADS);
307
308 const auto morton_compare = [](const sortstruct& a, const sortstruct& b) {
309 return (a.morton < b.morton);
310 };
311
312 std::sort(list_v.begin(), list_v.end(), morton_compare);
313
314 int interval = list_v.size() / NUM_THREADS + 1;
315
316 threading::parallel_for(
317 threading::range(0, list_v.size()),
318 [&](const threading::range& r) {
319 for (size_t i = r.begin(); i < r.end(); i++) {
320 m_vertex_attribute[list_v[i].order].partition_id = i / interval;
321 }
322 },
323 NUM_THREADS);
324 }
