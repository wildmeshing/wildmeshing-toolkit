192 // only used with unit tests
193 void create_mesh_attributes(
194 const std::vector<VertexAttributes>& _vertex_attribute,
195 const std::vector<TetAttributes>& _tet_attribute)
196 {
197 const size_t n_tet = _tet_attribute.size();
198 m_vertex_attribute.resize(_vertex_attribute.size());
199 // The extras carry no data a caller supplies -- every field defaults -- so they only
200 // have to be sized alongside the shared collection.
201 m_vertex_extra.resize(_vertex_attribute.size());
202 m_face_attribute.resize(4 * n_tet);
203 m_tet_attribute.resize(n_tet);
204
205 // new for edge
206 // m_edge_attribute.resize(6 * n_tet);
207
208 for (size_t i = 0; i < _vertex_attribute.size(); i++)
209 m_vertex_attribute[i] = _vertex_attribute[i];
210
211 // Keep whatever init() reserved. AttributeCollection::resize only ever grows, so the
212 // calls above cannot shrink a collection -- but assigning m_attributes directly does,
213 // and it used to drop the tet attributes to exactly n_tet. The attributes have to stay
214 // at least as large as the connectivity: an operation that creates a new element
215 // indexes them by its id, and a 5->6 swap creates one. That left m_tet_attribute one
216 // short of the tet the swap adds, and AttributeCollection::operator[] read past the end
217 // of it -- ASan reports a heap-buffer-overflow reached from swap_edge_56_after, and
218 // libstdc++ then aborts on the corrupted heap a few allocations later, while libc++
219 // carries on and the tests pass.
220 const size_t tcap = std::max(n_tet, m_tet_attribute.size());
221 m_tet_attribute.m_attributes = std::vector<TetAttributes>(tcap);
222 for (size_t i = 0; i < n_tet; i++) m_tet_attribute[i] = _tet_attribute[i];
223 for (size_t i = 0; i < n_tet; i++)
224 m_tet_attribute[i].m_quality = get_quality(tuple_from_tet(i));
225 }
