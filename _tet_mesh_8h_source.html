356 f[1] = j;
357 else if (v3_id == m_indices[j])

969 // ---- OPERATION ACCOUNTING (counting only) ----
970 // Every call into split_edge, collapse_edge, swap_edge, swap_edge_44, swap_edge_56 and
971 // swap_face reports `attempt` on entry and exactly ONE other event on the way out: `committed`,
972 // or the place that refused it. The application's hooks are two of those places (before_hook,
973 // after_hook); why a hook said no is the hook's own to count. So for each kind, attempt equals
974 // the sum of the other events, and a subclass that counts can check both that identity and a
975 // hook's refusals against the reasons the hook counted. Added because the refusals made here,
976 // below every hook, were visible only as a difference: on the deliverable cube at
977 // target_distance_rel 1e-2 / front_conv_rel 1e-4, 521142 of the 542123 edge swaps that
978 // passed every before-hook in turn 5 were refused inside this class with no reason recorded.
979 //
980 // The default does nothing and no site reads a count: the mesh behaves identically with or
981 // without an override.
982 enum class OpKind : int { split, collapse, swap_32, swap_44, swap_56, swap_face, COUNT };
983 enum class OpEvent : int {
984 attempt,
985 before_hook, // the application's *_before said no
986 link_condition, // collapse: the classical link condition (m_collapse_check_link_condition)
987 no_return_tet, // collapse: both probes for the return tuple hit a boundary face
988 duplicate_tet, // a new tet repeats an existing one (collapse manifold check, 4-4, 5-6)
989 topology, // collapse: m_collapse_check_topology found a face missing
990 valence, // edge swap: the edge does not have this swap's number of tets around it
991 boundary, // edge swap: the edge's ring of tets is open; face swap: a boundary face
992 exists, // 3-2: the new face already exists; face swap: the new edge already exists
993 no_case_allowed, // 4-4 / 5-6: every candidate was vetoed by *_accept_case
994 no_better_case, // 4-4 / 5-6: no allowed candidate scored below the current cells
995 out_of_slots, // the preallocated vertex or tet slots ran out
996 after_hook, // the application's *_after said no, so the operation was rolled back
997 invariants, // invariants() said no after the after-hook passed; rolled back
998 committed,
999 screened, // a dry run passed every check that precedes the change (see dry_run())
1000 COUNT
1001 };

1472 void remove_tets_by_ids(const std::vector<size_t>& tids)
1473 {
1474 // A face or edge attribute lives at ONE canonical slot, (lowest incident tet) * 4
1475 // or 6 + local index. Deleting a tet that is a simplex's canonical owner silently
1476 // re-canonicalizes the simplex onto a surviving tet whose cell was never
1477 // maintained -- the operations only ever write the canonical slot -- so the
1478 // surviving cell holds stale garbage. Measured: after filtering, interior faces
1479 // read back m_is_surface_fs = true 0.12 (eps 0.09) from the real surface, and the
1480 // extracted "tracked surface" grows phantom triangles. Migrate every attribute
1481 // whose owner dies to the simplex's new owner BEFORE touching the connectivity.
1482 if (p_face_attrs != nullptr || p_edge_attrs != nullptr) {
1483 std::vector<bool> removed(tet_capacity(), false);
1484 for (const size_t tid : tids) {
1485 removed[tid] = true;
1486 }
1487 for (const size_t tid : tids) {
1488 const auto& tconn = m_tet_connectivity[tid];
1489 if (p_face_attrs != nullptr) {
1490 for (int lf = 0; lf < 4; ++lf) {
1491 std::array<size_t, 3> vids = {
1492 {tconn[m_local_faces[lf][0]],
1493 tconn[m_local_faces[lf][1]],
1494 tconn[m_local_faces[lf][2]]}};
1495 // The face's other incident tet, if any.
1496 auto common = set_intersection(
1497 m_vertex_connectivity[vids[0]].m_conn_tets,
1498 m_vertex_connectivity[vids[1]].m_conn_tets);
1499 common =
1500 set_intersection(common, m_vertex_connectivity[vids[2]].m_conn_tets);
1501 size_t other = std::numeric_limits<size_t>::max();
1502 for (const size_t t : common) {
1503 if (t != tid) {
1504 other = t;
1505 }
1506 }
1507 // Migrate only when this tet was the canonical owner (tid < other)
1508 // and the other side survives; other < tid means the attribute
1509 // already lives on the survivor.
1510 if (other == std::numeric_limits<size_t>::max() || removed[other] ||
1511 other < tid) {
1512 continue;
1513 }
1514 std::array<size_t, 3> sorted = vids;
1515 std::sort(sorted.begin(), sorted.end());
1516 for (int lo = 0; lo < 4; ++lo) {
1517 std::array<size_t, 3> ovids = {
1518 {m_tet_connectivity[other][m_local_faces[lo][0]],
1519 m_tet_connectivity[other][m_local_faces[lo][1]],
1520 m_tet_connectivity[other][m_local_faces[lo][2]]}};
1521 std::sort(ovids.begin(), ovids.end());
1522 if (ovids == sorted) {
1523 p_face_attrs->move(tid * 4 + lf, other * 4 + lo);
1524 break;
1525 }
1526 }
1527 }
1528 }
1529 if (p_edge_attrs != nullptr) {
1530 for (int le = 0; le < 6; ++le) {
1531 const size_t v0 = tconn[m_local_edges[le][0]];
1532 const size_t v1 = tconn[m_local_edges[le][1]];
1533 const auto fan = set_intersection(
1534 m_vertex_connectivity[v0].m_conn_tets,
1535 m_vertex_connectivity[v1].m_conn_tets);
1536 // Only the canonical owner migrates, once, to the lowest survivor.
1537 if (fan.empty() || fan.front() != tid) {
1538 continue;
1539 }
1540 size_t new_owner = std::numeric_limits<size_t>::max();
1541 for (const size_t t : fan) {
1542 if (!removed[t]) {
1543 new_owner = t; // fan is sorted ascending; first kept wins
1544 break;
1545 }
1546 }
1547 if (new_owner == std::numeric_limits<size_t>::max()) {
1548 continue; // the edge dies with its whole fan
1549 }
1550 for (int lo = 0; lo < 6; ++lo) {
1551 const size_t w0 = m_tet_connectivity[new_owner][m_local_edges[lo][0]];
1552 const size_t w1 = m_tet_connectivity[new_owner][m_local_edges[lo][1]];
1553 if ((w0 == v0 && w1 == v1) || (w0 == v1 && w1 == v0)) {
1554 p_edge_attrs->move(tid * 6 + le, new_owner * 6 + lo);
1555 break;
1556 }
1557 }
1558 }
1559 }
1560 }
1561 }
1562
1563 for (size_t tid : tids) {
1564 m_tet_connectivity[tid].m_is_removed = true;
1565 for (int j = 0; j < 4; j++)
1566 vector_erase(m_vertex_connectivity[m_tet_connectivity[tid][j]].m_conn_tets, tid);
1567 }
1568 for (auto& v : m_vertex_connectivity) {
1569 if (v.m_is_removed) continue;
1570 if (v.m_conn_tets.empty()) v.m_is_removed = true;
1571 }
1572 if (m_surface_history) resync_surface_history();
