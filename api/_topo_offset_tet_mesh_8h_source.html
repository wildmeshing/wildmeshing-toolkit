2831 std::shared_ptr<polysolve::nonlinear::Problem> smoothing_extra_energy(
2832 const size_t vid) const override
2833 {
2834 // TetWild's rules (m_op_rules): no extra energy, so the engine's smoother minimises the
2835 // raw sum of AMIPS over the one-ring (AMIPSEnergy3D), as TetWild's does.
2836 if (tetwild_rules()) return nullptr;
2837 auto sum = std::make_shared<optimization::EnergySum>();
2838 sum->add_energy(amips3_energy(vid));
2839 // Before the march (repulsion_smoothing()) there is no front.
2840 if (is_repulsion_vertex(vid)) {
2841 sum->add_energy(repulsion_energy(vid));
2842 return sum;
2843 }
2844 if (vertex_carries_offset_term(vid)) {
2845 sum->add_energy(front_energy(vid));
2846 } else if (!vertex_is_held_on_input(vid)) {
2847 // E's band term moves with this vertex too -- except a vertex held on the input.
2848 // There d = 0 at the vertex and d has a kink: the term's slope along the input's
2849 // normal jumps across the surface, and its curvature there is 0 (or ~1e10 near an
2850 // input edge), so the Newton step along the normal is set by AMIPS alone and lands
2851 // far off; the projected step then discards the normal motion anyway. MEASURED
2852 // 2026-10-09, cube, serial, 3 turns: with the term these solves took 24.6 Newton
2853 // iterations and ~20 line-search trials per iteration (72% of the smoother's value
2854 // calls), without it 3.9 and 1; 19.3 s -> 11.3 s. E still guards the move:
2855 // smooth_vertex_under_energy() vetoes on the ring's sum of tet_energy().
2856 // Under "exact", not where it cancels (BAND TERM THAT CANCELS, tet_energy()): a ring
2857 // of band cells clear of the input's bisectors tiles the star of vid, so the term is a
2858 // constant, and its cells' exact integrals would add only work and their rounding to
2859 // every Newton step. The veto's own test (cancels_before in
2860 // smooth_vertex_under_energy()). Constant wherever the move can end, not only here: vid
2861 // is smoothed, so not on the domain wall (smooth_before()), and its ring closes around
2862 // it; the engine keeps no position that inverts a ring cell, so vid stays inside the
2863 // convex hull of its link; and the test reads the points only through their box, their
2864 // least side of each bisector polytope's planes, and min_P max d_P (d_P convex), none
2865 // of which a point inside that hull changes. So the test here certifies every position
2866 // the smoother can keep. "front_flux" and "exact_swept" leave the constant out by their
2867 // own forms
2868 // (band_volume_energy_at()). MEASURED (cube 1e-2 / 1e-4, 15 rounds, serial, two runs
2869 // each): 659 of 46,594 non-front solves had such a ring; at every one's end the
2870 // veto's test found it clear again, and every one was accepted; smoothing 7.7-7.9 s ->
2871 // 7.4-7.5 s.
2872 if (m_offset_params.band_volume_rule == "exact" &&
2873 band_term_cancels(get_one_ring_tids_for_vertex(vid)))
2874 return sum;
2875 if (const auto bv = band_volume_energy_at(vid)) sum->add_energy(bv);
2876 }
2877 return sum;
2878 }
