61struct ReplacedSurface
62{
64 std::vector<std::pair<int64_t, Eigen::Vector3d>> normals;
66 size_t against = 0;
69 size_t star_folds = 0;
73 struct Crease
74 {
75 size_t a = 0, b = 0;
76 int orientation = 0;
77 };
78 std::vector<Crease> creases;
80 std::optional<Eigen::Vector3d> normal(int64_t region) const
81 {
82 for (const auto& [r, n] : normals) {
83 if (r == region) return n;
84 }
85 return std::nullopt;
86 }
88 const Crease* crease(size_t a, size_t b) const
89 {
90 for (const Crease& c : creases) {
91 if ((c.a == a && c.b == b) || (c.a == b && c.b == a)) return &c;
92 }
93 return nullptr;
94 }
97 void rename(size_t from, size_t to)
98 {
99 for (Crease& c : creases) {
100 if (c.a == from) c.a = to;
101 if (c.b == from) c.b = to;
102 }
103 }
105 bool folded() const { return against > 0 || star_folds > 0; }
106};
