166 // only used with unit tests
167 void create_mesh_attributes(
168 const std::vector<VertexAttributes>& _vertex_attribute,
169 const std::vector<TetAttributes>& _tet_attribute)
170 {
171 const size_t n_tet = _tet_attribute.size();
172 m_vertex_attribute.resize(_vertex_attribute.size());
173 m_face_attribute.resize(4 * n_tet);
174 m_tet_attribute.resize(n_tet);
175
176 // new for edge
177 // m_edge_attribute.resize(6 * n_tet);
178
179 for (size_t i = 0; i < _vertex_attribute.size(); i++)
180 m_vertex_attribute[i] = _vertex_attribute[i];
181
182 // Keep whatever init() reserved. AttributeCollection::resize only ever grows, so the
183 // calls above cannot shrink a collection -- but assigning m_attributes directly does,
184 // and it used to drop the tet attributes to exactly n_tet. The attributes have to stay
185 // at least as large as the connectivity: an operation that creates a new element
186 // indexes them by its id, and a 5->6 swap creates one. That left m_tet_attribute one
187 // short of the tet the swap adds, and AttributeCollection::operator[] read past the end
188 // of it -- ASan reports a heap-buffer-overflow reached from swap_edge_56_after, and
189 // libstdc++ then aborts on the corrupted heap a few allocations later, while libc++
190 // carries on and the tests pass.
191 const size_t tcap = std::max(n_tet, m_tet_attribute.size());
192 m_tet_attribute.m_attributes = std::vector<TetAttributes>(tcap);
193 for (size_t i = 0; i < n_tet; i++) m_tet_attribute[i] = _tet_attribute[i];
194 for (size_t i = 0; i < n_tet; i++)
195 m_tet_attribute[i].m_quality = get_quality(tuple_from_tet(i));
196 }
