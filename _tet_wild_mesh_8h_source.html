188 if (!m_tet_params.preserve_feature_points) return false;
189 const size_t f1 = m_vertex_extra[v1_id].m_feature_point_id;

195 bool smoothing_position_is_allowed(const size_t vid, const Vector3d& p) const override
196 {
197 const size_t f = m_vertex_extra[vid].m_feature_point_id;
198 if (f == NO_FEATURE || !m_tet_params.preserve_feature_points) return true;
199 // A ball, not a pin -- the vertex may move anywhere within m_feature_eps of its
200 // anchor, exactly the freedom the envelope gives everywhere else.
201 return (p - m_feature_points[f]).squaredNorm() <= m_feature_eps * m_feature_eps;

273 // only used with unit tests
274 void create_mesh_attributes(
275 const std::vector<VertexAttributes>& _vertex_attribute,
276 const std::vector<TetAttributes>& _tet_attribute)
277 {
278 const size_t n_tet = _tet_attribute.size();
279 m_vertex_attribute.resize(_vertex_attribute.size());
280 // The extras carry no data a caller supplies -- every field defaults -- so they only
281 // have to be sized alongside the shared collection.
282 m_vertex_extra.resize(_vertex_attribute.size());
283 m_face_attribute.resize(4 * n_tet);
284 m_tet_attribute.resize(n_tet);
285
286 // new for edge
287 // m_edge_attribute.resize(6 * n_tet);
288
289 for (size_t i = 0; i < _vertex_attribute.size(); i++)
290 m_vertex_attribute[i] = _vertex_attribute[i];
291
292 // Keep whatever init() reserved. AttributeCollection::resize only ever grows, so the
293 // calls above cannot shrink a collection -- but assigning m_attributes directly does,
294 // and it used to drop the tet attributes to exactly n_tet. The attributes have to stay
295 // at least as large as the connectivity: an operation that creates a new element
296 // indexes them by its id, and a 5->6 swap creates one. That left m_tet_attribute one
297 // short of the tet the swap adds, and AttributeCollection::operator[] read past the end
298 // of it -- ASan reports a heap-buffer-overflow reached from swap_edge_56_after, and
299 // libstdc++ then aborts on the corrupted heap a few allocations later, while libc++
300 // carries on and the tests pass.
301 const size_t tcap = std::max(n_tet, m_tet_attribute.size());
302 m_tet_attribute.m_attributes = std::vector<TetAttributes>(tcap);
303 for (size_t i = 0; i < n_tet; i++) m_tet_attribute[i] = _tet_attribute[i];
304 for (size_t i = 0; i < n_tet; i++)
305 m_tet_attribute[i].m_quality = get_quality(tuple_from_tet(i));
306 }
