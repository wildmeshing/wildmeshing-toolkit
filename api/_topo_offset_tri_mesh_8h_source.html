1349 {
1350 // Boundary geometry first: a segment on any tag-region boundary may not
1351 // drift out of that boundary's tube, and one on several boundaries -- a junction -- is
1352 // held in their intersection. The mask carries the input complex too, since every complex
1353 // simplex lies on tag boundaries, and E_t is built from the input mesh before construction
1354 // touches it, so the per-tag tubes hold the as-loaded geometry.
1355 //
1356 // Keyed on the vertices, not the edge: every caller is an operation asking about a segment
1357 // it is about to create or has just created, whose own edge attributes are not written
1358 // yet. The endpoints' masks are, maintained by the operations themselves (AND at a split,
1359 // OR at a collapse).
1360 uint64_t mask = edge_mask(vids);
1361 bool all_offset = true;
1362 for (const size_t v : vids) {
1363 all_offset = all_offset && m_vertex_extra[v].m_is_on_offset;
1364 }
1365
1366 // The ambiguous case: both endpoints can be on region boundaries and on the offset front
1367 // at once -- a real state, and the point of tracking the two families separately. The
1368 // endpoint-mask AND is then necessary but not sufficient for the segment lying on a shared
1369 // boundary: two vertices on different junctions can share a tag bit by coincidence and be
1370 // joined by an offset chord nowhere near that tag's curve, which a container it was never
1371 // meant to satisfy then refuses to refine. Ask the edge's own class, the only record that
1372 // distinguishes a chord from a boundary.
1373 //
1374 // Reading the slot is safe here and would not be unconditionally: the base checks a
1375 // split's two child segments before writing their attributes, so their slots still hold a
1376 // recycled edge's class. A split child never reaches this branch -- its mask and its
1377 // offset flag are mutually exclusive by construction, so one of `mask` and `all_offset` is
1378 // always empty -- and the m_is_surface_fs guard leaves both constraints standing rather
1379 // than dropping one when a slot is illegible.
1380 if (mask != 0 && all_offset) {
1381 if (const auto found = try_tuple_from_edge(vids)) {
1382 const size_t eid = std::get<1>(*found);
1383 if (m_edge_attribute[eid].m_is_surface_fs) {
1384 if (edge_is_offset(eid)) {
1385 mask = 0; // an offset edge lies on no region boundary
1386 } else {
1387 all_offset = false; // a region edge is not the offset front
1388 }
1389 }
1390 }
1391 }
1392 // Both families compose: whatever holds this segment holds it at once. The offset side
1393 // holds only in the frozen-front final pass (see containment_for()); in the loop the
1394 // front is what the optimization moves, so it contributes nothing there.
1395 const std::shared_ptr<Envelope> base = containment_for(mask, all_offset);
1396 if (base || m_deform_tags.empty()) return base;
1397 // deform_others' ops-only tube: a released boundary is held by no mask -- its vertices
1398 // were freed so smoothing can carry the object -- which would leave the operations free
1399 // to decimate and reposition it. A segment the masks and the offset class do not claim,
1400 // but which lies on a released boundary by its incident faces' current tags, is held to
1401 // the tube around the boundary's current shape. A fresh split child can misread its
1402 // recycled face slots for this one check, and is at worst skipped or over-held once.
1403 if (const auto found = try_tuple_from_edge(vids)) {
1404 if (edge_borders_released_boundary(std::get<0>(*found))) return released_envelope();
1405 }
1406 return nullptr;
1407 }

1769 template <typename Visit>
1770 void for_each_edge_sample(const Vector2d& p0, const Vector2d& p1, Visit&& visit) const
1771 {
1772 const int k = m_offset_params.stencil_order;
1773 if (k < 0) return;
1774
1775 const auto emit = [&](const double wa, const double wb) {
1776 visit(Vector2d(wa * p0 + wb * p1), wa, wb);
1777 };
1778 // Order 0 is the two ENDS alone, each end's own placement error.
1779 if (k == 0) {
1780 emit(1., 0.);
1781 emit(0., 1.);
1782 return;
1783 }
1784 // Order k >= 1: the vertices of the chord cut into n = 2^(k-1) pieces, then the midpoint
1785 // of each piece -- 2n + 1 points: 3, 5, 9, 17, ...
1786 const int n = 1 << (k - 1);
1787 const double dn = double(n);
1788 for (int i = n; i >= 0; --i) {
1789 emit(double(i) / dn, double(n - i) / dn);
1790 }
1791 const double d2n = 2. * dn;
1792 for (int i = n - 1; i >= 0; --i) {
1793 emit((2. * i + 1.) / d2n, (2. * (n - 1 - i) + 1.) / d2n);
1794 }
1795 }
