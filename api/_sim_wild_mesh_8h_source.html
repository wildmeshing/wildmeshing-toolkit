209 // only used with unit tests
210 void create_mesh_attributes(
211 const std::vector<VertexAttributes>& _vertex_attribute,
212 const std::vector<TetAttributes>& _tet_attribute)
213 {
214 const size_t n_tet = _tet_attribute.size();
215 m_vertex_attribute.resize(_vertex_attribute.size());
216 m_face_attribute.resize(4 * n_tet);
217 m_tet_attribute.resize(n_tet);
218
219 for (size_t i = 0; i < _vertex_attribute.size(); i++)
220 m_vertex_attribute[i] = _vertex_attribute[i];
221
222 // Keep whatever init() reserved. AttributeCollection::resize only ever grows, so the
223 // calls above cannot shrink a collection -- but assigning m_attributes directly does,
224 // and it used to drop the tet attributes to exactly n_tet. The attributes have to stay
225 // at least as large as the connectivity: an operation that creates a new element
226 // indexes them by its id, and a 5->6 swap creates one. tetwild hit the same thing (see
227 // the note on its copy); there the overrun wrote a double past the end and libc++ let
228 // it pass, here it assigns a std::set and segfaults outright.
229 const size_t tcap = std::max(n_tet, m_tet_attribute.size());
230 m_tet_attribute.m_attributes = std::vector<TetAttributes>(tcap);
231 for (size_t i = 0; i < n_tet; i++) m_tet_attribute[i] = _tet_attribute[i];
232 for (size_t i = 0; i < n_tet; i++)
233 m_tet_attribute[i].m_quality = get_quality(tuple_from_tet(i));
234 }
