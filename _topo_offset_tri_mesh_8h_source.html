1330 std::shared_ptr<Envelope> surface_envelope_for_edge(
1331 const std::array<size_t, 2>& vids) const override
1332 {
1333 // Boundary geometry first: a segment on any tag-region boundary may not
1334 // drift out of that boundary's tube, and one on several boundaries -- a junction -- is
1335 // held in their intersection. The mask carries the input complex too, since every complex
1336 // simplex lies on tag boundaries, and E_t is built from the input mesh before construction
1337 // touches it, so the per-tag tubes hold the as-loaded geometry.
1338 //
1339 // Keyed on the vertices, not the edge: every caller is an operation asking about a segment
1340 // it is about to create or has just created, whose own edge attributes are not written
1341 // yet. The endpoints' masks are, maintained by the operations themselves (AND at a split,
1342 // OR at a collapse).
1343 uint64_t mask = edge_mask(vids);
1344 bool all_offset = true;
1345 for (const size_t v : vids) {
1346 all_offset = all_offset && m_vertex_extra[v].m_is_on_offset;
1347 }
1348
1349 // The ambiguous case: both endpoints can be on region boundaries and on the offset front
1350 // at once -- a real state, and the point of tracking the two families separately. The
1351 // endpoint-mask AND is then necessary but not sufficient for the segment lying on a shared
1352 // boundary: two vertices on different junctions can share a tag bit by coincidence and be
1353 // joined by an offset chord nowhere near that tag's curve, which a container it was never
1354 // meant to satisfy then refuses to refine. Ask the edge's own class, the only record that
1355 // distinguishes a chord from a boundary.
1356 //
1357 // Reading the slot is safe here and would not be unconditionally: the base checks a
1358 // split's two child segments before writing their attributes, so their slots still hold a
1359 // recycled edge's class. A split child never reaches this branch -- its mask and its
1360 // offset flag are mutually exclusive by construction, so one of `mask` and `all_offset` is
1361 // always empty -- and the m_is_surface_fs guard leaves both constraints standing rather
1362 // than dropping one when a slot is illegible.
1363 if (mask != 0 && all_offset) {
1364 if (const auto found = try_tuple_from_edge(vids)) {
1365 const size_t eid = std::get<1>(*found);
1366 if (m_edge_attribute[eid].m_is_surface_fs) {
1367 if (edge_is_offset(eid)) {
1368 mask = 0; // an offset edge lies on no region boundary
1369 } else {
1370 all_offset = false; // a region edge is not the offset front
1371 }
1372 }
1373 }
1374 }
1375 // Both families compose: whatever holds this segment holds it at once. The offset side
1376 // holds only in the frozen-front final pass (see containment_for()); in the loop the
1377 // front is what the optimization moves, so it contributes nothing there.
1378 const std::shared_ptr<Envelope> base = containment_for(mask, all_offset);
1379 if (base || m_deform_tags.empty()) return base;
1380 // deform_others' ops-only tube: a released boundary is held by no mask -- its vertices
1381 // were freed so smoothing can carry the object -- which would leave the operations free
1382 // to decimate and reposition it. A segment the masks and the offset class do not claim,
1383 // but which lies on a released boundary by its incident faces' current tags, is held to
1384 // the tube around the boundary's current shape. A fresh split child can misread its
1385 // recycled face slots for this one check, and is at worst skipped or over-held once.
1386 if (const auto found = try_tuple_from_edge(vids)) {
1387 if (edge_borders_released_boundary(std::get<0>(*found))) return released_envelope();
1388 }
1389 return nullptr;
1390 }

1743 template <typename Visit>
1744 void for_each_edge_sample(const Vector2d& p0, const Vector2d& p1, Visit&& visit) const
1745 {
1746 const int k = m_offset_params.stencil_order;
1747 if (k < 0) return;
1748
1749 const auto emit = [&](const double wa, const double wb) {
1750 visit(Vector2d(wa * p0 + wb * p1), wa, wb);
1751 };
1752 // Order 0 is the two ENDS alone, each end's own placement error.
1753 if (k == 0) {
1754 emit(1., 0.);
1755 emit(0., 1.);
1756 return;
1757 }
1758 // Order k >= 1: the vertices of the chord cut into n = 2^(k-1) pieces, then the midpoint
1759 // of each piece -- 2n + 1 points: 3, 5, 9, 17, ...
1760 const int n = 1 << (k - 1);
1761 const double dn = double(n);
1762 for (int i = n; i >= 0; --i) {
1763 emit(double(i) / dn, double(n - i) / dn);
1764 }
1765 const double d2n = 2. * dn;
1766 for (int i = n - 1; i >= 0; --i) {
1767 emit((2. * i + 1.) / d2n, (2. * (n - 1 - i) + 1.) / d2n);
1768 }
1769 }

1777 m_vertex_attribute[e.switch_vertex(*this).vid(*this)].m_posf,
1778 std::forward<Visit>(visit));
