76 bool write_mesh(
77 const std::string& path,
78 const Eigen::MatrixXd& V,
79 const Eigen::MatrixXi& C,
80 paraviewo::CellType type)
81 {
82 if (!output) return file_written(path, file.write_mesh(path, V, C, type));
83 std::vector<paraviewo::CellElement> cells(C.rows());
84 for (Eigen::Index i = 0; i < C.rows(); ++i) {

90 bool write_mesh(
91 const std::string& path,
92 const Eigen::MatrixXd& V,
93 const std::vector<paraviewo::CellElement>& cells)
94 {
95 if (!output) return file_written(path, file.write_mesh(path, V, cells));
96 MeshData data;
97 data.V = DataMatrix<double>::Zero(V.rows(), 3);
98 data.V.leftCols(V.cols()) = V;
99 data.vertex_attributes = points;
100 std::map<CellType, std::vector<size_t>> rows;
101 for (size_t i = 0; i < cells.size(); ++i) {
102 const auto& c = cells[i];
103 const auto type = static_cast<CellType>(
104 paraviewo::paraview_tags::VTKTag(int(c.vertices.size()), c.ctype));
105 auto& block = data.cells[type].connectivity;
106 block.conservativeResize(block.rows() + 1, cell_size(type));
107 for (size_t j = 0; j < c.vertices.size(); ++j)
108 block(block.rows() - 1, j) = c.vertices[j];
109 rows[type].push_back(i);
110 }
111 // Each block its rows of every cell field, in the field's own type, with its labels.
112 for (const auto& [type, indices] : rows)
113 for (const auto& [name, a] : fields) {
114 MeshAttribute& field = data.cells[type].attributes[name];
115 field.labels = a.labels;
116 field.values = std::visit(
117 [&indices](const auto& values) -> AttributeValues {
118 std::decay_t<decltype(values)> sub(indices.size(), values.cols());
119 for (size_t r = 0; r < indices.size(); ++r)
120 sub.row(r) = values.row(indices[r]);
121 return sub;
122 },
123 a.values);
124 }
125 data.validate();
126 output->insert_or_assign(path, std::move(data));
127 // Like VTUWriter::write_mesh, which clears its fields after every write.
128 points.clear();
129 fields.clear();
130 return true;
131 }

153// Export arrays are sometimes indexed by reserved slots. Remove dead slots without
154// consolidating or otherwise mutating the algorithm's mesh, and keep every field aligned.
155inline void compact_vertices(MeshData& data, const std::vector<size_t>& rows)
156{
157 DataMatrix<double> V(rows.size(), 3);
158 std::vector<int64_t> remap(data.V.rows(), -1);
159 for (size_t i = 0; i < rows.size(); ++i) {
160 V.row(i) = data.V.row(rows[i]);
161 remap[rows[i]] = i;
162 }
163 data.V = std::move(V);
164 for (auto& r : data.V_exact.rows) {
165 r = remap.at(r);
166 if (r < 0) throw std::runtime_error("Exact position of an inactive vertex");
167 }
168 for (auto& [name, a] : data.vertex_attributes)
169 a.values = std::visit(
170 [&](const auto& values) -> AttributeValues {
171 using M = std::decay_t<decltype(values)>;
172 M compact(rows.size(), values.cols());
173 for (size_t i = 0; i < rows.size(); ++i) compact.row(i) = values.row(rows[i]);
174 return compact;
175 },
176 a.values);
177 for (auto& [type, block] : data.cells)
178 for (Eigen::Index i = 0; i < block.connectivity.size(); ++i) {
179 const auto v = remap.at(block.connectivity.data()[i]);
180 if (v < 0) throw std::runtime_error("Export references an inactive vertex");
181 block.connectivity.data()[i] = v;
182 }
183 data.validate();
184}
185// The exact positions of the mesh's un-rounded vertices (MeshData::V_exact), in the rows of the
186// memory export `data`: by vertex slot, as most exports are before compact_slot_outputs()
187// renumbers V and V_exact together, or, `packed`, in get_vertices() order, as an export that
188// skips the dead slots itself is. V_exact has V's three columns; a 2D mesh's third is 0.
189template <typename Mesh>
190void collect_exact_positions(const Mesh& mesh, MeshData& data, const bool packed = false)
191{
192 data.V_exact = {};
193 const auto vs = mesh.get_vertices();
194 for (size_t k = 0; k < vs.size(); ++k) {
195 const size_t vid = vs[k].vid(mesh);
196 const auto& va = mesh.m_vertex_attribute[vid];
197 if (va.m_is_rounded) continue;
198 auto p = va.pos();
199 data.V_exact.rows.push_back(int64_t(packed ? k : vid));
200 for (Eigen::Index k = 0; k < p.size(); ++k) data.V_exact.values.push_back(p[k].get_str());
201 for (Eigen::Index k = p.size(); k < data.V.cols(); ++k) data.V_exact.values.push_back("0");
202 }
203}
204template <typename Mesh>
205void compact_slot_outputs(const Mesh& mesh, MeshMap& outputs)
206{
207 std::vector<size_t> rows;
208 for (const auto& v : mesh.get_vertices()) rows.push_back(v.vid(mesh));
209 for (auto& [name, data] : outputs) compact_vertices(data, rows);
210}
211// The result key of each memory export. The exporters keep their legacy file naming,
212// <stem><suffix>.vtu, and the memory calls use the stem "mesh". An empty key drops an export
213// that the caller returns in another form or that the mesh already holds: the boundary
214// (`_surf`), the non-manifold or feature edges (`_edge`) and the offset's front (`_off`,
215// `_front`) are the mesh's tags -- its face and edge blocks, and the tet (2D: triangle) tags
216// that differ across a face -- and each was a second copy of the mesh's vertices. Until 2026-11
217// they were returned as `surface`, `edges` and `offset`; isotropic_remeshing, whose mesh has no
218// edge block, still returns its feature edges as `edges`, and prismatic_mesh its classified
219// offset faces as `offset`. The .msh among them are the meshes a history names its points by:
220// <output>_history_input.msh, simwild's and tetwild's, where the surface history (and the volume
221// history) starts, and tetwild's <output>_history_output.msh, the mesh before its filter, where
222// the volume history ends.
223using OutputKeys = std::map<std::string, std::string>;
224inline OutputKeys default_output_keys()
225{
226 return {
227 {"mesh.vtu", "mesh"},
228 {"mesh_surf.vtu", ""},
229 {"mesh_edge.vtu", ""},
230 {"mesh_off.vtu", ""},
231 {"mesh_front.vtu", ""},
232 {"mesh_offset_faces.vtu", "offset"},
233 {"mesh_history_input.msh", "history_input"},
234 {"mesh_history_output.msh", "history_output"}};
235}
236// Rename the exports into public result keys. An export missing from `keys`, or two exports
237// with the same key, is an error rather than a silently dropped output.
238inline MeshMap named_outputs(MeshMap raw, const OutputKeys& keys = default_output_keys())
239{
240 MeshMap result;
241 for (auto& [path, data] : raw) {
242 const auto key = keys.find(path);
243 if (key == keys.end()) throw std::runtime_error("Unexpected memory export: " + path);
244 if (key->second.empty()) continue;
245 if (!result.emplace(key->second, std::move(data)).second)
246 throw std::runtime_error("Two memory exports map to the result key " + key->second);
247 }
248 return result;
249}
