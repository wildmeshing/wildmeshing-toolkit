482 {
483 const size_t floor = 64;
484 // A mesh that grows on demand only needs room for the passes not to grow at once.
485 const double factor = storage_grows_on_demand() ? 1.25 : m_preallocation_factor;
486 double c = std::ceil(factor * static_cast<double>(live_count));

488 if (capacity < live_count) capacity = live_count;
489 return capacity < floor ? floor : capacity;
