172 bool boundary_edges_at_vertex_inside(size_t vid) const override
173 {
174 if (!m_vertex_extra[vid].m_is_on_open_boundary || !m_order2_envelope) {
175 return true;
176 }
177 for (const size_t u : get_one_ring_vids_for_vertex(vid)) {
178 if (!m_vertex_extra[u].m_is_on_open_boundary) {
179 continue;
180 }
181 // Exact membership: an open boundary edge has exactly one incident tracked
182 // face. Flags alone would also catch interior chords between two boundary
183 // vertices (e.g. across a corner), which must not be tube-tested.
184 if (get_num_surface_faces_for_edge({{vid, u}}) != 1) {
185 continue;
186 }
187 if (m_order2_envelope->is_outside(
188 std::array<Eigen::Vector3d, 2>{
189 {m_vertex_attribute[vid].m_posf, m_vertex_attribute[u].m_posf}})) {
190 return false;
191 }
192 }
193 return true;
194 }
195 bool collapse_before_vertex(size_t v1, size_t v2, double edge_length) override
196 {
197 if (collapse_breaks_feature_point(v1, v2)) return false;
198 if (edge_length <= 0 || !m_vertex_extra[v1].m_is_on_open_boundary) return true;
199 return m_vertex_extra[v2].m_is_on_open_boundary ||
200 !m_order2_envelope->is_outside(m_vertex_attribute[v2].m_posf);
201 }
202
209 bool collapse_breaks_feature_point(const size_t v1_id, const size_t v2_id) const
210 {
211 if (!m_tet_params.preserve_feature_points) return false;
212 const size_t f1 = m_vertex_extra[v1_id].m_feature_point_id;
213 if (f1 == NO_FEATURE) return false;
214 return (m_vertex_attribute[v2_id].m_posf - m_feature_points[f1]).squaredNorm() >
215 m_feature_eps * m_feature_eps;
216 }
217
218 bool smoothing_position_is_allowed(const size_t vid, const Vector3d& p) const override
219 {
220 const size_t f = m_vertex_extra[vid].m_feature_point_id;
221 if (f == NO_FEATURE || !m_tet_params.preserve_feature_points) return true;
222 // A ball, not a pin -- the vertex may move anywhere within m_feature_eps of its
223 // anchor, exactly the freedom the envelope gives everywhere else.
224 return (p - m_feature_points[f]).squaredNorm() <= m_feature_eps * m_feature_eps;
225 }

296 // only used with unit tests
297 void create_mesh_attributes(
298 const std::vector<VertexAttributes>& _vertex_attribute,
299 const std::vector<TetAttributes>& _tet_attribute)
300 {
301 const size_t n_tet = _tet_attribute.size();
302 m_vertex_attribute.resize(_vertex_attribute.size());
303 // The extras carry no data a caller supplies -- every field defaults -- so they only
304 // have to be sized alongside the shared collection.
305 m_vertex_extra.resize(_vertex_attribute.size());
306 m_face_attribute.resize(4 * n_tet);
307 m_tet_attribute.resize(n_tet);
308
309 // new for edge
310 // m_edge_attribute.resize(6 * n_tet);
311
312 for (size_t i = 0; i < _vertex_attribute.size(); i++)
313 m_vertex_attribute[i] = _vertex_attribute[i];
314
315 // Keep whatever init() reserved. AttributeCollection::resize only ever grows, so the
316 // calls above cannot shrink a collection -- but assigning m_attributes directly does,
317 // and it used to drop the tet attributes to exactly n_tet. The attributes have to stay
318 // at least as large as the connectivity: an operation that creates a new element
319 // indexes them by its id, and a 5->6 swap creates one. That left m_tet_attribute one
320 // short of the tet the swap adds, and AttributeCollection::operator[] read past the end
321 // of it -- ASan reports a heap-buffer-overflow reached from swap_edge_56_after, and
322 // libstdc++ then aborts on the corrupted heap a few allocations later, while libc++
323 // carries on and the tests pass.
324 const size_t tcap = std::max(n_tet, m_tet_attribute.size());
325 m_tet_attribute.m_attributes = std::vector<TetAttributes>(tcap);
326 for (size_t i = 0; i < n_tet; i++) m_tet_attribute[i] = _tet_attribute[i];
327 for (size_t i = 0; i < n_tet; i++)
328 m_tet_attribute[i].m_quality = get_quality(tuple_from_tet(i));
329 }
