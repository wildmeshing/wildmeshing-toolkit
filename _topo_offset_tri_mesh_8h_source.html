1322 std::shared_ptr<SampleEnvelope> surface_envelope_for_edge(
1323 const std::array<size_t, 2>& vids) const override
1324 {
1325 // Boundary geometry first: a segment on any tag-region boundary may not
1326 // drift out of that boundary's tube, and one on several boundaries -- a junction -- is
1327 // held in their intersection. The mask carries the input complex too, since every complex
1328 // simplex lies on tag boundaries, and E_t is built from the input mesh before construction
1329 // touches it, so the per-tag tubes hold the as-loaded geometry.
1330 //
1331 // Keyed on the vertices, not the edge: every caller is an operation asking about a segment
1332 // it is about to create or has just created, whose own edge attributes are not written
1333 // yet. The endpoints' masks are, maintained by the operations themselves (AND at a split,
1334 // OR at a collapse).
1335 uint64_t mask = edge_mask(vids);
1336 bool all_offset = true;
1337 for (const size_t v : vids) {
1338 all_offset = all_offset && m_vertex_extra[v].m_is_on_offset;
1339 }
1340
1341 // The ambiguous case: both endpoints can be on region boundaries and on the offset front
1342 // at once -- a real state, and the point of tracking the two families separately. The
1343 // endpoint-mask AND is then necessary but not sufficient for the segment lying on a shared
1344 // boundary: two vertices on different junctions can share a tag bit by coincidence and be
1345 // joined by an offset chord nowhere near that tag's curve, which a container it was never
1346 // meant to satisfy then refuses to refine. Ask the edge's own class, the only record that
1347 // distinguishes a chord from a boundary.
1348 //
1349 // Reading the slot is safe here and would not be unconditionally: the base checks a
1350 // split's two child segments before writing their attributes, so their slots still hold a
1351 // recycled edge's class. A split child never reaches this branch -- its mask and its
1352 // offset flag are mutually exclusive by construction, so one of `mask` and `all_offset` is
1353 // always empty -- and the m_is_surface_fs guard leaves both constraints standing rather
1354 // than dropping one when a slot is illegible.

1373 // were freed so smoothing can carry the object -- which would leave the operations free
1374 // to decimate and reposition it. A segment the masks and the offset class do not claim,
1375 // but which lies on a released boundary by its incident faces' current tags, is held to
1376 // the tube around the boundary's current shape. A fresh split child can misread its
1377 // recycled face slots for this one check, and is at worst skipped or over-held once.

1735 template <typename Visit>
1736 void for_each_edge_sample(const Vector2d& p0, const Vector2d& p1, Visit&& visit) const
1737 {
1738 const int k = m_offset_params.stencil_order;
1739 if (k < 0) return;
1740
1741 const auto emit = [&](const double wa, const double wb) {
1742 visit(Vector2d(wa * p0 + wb * p1), wa, wb);
1743 };
1744 // Order 0 is the two ENDS alone, each end's own placement error.
1745 if (k == 0) {
1746 emit(1., 0.);
1747 emit(0., 1.);
1748 return;
1749 }
1750 // Order k >= 1: the vertices of the chord cut into n = 2^(k-1) pieces, then the midpoint
1751 // of each piece -- 2n + 1 points: 3, 5, 9, 17, ...
1752 const int n = 1 << (k - 1);
1753 const double dn = double(n);
1754 for (int i = n; i >= 0; --i) {
1755 emit(double(i) / dn, double(n - i) / dn);
1756 }
1757 const double d2n = 2. * dn;
1758 for (int i = n - 1; i >= 0; --i) {
1759 emit((2. * i + 1.) / d2n, (2. * (n - 1 - i) + 1.) / d2n);
1760 }
1761 }
