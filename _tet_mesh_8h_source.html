351 else if (v2_id == m_indices[j])
352 f[1] = j;
353 else if (v3_id == m_indices[j])

965 // ---- OPERATION ACCOUNTING (counting only) ----
966 // Every call into split_edge, collapse_edge, swap_edge, swap_edge_44, swap_edge_56 and
967 // swap_face reports `attempt` on entry and exactly ONE other event on the way out: `committed`,
968 // or the place that refused it. The application's hooks are two of those places (before_hook,
969 // after_hook); why a hook said no is the hook's own to count. So for each kind, attempt equals
970 // the sum of the other events, and a subclass that counts can check both that identity and a
971 // hook's refusals against the reasons the hook counted. Added because the refusals made here,
972 // below every hook, were visible only as a difference: on the deliverable cube at
973 // target_distance_rel 1e-2 / front_conv_rel 1e-4, 521142 of the 542123 edge swaps that
974 // passed every before-hook in turn 5 were refused inside this class with no reason recorded.
975 //
976 // The default does nothing and no site reads a count: the mesh behaves identically with or
977 // without an override.
978 enum class OpKind : int { split, collapse, swap_32, swap_44, swap_56, swap_face, COUNT };
979 enum class OpEvent : int {
980 attempt,
981 before_hook, // the application's *_before said no
982 link_condition, // collapse: the classical link condition (m_collapse_check_link_condition)
983 no_return_tet, // collapse: both probes for the return tuple hit a boundary face
984 duplicate_tet, // a new tet repeats an existing one (collapse manifold check, 4-4, 5-6)
985 topology, // collapse: m_collapse_check_topology found a face missing
986 valence, // edge swap: the edge does not have this swap's number of tets around it
987 boundary, // edge swap: the edge's ring of tets is open; face swap: a boundary face
988 exists, // 3-2: the new face already exists; face swap: the new edge already exists
989 no_case_allowed, // 4-4 / 5-6: every candidate was vetoed by *_accept_case
990 no_better_case, // 4-4 / 5-6: no allowed candidate scored below the current cells
991 out_of_slots, // the preallocated vertex or tet slots ran out
992 after_hook, // the application's *_after said no, so the operation was rolled back
993 invariants, // invariants() said no after the after-hook passed; rolled back
994 committed,
995 screened, // a dry run passed every check that precedes the change (see dry_run())
996 COUNT
997 };

1468 void remove_tets_by_ids(const std::vector<size_t>& tids)
1469 {
1470 // A face or edge attribute lives at ONE canonical slot, (lowest incident tet) * 4
1471 // or 6 + local index. Deleting a tet that is a simplex's canonical owner silently
1472 // re-canonicalizes the simplex onto a surviving tet whose cell was never
1473 // maintained -- the operations only ever write the canonical slot -- so the
1474 // surviving cell holds stale garbage. Measured: after filtering, interior faces
1475 // read back m_is_surface_fs = true 0.12 (eps 0.09) from the real surface, and the
1476 // extracted "tracked surface" grows phantom triangles. Migrate every attribute
1477 // whose owner dies to the simplex's new owner BEFORE touching the connectivity.
1478 if (p_face_attrs != nullptr || p_edge_attrs != nullptr) {
1479 std::vector<bool> removed(tet_capacity(), false);
1480 for (const size_t tid : tids) {
1481 removed[tid] = true;
1482 }
1483 for (const size_t tid : tids) {
1484 const auto& tconn = m_tet_connectivity[tid];
1485 if (p_face_attrs != nullptr) {
1486 for (int lf = 0; lf < 4; ++lf) {
1487 std::array<size_t, 3> vids = {
1488 {tconn[m_local_faces[lf][0]],
1489 tconn[m_local_faces[lf][1]],
1490 tconn[m_local_faces[lf][2]]}};
1491 // The face's other incident tet, if any.
1492 auto common = set_intersection(
1493 m_vertex_connectivity[vids[0]].m_conn_tets,
1494 m_vertex_connectivity[vids[1]].m_conn_tets);
1495 common =
1496 set_intersection(common, m_vertex_connectivity[vids[2]].m_conn_tets);
1497 size_t other = std::numeric_limits<size_t>::max();
1498 for (const size_t t : common) {
1499 if (t != tid) {
1500 other = t;
1501 }
1502 }
1503 // Migrate only when this tet was the canonical owner (tid < other)
1504 // and the other side survives; other < tid means the attribute
1505 // already lives on the survivor.
1506 if (other == std::numeric_limits<size_t>::max() || removed[other] ||
1507 other < tid) {
1508 continue;
1509 }
1510 std::array<size_t, 3> sorted = vids;
1511 std::sort(sorted.begin(), sorted.end());
1512 for (int lo = 0; lo < 4; ++lo) {
1513 std::array<size_t, 3> ovids = {
1514 {m_tet_connectivity[other][m_local_faces[lo][0]],
1515 m_tet_connectivity[other][m_local_faces[lo][1]],
1516 m_tet_connectivity[other][m_local_faces[lo][2]]}};
1517 std::sort(ovids.begin(), ovids.end());
1518 if (ovids == sorted) {
1519 p_face_attrs->move(tid * 4 + lf, other * 4 + lo);
1520 break;
1521 }
1522 }
1523 }
1524 }
1525 if (p_edge_attrs != nullptr) {
1526 for (int le = 0; le < 6; ++le) {
1527 const size_t v0 = tconn[m_local_edges[le][0]];
1528 const size_t v1 = tconn[m_local_edges[le][1]];
1529 const auto fan = set_intersection(
1530 m_vertex_connectivity[v0].m_conn_tets,
1531 m_vertex_connectivity[v1].m_conn_tets);
1532 // Only the canonical owner migrates, once, to the lowest survivor.
1533 if (fan.empty() || fan.front() != tid) {
1534 continue;
1535 }
1536 size_t new_owner = std::numeric_limits<size_t>::max();
1537 for (const size_t t : fan) {
1538 if (!removed[t]) {
1539 new_owner = t; // fan is sorted ascending; first kept wins
1540 break;
1541 }
1542 }
1543 if (new_owner == std::numeric_limits<size_t>::max()) {
1544 continue; // the edge dies with its whole fan
1545 }
1546 for (int lo = 0; lo < 6; ++lo) {
1547 const size_t w0 = m_tet_connectivity[new_owner][m_local_edges[lo][0]];
1548 const size_t w1 = m_tet_connectivity[new_owner][m_local_edges[lo][1]];
1549 if ((w0 == v0 && w1 == v1) || (w0 == v1 && w1 == v0)) {
1550 p_edge_attrs->move(tid * 6 + le, new_owner * 6 + lo);
1551 break;
1552 }
1553 }
1554 }
1555 }
1556 }
1557 }
1558
1559 for (size_t tid : tids) {
1560 m_tet_connectivity[tid].m_is_removed = true;
1561 for (int j = 0; j < 4; j++)
1562 vector_erase(m_vertex_connectivity[m_tet_connectivity[tid][j]].m_conn_tets, tid);
1563 }
1564 for (auto& v : m_vertex_connectivity) {
1565 if (v.m_is_removed) continue;
1566 if (v.m_conn_tets.empty()) v.m_is_removed = true;
1567 }
1568 if (m_surface_history) resync_surface_history();
