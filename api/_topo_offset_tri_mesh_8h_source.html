2370 const std::array<size_t, 2>& vids,
2371 const EdgeAttributes& /*attr*/) const override
2372 {
2373 // Boundary geometry first: a segment on any tag-region boundary may not
2374 // drift out of that boundary's tube, and one on several boundaries -- a junction -- is
2375 // held in their intersection. The mask carries the input complex too, since every complex
2376 // simplex lies on tag boundaries, and E_t is built from the input mesh before construction
2377 // touches it, so the per-tag tubes hold the as-loaded geometry.
2378 //
2379 // Keyed on the vertices, not the edge: every caller is an operation asking about a segment
2380 // it is about to create or has just created, whose own edge attributes are not written
2381 // yet. The endpoints' masks are, maintained by the operations themselves (AND at a split,
2382 // OR at a collapse).
2383 uint64_t mask = edge_mask(vids);
2384 bool all_offset = true;
2385 for (const size_t v : vids) {
2386 all_offset = all_offset && m_vertex_extra[v].m_is_on_offset;
2387 }
2388
2389 // The ambiguous case: both endpoints can be on region boundaries and on the offset front
2390 // at once -- a real state, and the point of tracking the two families separately. The
2391 // endpoint-mask AND is then necessary but not sufficient for the segment lying on a shared
2392 // boundary: two vertices on different junctions can share a tag bit by coincidence and be
2393 // joined by an offset chord nowhere near that tag's curve, which a container it was never
2394 // meant to satisfy then refuses to refine. Ask the edge's own class, the only record that
2395 // distinguishes a chord from a boundary.
2396 //
2397 // Reading the slot is safe here and would not be unconditionally: the base checks a
2398 // split's two child segments before writing their attributes, so their slots still hold a
2399 // recycled edge's class. A split child never reaches this branch -- its mask and its
2400 // offset flag are mutually exclusive by construction, so one of `mask` and `all_offset` is
2401 // always empty -- and the m_is_surface_fs guard leaves both constraints standing rather
2402 // than dropping one when a slot is illegible.
2403 if (mask != 0 && all_offset) {
2404 if (const auto found = try_tuple_from_edge(vids)) {
2405 const size_t eid = std::get<1>(*found);
2406 if (m_edge_attribute[eid].m_is_surface_fs) {
2407 if (edge_is_offset(eid)) {
2408 mask = 0; // an offset edge lies on no region boundary
2409 } else {
2410 all_offset = false; // a region edge is not the offset front
2411 }
2412 }
2413 }
2414 }
2415 // Both families compose: whatever holds this segment holds it at once. The offset side
2416 // holds only in the frozen-front final pass (see containment_for()); in the loop the
2417 // front is what the optimization moves, so it contributes nothing there.
2418 const std::shared_ptr<Envelope> base = containment_for(mask, all_offset);
2419 if (base || m_deform_tags.empty()) return base;
2420 // deform_others' ops-only tube: a released boundary is held by no mask -- its vertices
2421 // were freed so smoothing can carry the object -- which would leave the operations free
2422 // to decimate and reposition it. A segment the masks and the offset class do not claim,
2423 // but which lies on a released boundary by its incident faces' current tags, is held to
2424 // the tube around the boundary's current shape. A fresh split child can misread its
2425 // recycled face slots for this one check, and is at worst skipped or over-held once.
2426 if (const auto found = try_tuple_from_edge(vids)) {
2427 if (edge_borders_released_boundary(std::get<0>(*found))) return released_envelope();
2428 }
2429 return nullptr;
2430 }

2433 // as in 3D. Until 2026-11-11 this mesh anchored nothing and answered true. An isolated point
2434 // of a complex with faces arises only where two or more selected tags meet (the selection
2435 // labels an isolated simplex only where its face star is tag-heterogeneous), so the edges at
2436 // it are tag boundaries and the intersection of their tubes holds it
2437 // (smoothing_containment_envelope()); one of a complex without faces is anchored.

2452 std::shared_ptr<Envelope> smoothing_energy_envelope(const size_t vid) const override
2453 {
2454 if (m_vertex_extra[vid].m_is_on_offset && !vertex_is_on_region(vid)) {
2455 return nullptr;
2456 }
2457 const uint64_t mask = vertex_boundary_mask(vid);
2458 if (mask == 0) {
2459 // Reachable only for a construction artefact (a wall-chord midpoint flagged
2460 // on-surface with disjoint endpoint masks); its containment is vacuous too, so no
2461 // pull is behavior-neutral. Not an error.
2462 return nullptr;
2463 }
2464 std::shared_ptr<Envelope> best;
2465 double worst_d2 = -1.;
2466 for (const auto& [tag, env] : m_tag_envelopes) {
2467 const auto it = m_tag_bit.find(tag);

2812 void for_each_edge_sample(const Vector2d& p0, const Vector2d& p1, Visit&& visit) const
2813 {
2814 const int k = m_offset_params.stencil_order;
2815 if (k < 0) return;
2816
2817 const auto emit = [&](const double wa, const double wb) {
2818 visit(Vector2d(wa * p0 + wb * p1), wa, wb);
2819 };
2820 // Order 0 is the two ENDS alone, each end's own placement error.
2821 if (k == 0) {
2822 emit(1., 0.);
2823 emit(0., 1.);
2824 return;
2825 }
2826 // Order k >= 1: the vertices of the chord cut into n = 2^(k-1) pieces, then the midpoint
2827 // of each piece -- 2n + 1 points: 3, 5, 9, 17, ...
2828 const int n = 1 << (k - 1);
2829 const double dn = double(n);
2830 for (int i = n; i >= 0; --i) {
2831 emit(double(i) / dn, double(n - i) / dn);
2832 }
2833 const double d2n = 2. * dn;
2834 for (int i = n - 1; i >= 0; --i) {
2835 emit((2. * i + 1.) / d2n, (2. * (n - 1 - i) + 1.) / d2n);
2836 }
2837 }
