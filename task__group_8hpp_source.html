80 auto& slot = current();
81 worker_pool* p = slot.load(std::memory_order_acquire);
82 if (p != nullptr) return *p;
83 // Created lazily, so that a forked child can start over. Threads racing to create it
84 // agree on one; the losers' pools have no workers yet and are simply deleted.
85 auto* fresh = new worker_pool();
86 if (slot.compare_exchange_strong(p, fresh, std::memory_order_acq_rel)) return *fresh;
87 delete fresh;
88 return *p;
89 }

92 void submit(std::unique_ptr<job> j)
93 {
94 {
95 std::unique_lock<std::mutex> lock(m_mutex);
96 // Every queued job must have an idle worker of its own: a worker that is busy may be
97 // running a task that waits for this one. A worker being started is already owed to
98 // the job that started it, so it does not count as free here: another thread taking
99 // it in the meantime would leave that job with no worker. Waking an idle worker that
100 // then finds the queue already drained by a worker finishing early is harmless; it
101 // waits again.
102 if (m_jobs.size() + m_spawning < m_idle) {
103 m_jobs.push_back(std::move(j));
104 lock.unlock();
105 m_cv.notify_one();
106 return;
107 }
108 ++m_spawning;
109 }
110 // No idle worker is free for this job. Start one BEFORE queueing the job, so that a
111 // failure to start it (std::system_error) leaves nothing behind for wait() to hang on.
112 // If an idle worker turned up in the meantime, one of the two just waits for the next job.
113 try {
114 std::thread([this] { work(); }).detach();
115 } catch (...) {
116 std::lock_guard<std::mutex> lock(m_mutex);
117 --m_spawning;
118 throw;
119 }
120 {
121 std::lock_guard<std::mutex> lock(m_mutex);
122 ++m_workers;
123 --m_spawning;
124 m_jobs.push_back(std::move(j));
125 }
126 m_cv.notify_one();
127 }

152 [[noreturn]] void work()
153 {
154 std::unique_lock<std::mutex> lock(m_mutex);
155 ++m_idle;
156 for (;;) {
157 m_cv.wait(lock, [this] { return !m_jobs.empty(); });
158 --m_idle;
159 std::unique_ptr<job> j = std::move(m_jobs.front());
160 m_jobs.pop_front();
161 lock.unlock();
162 // A task expects round-to-nearest, the mode the code that created its thread ran in.
163 // A reused worker starts in whatever the previous task left behind, and the exact
164 // predicates switch the rounding mode around their interval filters, so an exception
165 // in the middle of one would leak an upward mode into every later task on this thread.
166 std::fesetround(FE_TONEAREST);
167 j->run();
168 // Idle again BEFORE the task is reported finished: once it is, wait() may return and
169 // its caller start the next group, which must find this worker free rather than start
170 // another one. Until it waits again this worker does nothing that can block.
171 lock.lock();
172 ++m_idle;
173 lock.unlock();
174 j->finish();
175 j.reset();
176 lock.lock();
177 }
178 }

190// ---------------------------------------------------------------------------
191// task_group: replaces tbb::task_group. run() hands the task to a worker of the persistent
192// pool above (one worker per outstanding task, so the semantics are a thread per task);
193// wait() blocks until every task run since the last wait has finished, then rethrows the first
194// exception any of them threw.
195// ---------------------------------------------------------------------------
196class task_group
197{
198 std::mutex m_mutex;
199 std::condition_variable m_done;
200 size_t m_pending = 0;
201 std::exception_ptr m_eptr;
202
203public:

208 template <typename F>
209 void run(F&& f)
210 {
211 // The task is copied or moved into its job here, on the calling thread: a task whose copy
212 // or move throws makes run() throw -- as std::thread's constructor did -- before anything
213 // is counted or queued.
214 auto j = std::make_unique<task_job<std::decay_t<F>>>(*this, std::forward<F>(f));
215 {
216 std::lock_guard<std::mutex> lock(m_mutex);
217 ++m_pending;
218 }
219 try {
220 detail::worker_pool::instance().submit(std::move(j));
221 } catch (...) {
222 // Nothing was queued (see worker_pool::submit), so nothing will decrement this.
223 std::lock_guard<std::mutex> lock(m_mutex);
224 --m_pending;
225 throw;
226 }
227 }
228
229 void wait()
230 {
231 std::unique_lock<std::mutex> lock(m_mutex);
232 m_done.wait(lock, [this] { return m_pending == 0; });
233 if (m_eptr) {
234 std::exception_ptr e = m_eptr;
235 m_eptr = nullptr;
236 lock.unlock();
237 std::rethrow_exception(e);
238 }
239 }
240
241 ~task_group()
242 {
243 std::unique_lock<std::mutex> lock(m_mutex);
244 m_done.wait(lock, [this] { return m_pending == 0; });
245 }

262 void run() noexcept override
263 {
264 try {
265 (*m_fn)();
266 } catch (...) {
267 m_group.record(std::current_exception());
268 }
269 // Destroy the task's own state before it is reported finished: once it is, wait()
270 // may return and the caller may tear down whatever the task captured by reference.
271 m_fn.reset();
272 }
273
274 // After this the job touches nothing of the group's.
275 void finish() noexcept override { m_group.finish_one(); }
276 };
277
278 void record(std::exception_ptr e)
279 {
280 std::lock_guard<std::mutex> lock(m_mutex);
281 if (!m_eptr) {
282 m_eptr = std::move(e);
283 }
284 }

286 void finish_one()
287 {
288 // Notify while holding the lock, so wait() cannot return -- and the group cannot be
289 // destroyed -- until this thread has let go of the group's mutex.
290 std::lock_guard<std::mutex> lock(m_mutex);
291 if (--m_pending == 0) {
292 m_done.notify_all();
293 }
294 }
