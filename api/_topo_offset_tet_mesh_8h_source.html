2710 std::shared_ptr<polysolve::nonlinear::Problem> smoothing_extra_energy(
2711 const size_t vid) const override
2712 {
2713 // TetWild's rules (m_op_rules): no extra energy, so the engine's smoother minimises the
2714 // raw sum of AMIPS over the one-ring (AMIPSEnergy3D), as TetWild's does.
2715 if (tetwild_rules()) return nullptr;
2716 auto sum = std::make_shared<optimization::EnergySum>();
2717 sum->add_energy(amips3_energy(vid));
2718 // Before the march (repulsion_smoothing()) there is no front.
2719 if (is_repulsion_vertex(vid)) {
2720 sum->add_energy(repulsion_energy(vid));
2721 return sum;
2722 }
2723 if (vertex_carries_offset_term(vid)) {
2724 sum->add_energy(front_energy(vid));
2725 } else if (!vertex_is_held_on_input(vid)) {
2726 // E's band term moves with this vertex too -- except a vertex held on the input.
2727 // There d = 0 at the vertex and d has a kink: the term's slope along the input's
2728 // normal jumps across the surface, and its curvature there is 0 (or ~1e10 near an
2729 // input edge), so the Newton step along the normal is set by AMIPS alone and lands
2730 // far off; the projected step then discards the normal motion anyway. MEASURED
2731 // 2026-10-09, cube, serial, 3 turns: with the term these solves took 24.6 Newton
2732 // iterations and ~20 line-search trials per iteration (72% of the smoother's value
2733 // calls), without it 3.9 and 1; 19.3 s -> 11.3 s. E still guards the move:
2734 // smooth_nonfront_vertex() vetoes on the ring's sum of tet_energy().
2735 // Under "exact", not where it cancels (BAND TERM THAT CANCELS, tet_energy()): a ring
2736 // of band cells clear of the input's bisectors tiles the star of vid, so the term is a
2737 // constant, and its cells' exact integrals would add only work and their rounding to
2738 // every Newton step. The veto's own test (cancels_before in smooth_nonfront_vertex()).
2739 // Constant wherever the move can end, not only here: vid is smoothed, so not on the
2740 // domain wall (smooth_before()), and its ring closes around it; the engine keeps no
2741 // position that inverts a ring cell, so vid stays inside the convex hull of its link;
2742 // and the test reads the points only through their box, their least side of each
2743 // bisector polytope's planes, and min_P max d_P (d_P convex), none of which a point
2744 // inside that hull changes. So the test here certifies every position the smoother
2745 // can keep. "front_flux" and "exact_swept" leave the constant out by their own forms
2746 // (band_volume_energy_at()). MEASURED (cube 1e-2 / 1e-4, 15 rounds, serial, two runs
2747 // each): 659 of 46,594 non-front solves had such a ring; at every one's end the
2748 // veto's test found it clear again, and every one was accepted; smoothing 7.7-7.9 s ->
2749 // 7.4-7.5 s.
2750 if (m_offset_params.band_volume_rule == "exact" &&
2751 band_term_cancels(get_one_ring_tids_for_vertex(vid)))
2752 return sum;
2753 if (const auto bv = band_volume_energy_at(vid)) sum->add_energy(bv);
2754 }
2755 return sum;
2756 }
