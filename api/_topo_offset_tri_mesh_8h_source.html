2412 const std::array<size_t, 2>& vids,
2413 const EdgeAttributes& /*attr*/) const override
2414 {
2415 // Boundary geometry first: a segment on any tag-region boundary may not
2416 // drift out of that boundary's tube, and one on several boundaries -- a junction -- is
2417 // held in their intersection. The mask carries the input complex too, since every complex
2418 // simplex lies on tag boundaries, and E_t is built from the input mesh before construction
2419 // touches it, so the per-tag tubes hold the as-loaded geometry.
2420 //
2421 // Keyed on the vertices, not the edge: every caller is an operation asking about a segment
2422 // it is about to create or has just created, whose own edge attributes are not written
2423 // yet. The endpoints' masks are, maintained by the operations themselves (AND at a split,
2424 // OR at a collapse).
2425 LabelIds labels = edge_boundary_labels(vids);
2426 bool all_offset = true;
2427 for (const size_t v : vids) {
2428 all_offset = all_offset && m_vertex_extra[v].m_is_on_offset;
2429 }
2430
2431 // The ambiguous case: both endpoints can be on region boundaries and on the offset front
2432 // at once -- a real state, and the point of tracking the two families separately. The
2433 // endpoint-mask AND is then necessary but not sufficient for the segment lying on a shared
2434 // boundary: two vertices on different junctions can share a tag bit by coincidence and be
2435 // joined by an offset chord nowhere near that tag's curve, which a container it was never
2436 // meant to satisfy then refuses to refine. Ask the edge's own class, the only record that
2437 // distinguishes a chord from a boundary.
2438 //
2439 // Reading the slot is safe here and would not be unconditionally: the base checks a
2440 // split's two child segments before writing their attributes, so their slots still hold a
2441 // recycled edge's class. A split child never reaches this branch -- its mask and its
2442 // offset flag are mutually exclusive by construction, so one of `mask` and `all_offset` is
2443 // always empty -- and the m_is_surface_fs guard leaves both constraints standing rather
2444 // than dropping one when a slot is illegible.
2445 if (!labels.none() && all_offset) {
2446 if (const auto found = try_tuple_from_edge(vids)) {
2447 const size_t eid = std::get<1>(*found);
2448 if (m_edge_attribute[eid].m_is_surface_fs) {
2449 if (edge_is_offset(eid)) {
2450 labels.clear(); // an offset edge lies on no region boundary
2451 } else {
2452 all_offset = false; // a region edge is not the offset front
2453 }
2454 }
2455 }
2456 }
2457 // Both families compose: whatever holds this segment holds it at once. The offset side
2458 // holds only in the frozen-front final pass (see containment_for()); in the loop the
2459 // front is what the optimization moves, so it contributes nothing there.
2460 const std::shared_ptr<Envelope> base = containment_for(labels, all_offset);
2461 if (base || m_deform_tags.empty()) return base;
2462 // deform_others' ops-only tube: a released boundary is held by no mask -- its vertices
2463 // were freed so smoothing can carry the object -- which would leave the operations free
2464 // to decimate and reposition it. A segment the masks and the offset class do not claim,
2465 // but which lies on a released boundary by its incident faces' current tags, is held to
2466 // the tube around the boundary's current shape. A fresh split child can misread its
2467 // recycled face slots for this one check, and is at worst skipped or over-held once.
2468 if (const auto found = try_tuple_from_edge(vids)) {
2469 if (edge_borders_released_boundary(std::get<0>(*found))) return released_envelope();
2470 }
2471 return nullptr;
2472 }

2838 template <typename Visit>
2839 void for_each_edge_sample(const Vector2d& p0, const Vector2d& p1, Visit&& visit) const
2840 {
2841 const int k = m_offset_params.stencil_order;
2842 if (k < 0) return;
2843
2844 const auto emit = [&](const double wa, const double wb) {
2845 visit(Vector2d(wa * p0 + wb * p1), wa, wb);
2846 };
2847 // Order 0 is the two ENDS alone, each end's own placement error.
2848 if (k == 0) {
2849 emit(1., 0.);
2850 emit(0., 1.);
2851 return;
2852 }
2853 // Order k >= 1: the vertices of the chord cut into n = 2^(k-1) pieces, then the midpoint
2854 // of each piece -- 2n + 1 points: 3, 5, 9, 17, ...
2855 const int n = 1 << (k - 1);
2856 const double dn = double(n);
2857 for (int i = n; i >= 0; --i) {
2858 emit(double(i) / dn, double(n - i) / dn);
2859 }
2860 const double d2n = 2. * dn;
2861 for (int i = n - 1; i >= 0; --i) {
2862 emit((2. * i + 1.) / d2n, (2. * (n - 1 - i) + 1.) / d2n);
2863 }
2864 }
