28 mesh.get_tets(),

80inline OffsetRows offset_rows(const TopoOffsetTetMesh& mesh)
81{
82 OffsetRows out;
83 const std::vector<std::string> labels =
84 io::tag_labels(mesh.m_tag_id_to_name, mesh.m_tags_count);
85 std::vector<int> row(mesh.vert_capacity(), -1);
86 std::vector<Vector3d> V;
87 std::vector<std::array<int, 3>> F;
88 for (const TetMesh::Tuple& f : mesh.get_faces()) {
89 const auto opp = f.switch_tetrahedron(mesh);
90 if (!opp ||
91 mesh.cell_is_offset_band(f.tid(mesh)) == mesh.cell_is_offset_band(opp->tid(mesh)))
92 continue;
93 const auto& a = mesh.m_tet_attribute[f.tid(mesh)].tag;
94 const auto& b = mesh.m_tet_attribute[opp->tid(mesh)].tag;
95 std::vector<std::string> names;
96 for (const auto& [x, y] : {std::pair(&a, &b), std::pair(&b, &a)}) {
97 for (const int64_t t : *x)
98 if (t > 0 && !y->count(t) && size_t(t) < labels.size())
99 names.push_back(labels[size_t(t)]); // ambient carries no label
100 }
101 if (names.empty()) continue;
102 out.names.push_back(std::move(names));
103 std::array<int, 3> r;
104 const auto vs = mesh.get_face_vids(f);
105 for (int k = 0; k < 3; ++k) {
106 int& v = row[vs[k]];
107 if (v < 0) {
108 v = int(V.size());
109 V.push_back(mesh.m_vertex_attribute[vs[k]].m_posf);
110 }
111 r[k] = v;
112 }
113 F.push_back(r);
114 }
115 out.V.resize(Eigen::Index(V.size()), 3);
116 for (size_t i = 0; i < V.size(); ++i) out.V.row(Eigen::Index(i)) = V[i].transpose();
117 out.F.resize(Eigen::Index(F.size()), 3);
118 for (size_t i = 0; i < F.size(); ++i)
119 for (int k = 0; k < 3; ++k) out.F(Eigen::Index(i), k) = F[i][k];
120 return out;
121}

127inline MeshData with_offset_surface(const MeshData& reference, const OffsetRows& offset)
128{
129 struct Block
130 {
131 MatrixXd V = MatrixXd(0, 3);
132 MatrixXi C;
133 std::vector<InputSet> sets;
134 std::vector<std::string> names;
135 };
136 // The `type` rows of the reference on vertices of their own, in first use, with their labels.
137 const auto block = [&](const CellType type) {
138 Block b;
139 const auto it = reference.cells.find(type);
140 if (it == reference.cells.end()) return b;
141 const auto& C = it->second.connectivity;
142 std::vector<int> row(size_t(reference.V.rows()), -1);
143 std::vector<Eigen::Index> used;
144 b.C.resize(C.rows(), C.cols());
145 for (Eigen::Index r = 0; r < C.rows(); ++r) {
146 for (Eigen::Index k = 0; k < C.cols(); ++k) {
147 int& v = row[size_t(C(r, k))];
148 if (v < 0) {
149 v = int(used.size());
150 used.push_back(Eigen::Index(C(r, k)));
151 }
152 b.C(r, k) = v;
153 }
154 }
155 b.V.resize(Eigen::Index(used.size()), 3);
156 for (size_t i = 0; i < used.size(); ++i)
157 b.V.row(Eigen::Index(i)) = reference.V.row(used[i]);
158 const auto tags = it->second.attributes.find("tags");
159 if (tags == it->second.attributes.end()) return b;
160 b.names = tags->second.labels;
161 const auto member = tags->second.integers();
162 b.sets.assign(size_t(C.rows()), 0);
163 for (Eigen::Index r = 0; r < C.rows(); ++r) {
164 for (size_t j = 0; j < b.names.size() && j < size_t(SurfaceTagAttributes::max_inputs);
165 ++j) {
166 if (member(r, Eigen::Index(j))) b.sets[size_t(r)] |= InputSet(1) << j;
167 }
168 }
169 return b;
170 };
171 Block tri = block(CellType::Triangle);
172 tri.sets.resize(size_t(tri.C.rows()), 0);
173 for (const auto& names : offset.names) {
174 InputSet set = 0;
175 for (const std::string& name : names) {
176 auto k = std::find(tri.names.begin(), tri.names.end(), name);
177 if (k == tri.names.end()) k = tri.names.insert(tri.names.end(), name);
178 const size_t j = size_t(k - tri.names.begin());
179 if (j < size_t(SurfaceTagAttributes::max_inputs)) set |= InputSet(1) << j;
180 }
181 tri.sets.push_back(set);
182 }
183 const Eigen::Index nv = tri.V.rows(), nf = tri.C.rows();
184 tri.V.conservativeResize(nv + offset.V.rows(), 3);
185 tri.V.bottomRows(offset.V.rows()) = offset.V;
186 tri.C.conservativeResize(nf + offset.F.rows(), 3);
187 tri.C.bottomRows(offset.F.rows()) = (offset.F.array() + int(nv)).matrix();
188 MeshData out;
189 io::append_envelope_block(
190 out,
191 io::geometry_data(tri.V, tri.C, CellType::Triangle),
192 CellType::Triangle,
193 tri.sets,
194 tri.names);
195 const Block curves = block(CellType::Edge);
196 if (curves.C.rows() > 0) {
197 io::append_envelope_block(
198 out,
199 io::geometry_data(curves.V, curves.C, CellType::Edge),
200 CellType::Edge,
201 curves.sets,
202 curves.names);
203 }
204 const Block points = block(CellType::Point);
205 std::vector<Vector3d> P(size_t(points.V.rows()));
206 for (size_t i = 0; i < P.size(); ++i) P[i] = points.V.row(Eigen::Index(i)).transpose();
207 io::append_envelope_points(out, P);
208 return out;
209}
