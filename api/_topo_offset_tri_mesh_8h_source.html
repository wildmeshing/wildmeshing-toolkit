2425 std::shared_ptr<Envelope> surface_envelope_for_edge(
2426 const std::array<size_t, 2>& vids,
2427 const EdgeAttributes& /*attr*/) const override
2428 {
2429 // Boundary geometry first: a segment on any tag-region boundary may not
2430 // drift out of that boundary's tube, and one on several boundaries -- a junction -- is
2431 // held in their intersection. The mask carries the input complex too, since every complex
2432 // simplex lies on tag boundaries, and E_t is built from the input mesh before construction
2433 // touches it, so the per-tag tubes hold the as-loaded geometry.
2434 //
2435 // Keyed on the vertices, not the edge: every caller is an operation asking about a segment
2436 // it is about to create or has just created, whose own edge attributes are not written
2437 // yet. The endpoints' masks are, maintained by the operations themselves (AND at a split,
2438 // OR at a collapse).
2439 LabelIds labels = edge_boundary_labels(vids);
2440 bool all_offset = true;
2441 for (const size_t v : vids) {
2442 all_offset = all_offset && m_vertex_extra[v].m_is_on_offset;
2443 }
2444
2445 // The ambiguous case: both endpoints can be on region boundaries and on the offset front
2446 // at once -- a real state, and the point of tracking the two families separately. The
2447 // endpoint-mask AND is then necessary but not sufficient for the segment lying on a shared
2448 // boundary: two vertices on different junctions can share a tag bit by coincidence and be
2449 // joined by an offset chord nowhere near that tag's curve, which a container it was never
2450 // meant to satisfy then refuses to refine. Ask the edge's own class, the only record that
2451 // distinguishes a chord from a boundary.
2452 //
2453 // Reading the slot is safe here and would not be unconditionally: the base checks a
2454 // split's two child segments before writing their attributes, so their slots still hold a
2455 // recycled edge's class. A split child never reaches this branch -- its mask and its
2456 // offset flag are mutually exclusive by construction, so one of `mask` and `all_offset` is
2457 // always empty -- and the m_is_surface_fs guard leaves both constraints standing rather
2458 // than dropping one when a slot is illegible.
2459 if (!labels.none() && all_offset) {
2460 if (const auto found = try_tuple_from_edge(vids)) {
2461 const size_t eid = std::get<1>(*found);
2462 if (m_edge_attribute[eid].m_is_surface_fs) {
2463 if (edge_is_offset(eid)) {
2464 labels.clear(); // an offset edge lies on no region boundary
2465 } else {
2466 all_offset = false; // a region edge is not the offset front
2467 }
2468 }
2469 }
2470 }
2471 // Both families compose: whatever holds this segment holds it at once. The offset side
2472 // holds only in the frozen-front final pass (see containment_for()); in the loop the
2473 // front is what the optimization moves, so it contributes nothing there.
2474 const std::shared_ptr<Envelope> base = containment_for(labels, all_offset);
2475 if (base || m_deform_tags.empty()) return base;
2476 // deform_others' ops-only tube: a released boundary is held by no mask -- its vertices
2477 // were freed so smoothing can carry the object -- which would leave the operations free
2478 // to decimate and reposition it. A segment the masks and the offset class do not claim,
2479 // but which lies on a released boundary by its incident faces' current tags, is held to
2480 // the tube around the boundary's current shape. A fresh split child can misread its
2481 // recycled face slots for this one check, and is at worst skipped or over-held once.
2482 if (const auto found = try_tuple_from_edge(vids)) {
2483 if (edge_borders_released_boundary(std::get<0>(*found))) return released_envelope();
2484 }
2485 return nullptr;
2486 }

2852 template <typename Visit>
2853 void for_each_edge_sample(const Vector2d& p0, const Vector2d& p1, Visit&& visit) const
2854 {
2855 const int k = m_offset_params.stencil_order;
2856 if (k < 0) return;
2857
2858 const auto emit = [&](const double wa, const double wb) {
2859 visit(Vector2d(wa * p0 + wb * p1), wa, wb);
2860 };
2861 // Order 0 is the two ENDS alone, each end's own placement error.
2862 if (k == 0) {
2863 emit(1., 0.);
2864 emit(0., 1.);
2865 return;
2866 }
2867 // Order k >= 1: the vertices of the chord cut into n = 2^(k-1) pieces, then the midpoint
2868 // of each piece -- 2n + 1 points: 3, 5, 9, 17, ...
2869 const int n = 1 << (k - 1);
2870 const double dn = double(n);
2871 for (int i = n; i >= 0; --i) {
2872 emit(double(i) / dn, double(n - i) / dn);
2873 }
2874 const double d2n = 2. * dn;
2875 for (int i = n - 1; i >= 0; --i) {
2876 emit((2. * i + 1.) / d2n, (2. * (n - 1 - i) + 1.) / d2n);
2877 }
2878 }
