145 void run()
146 {
147 for (size_t o = 0; o < m_originals.size(); ++o) {
148 const size_t cell = *m_originals[o].children.begin();
149 m_label[cell] = finalized_label(cell);
150 m_stats.n_original_finalized += m_label[cell] >= 0 ? 1 : 0;
151 }
152
153 std::vector<std::pair<size_t, size_t>> children;
154 for (size_t o = 0; o < m_originals.size(); ++o) {
155 Original& t = m_originals[o];
156 if (std::all_of(t.children.begin(), t.children.end(), [&](const size_t cell) {
157 return m_label[cell] >= 0;
158 })) {
159 continue;
160 }
161 const std::vector<size_t> functions = undominated(t);
162 m_stats.n_dominated += m_n_functions[t.comp] - functions.size();
163 for (size_t p = 0; p < functions.size(); ++p) {
164 for (size_t q = p + 1; q < functions.size(); ++q) {
165 const size_t i = functions[p];
166 const size_t j = functions[q];
167 ++m_stats.n_pairs;
168 std::deque<size_t> queue;
169 for (const size_t cell : t.children) {
170 if (m_label[cell] < 0) {
171 queue.push_back(cell);
172 }
173 }
174 while (!queue.empty()) {
175 const size_t cell = queue.front();
176 queue.pop_front();
177 if (t.children.count(cell) == 0 || m_label[cell] >= 0) {
178 continue;
179 }
180 const auto [a, b] = crossing_edge(t.comp, cell, i, j);
181 if (a == size_t(-1)) {
182 continue;
183 }
184 const Vertex& A = vertex(t.comp, a);
185 const Vertex& B = vertex(t.comp, b);
186 const double delta_a = A.d[i] - A.d[j];
187 const double delta_b = B.d[i] - B.d[j];
188 const double s = delta_a / (delta_a - delta_b);
189 const size_t x = m_mesh.split(a, b, delta_a, delta_b, children);
190 if (x == size_t(-1)) {
191 ++m_stats.n_refused;
192 continue;
193 }
194 ++m_stats.n_split;
195 m_new_vertices.push_back(x);
196 m_made.emplace(x, Made{a, b, s, t.comp, i, j});
197 adopt(children, o, queue);
198 }
199 }
200 }
201 }
202
203 for (Original& t : m_originals) {
204 for (const size_t cell : t.children) {
205 ++m_stats.n_cells;
206 if (m_label[cell] < 0) {
207 m_label[cell] = smallest_at_centroid(t.comp, cell);
208 ++m_stats.n_fallback;
209 }
210 }
211 }

247 static int compare(const Vertex& v, const size_t k, const size_t l)
248 {

253 static std::vector<size_t> label_set(const Vertex& v)
254 {
255 const double least = *std::min_element(v.d.begin(), v.d.end());
256 std::vector<size_t> labels;

267 Vertex interpolate(
268 const Vertex& A,
269 const Vertex& B,
270 const double s,
271 const std::pair<size_t, size_t>* pair)
272 {
273 const size_t K = A.d.size();
274 Vertex x;
275 x.d.resize(K);
276 const double r = 1. - s;
277 for (size_t k = 0; k < K; ++k) {
278 x.d[k] = r * A.d[k] + s * B.d[k];
279 }
280 // the tie classes, each rooted at its smallest function
281 std::vector<size_t> root(K);
282 for (size_t k = 0; k < K; ++k) {
283 root[k] = k;
284 }
285 const auto find = [&](size_t k) {
286 while (root[k] != k) {
287 k = root[k] = root[root[k]];
288 }
289 return k;
290 };
291 const auto tie = [&](size_t k, size_t l) {
292 k = find(k);
293 l = find(l);
294 if (k > l) {
295 std::swap(k, l);
296 }
297 root[l] = k;
298 };
299 if (pair) {
300 tie(pair->first, pair->second);
301 }
302 for (bool tied = true; tied;) {
303 // each class takes its least value
304 std::vector<double> least(K, std::numeric_limits<double>::infinity());
305 for (size_t k = 0; k < K; ++k) {
306 least[find(k)] = std::min(least[find(k)], x.d[k]);
307 }
308 for (size_t k = 0; k < K; ++k) {
309 x.d[k] = least[find(k)];
310 }
311 // then each pair stays on the closed side its edge's ends share
312 tied = false;
313 for (size_t k = 0; k < K; ++k) {
314 for (size_t l = k + 1; l < K; ++l) {
315 if (find(k) == find(l)) {
316 continue;
317 }
318 const int sx = compare(x, k, l);
319 const int sa = compare(A, k, l);
320 const int sb = compare(B, k, l);
321 if ((sa >= 0 && sb >= 0 && sx < 0) || (sa <= 0 && sb <= 0 && sx > 0)) {
322 tie(k, l);
323 tied = true;
324 ++m_stats.n_closed;
325 }
326 }
327 }
328 }
329 return x;
330 }

334 const Vertex& vertex(const size_t c, const size_t v)
335 {
336 std::unordered_map<size_t, Vertex>& data = m_vertices[c];
337 if (const auto it = data.find(v); it != data.end()) {
338 return it->second;
339 }
340 Vertex x;
341 if (const auto it = m_made.find(v); it == m_made.end()) {
342 x.d.resize(m_n_functions[c]);
343 for (size_t k = 0; k < x.d.size(); ++k) {
344 x.d[k] = m_mesh.value(c, k, v);
345 }
346 } else {
347 const Made made = it->second;
348 const std::pair<size_t, size_t> pair(made.i, made.j);
349 // references into an unordered_map stay valid as it grows
350 const Vertex& A = vertex(c, made.a);
351 const Vertex& B = vertex(c, made.b);
352 x = interpolate(A, B, made.s, made.comp == c ? &pair : nullptr);
353 }
354 x.labels = label_set(x);
355 return data.emplace(v, std::move(x)).first->second;
356 }

359 int finalized_label(const size_t cell)
360 {
361 const size_t c = m_originals[m_orig[cell]].comp;
362 const Cell vs = m_mesh.corners(cell);
363 std::vector<size_t> common = vertex(c, vs[0]).labels;
364 for (int q = 1; q < N && !common.empty(); ++q) {
365 const std::vector<size_t>& labels = vertex(c, vs[q]).labels;
366 std::vector<size_t> both;
367 std::set_intersection(
368 common.begin(),
369 common.end(),
370 labels.begin(),
371 labels.end(),
372 std::back_inserter(both));
373 common = std::move(both);
374 }
375 return common.empty() ? -1 : int(common.front());
376 }

380 std::vector<size_t> undominated(const Original& t)
381 {
382 const size_t K = m_n_functions[t.comp];
383 std::array<const Vertex*, N> corner;
384 for (int q = 0; q < N; ++q) {
385 corner[q] = &vertex(t.comp, t.corners[q]);
386 }
387 // the functions smallest at a corner first: they dominate most of the others
388 std::vector<size_t> order;
389 for (int q = 0; q < N; ++q) {
390 order.insert(order.end(), corner[q]->labels.begin(), corner[q]->labels.end());
391 }
392 for (size_t l = 0; l < K; ++l) {
393 order.push_back(l);
394 }
395 std::vector<size_t> functions;
396 for (size_t k = 0; k < K; ++k) {
397 const bool dominated = std::any_of(order.begin(), order.end(), [&](const size_t l) {
398 if (l == k) {
399 return false;
400 }
401 for (int q = 0; q < N; ++q) {
402 if (compare(*corner[q], l, k) >= 0) {
403 return false;
404 }
405 }
406 return true;
407 });
408 if (!dominated) {
409 functions.push_back(k);
410 }
411 }
412 return functions;
413 }

417 std::pair<size_t, size_t>
418 crossing_edge(const size_t c, const size_t cell, const size_t i, const size_t j)
419 {
420 Cell vs = m_mesh.corners(cell);
421 std::sort(vs.begin(), vs.end());
422 std::array<int, N> signs;
423 for (int q = 0; q < N; ++q) {
424 signs[q] = compare(vertex(c, vs[q]), i, j);
425 }
426 for (int p = 0; p < N; ++p) {
427 for (int q = p + 1; q < N; ++q) {
428 if (signs[p] * signs[q] < 0) {
429 return {vs[p], vs[q]};
430 }
431 }
432 }
433 return {size_t(-1), size_t(-1)};
434 }

439 void adopt(
440 const std::vector<std::pair<size_t, size_t>>& children,
441 const size_t o,
442 std::deque<size_t>& queue)
443 {
444 // the parents' states first: a child may take its parent's id
445 std::map<size_t, std::pair<int, int>> parents;
446 for (const auto& [parent, child] : children) {
447 grow(parent);
448 parents.emplace(parent, std::make_pair(m_orig[parent], m_label[parent]));
449 }
450 for (const auto& [parent, state] : parents) {
451 if (state.first >= 0) {
452 m_originals[state.first].children.erase(parent);
453 }
454 }
455 for (const auto& [parent, child] : children) {
456 grow(child);
457 const auto [orig, label] = parents.at(parent);
458 m_orig[child] = orig;
459 m_label[child] = -1;
460 if (orig < 0) {
461 continue; // in no component
462 }
463 m_originals[orig].children.insert(child);
464 if (label >= 0) {
465 m_label[child] = label;
466 continue;
467 }
468 m_label[child] = finalized_label(child);
469 if (m_label[child] < 0 && size_t(orig) == o) {
470 queue.push_back(child);
471 }
472 }
473 }

477 int smallest_at_centroid(const size_t c, const size_t cell)
478 {
479 std::vector<double> sum(m_n_functions[c], 0.);
480 for (const size_t v : m_mesh.corners(cell)) {
481 const Vertex& x = vertex(c, v);
482 for (size_t k = 0; k < sum.size(); ++k) {
483 sum[k] += x.d[k];
484 }
485 }
486 return int(std::min_element(sum.begin(), sum.end()) - sum.begin());
487 }
488
489 void grow(const size_t cell)
490 {
491 if (cell >= m_orig.size()) {
492 m_orig.resize(cell + 1, -1);
493 m_label.resize(cell + 1, -1);
494 }
495 }
