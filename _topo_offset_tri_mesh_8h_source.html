1140 std::shared_ptr<SampleEnvelope> surface_envelope_for_edge(
1141 const std::array<size_t, 2>& vids) const override
1142 {
1143 // Boundary geometry first, in both phases: a segment on any tag-region boundary may not
1144 // drift out of that boundary's tube, and one on several boundaries -- a junction -- is
1145 // held in their intersection. The mask carries the input complex too, since every complex
1146 // simplex lies on tag boundaries, and E_t is built from the input mesh before construction
1147 // touches it, so the per-tag tubes hold the as-loaded geometry.
1148 //
1149 // Keyed on the vertices, not the edge: every caller is an operation asking about a segment
1150 // it is about to create or has just created, whose own edge attributes are not written
1151 // yet. The endpoints' masks are, maintained by the operations themselves (AND at a split,
1152 // OR at a collapse).
1153 uint64_t mask = edge_mask(vids);
1154 bool all_offset = true;
1155 for (const size_t v : vids) {
1156 all_offset = all_offset && m_vertex_extra[v].m_is_on_offset;
1157 }
1158
1159 // The ambiguous case: both endpoints can be on region boundaries and on the offset front
1160 // at once -- a real state, and the point of tracking the two families separately. The
1161 // endpoint-mask AND is then necessary but not sufficient for the segment lying on a shared
1162 // boundary: two vertices on different junctions can share a tag bit by coincidence and be
1163 // joined by an offset chord nowhere near that tag's curve, which a container it was never
1164 // meant to satisfy then refuses to refine. Ask the edge's own class, the only record that
1165 // distinguishes a chord from a boundary.
1166 //
1167 // Reading the slot is safe here and would not be unconditionally: the base checks a
1168 // split's two child segments before writing their attributes, so their slots still hold a
1169 // recycled edge's class. A split child never reaches this branch -- its mask and its
1170 // offset flag are mutually exclusive by construction, so one of `mask` and `all_offset` is
1171 // always empty -- and the m_is_surface_fs guard leaves both constraints standing rather
1172 // than dropping one when a slot is illegible.
1173 if (mask != 0 && all_offset) {
1174 if (const auto found = try_tuple_from_edge(vids)) {
1175 const size_t eid = std::get<1>(*found);
1176 if (m_edge_attribute[eid].m_is_surface_fs) {
1177 if (edge_is_offset(eid)) {
1178 mask = 0; // an offset edge lies on no region boundary
1179 } else {
1180 all_offset = false; // a region edge is not the offset front
1181 }
1182 }
1183 }
1184 }
1185 // Both families compose: whatever holds this segment holds it at once. Phase A holds the
1186 // offset where Phase B left it; Phase B is what moves it, so it contributes nothing there
1187 // -- and null before the offset exists at all, which is the pre-pass.
1188 const std::shared_ptr<SampleEnvelope> base = containment_for(mask, all_offset);
1189 if (base || m_deform_tags.empty()) return base;
1190 // deform_others' ops-only tube: a released boundary is held by no mask -- its vertices
1191 // were freed so smoothing can carry the object -- which would leave the operations free
1192 // to decimate and reposition it. A segment the masks and the offset class do not claim,
1193 // but which lies on a released boundary by its incident faces' current tags, is held to
1194 // the tube around the boundary's current shape. A fresh split child can misread its
1195 // recycled face slots for this one check, and is at worst skipped or over-held once.
1196 if (const auto found = try_tuple_from_edge(vids)) {
1197 if (edge_borders_released_boundary(std::get<0>(*found))) return released_envelope();
1198 }
1199 return nullptr;
1200 }

1232 std::shared_ptr<SampleEnvelope> smoothing_energy_envelope(const size_t vid) const override
1233 {
1234 if (m_vertex_extra[vid].m_is_on_offset && !vertex_is_on_region(vid)) {
1235 return nullptr;
1236 }
1237 const uint64_t mask = vertex_boundary_mask(vid);
1238 if (mask == 0) {
1239 // Reachable only for a construction artefact (a wall-chord midpoint flagged
1240 // on-surface with disjoint endpoint masks); its containment is vacuous too, so no
1241 // pull is behavior-neutral. Not an error.
1242 return nullptr;
1243 }
1244 std::shared_ptr<SampleEnvelope> best;
1245 double worst_d2 = -1.;
1246 for (const auto& [tag, env] : m_tag_envelopes) {
1247 const auto it = m_tag_bit.find(tag);
1248 if (it == m_tag_bit.end() || !(mask & (uint64_t(1) << it->second))) continue;
1249 if (!best) {
1250 best = env;
1251 if ((mask & (mask - 1)) == 0) break; // single bit: no violation contest to run
1252 worst_d2 = env->squared_distance(m_vertex_attribute[vid].m_posf);
