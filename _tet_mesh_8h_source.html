861 // ---- OPERATION ACCOUNTING (counting only) ----
862 // Every call into split_edge, collapse_edge, swap_edge, swap_edge_44, swap_edge_56 and
863 // swap_face reports `attempt` on entry and exactly ONE other event on the way out: `committed`,
864 // or the place that refused it. The application's hooks are two of those places (before_hook,
865 // after_hook); why a hook said no is the hook's own to count. So for each kind, attempt equals
866 // the sum of the other events, and a subclass that counts can check both that identity and a
867 // hook's refusals against the reasons the hook counted. Added because the refusals made here,
868 // below every hook, were visible only as a difference: on the deliverable cube at
869 // target_distance_rel 1e-2 / front_conv_rel 1e-4, 521142 of the 542123 edge swaps that
870 // passed every before-hook in turn 5 were refused inside this class with no reason recorded.
871 //
872 // The default does nothing and no site reads a count: the mesh behaves identically with or
873 // without an override.
874 enum class OpKind : int { split, collapse, swap_32, swap_44, swap_56, swap_face, COUNT };
875 enum class OpEvent : int {
876 attempt,
877 before_hook, // the application's *_before said no
878 link_condition, // collapse: the classical link condition (m_collapse_check_link_condition)
879 no_return_tet, // collapse: both probes for the return tuple hit a boundary face
880 duplicate_tet, // a new tet repeats an existing one (collapse manifold check, 4-4, 5-6)
881 topology, // collapse: m_collapse_check_topology found a face missing
882 valence, // edge swap: the edge does not have this swap's number of tets around it
883 boundary, // edge swap: the edge's ring of tets is open; face swap: a boundary face
884 exists, // 3-2: the new face already exists; face swap: the new edge already exists
885 no_case_allowed, // 4-4 / 5-6: every candidate was vetoed by *_accept_case
886 no_better_case, // 4-4 / 5-6: no allowed candidate scored below the current cells
887 out_of_slots, // the preallocated vertex or tet slots ran out
888 after_hook, // the application's *_after said no, so the operation was rolled back
889 invariants, // invariants() said no after the after-hook passed; rolled back
890 committed,
891 screened, // a dry run passed every check that precedes the change (see dry_run())
892 COUNT
893 };

1364 void remove_tets_by_ids(const std::vector<size_t>& tids)
1365 {
1366 // A face or edge attribute lives at ONE canonical slot, (lowest incident tet) * 4
1367 // or 6 + local index. Deleting a tet that is a simplex's canonical owner silently
1368 // re-canonicalizes the simplex onto a surviving tet whose cell was never
1369 // maintained -- the operations only ever write the canonical slot -- so the
1370 // surviving cell holds stale garbage. Measured: after filtering, interior faces
1371 // read back m_is_surface_fs = true 0.12 (eps 0.09) from the real surface, and the
1372 // extracted "tracked surface" grows phantom triangles. Migrate every attribute
1373 // whose owner dies to the simplex's new owner BEFORE touching the connectivity.
1374 if (p_face_attrs != nullptr || p_edge_attrs != nullptr) {
1375 std::vector<bool> removed(tet_capacity(), false);
1376 for (const size_t tid : tids) {
1377 removed[tid] = true;
1378 }
1379 for (const size_t tid : tids) {
1380 const auto& tconn = m_tet_connectivity[tid];
1381 if (p_face_attrs != nullptr) {
1382 for (int lf = 0; lf < 4; ++lf) {
1383 std::array<size_t, 3> vids = {
1384 {tconn[m_local_faces[lf][0]],
1385 tconn[m_local_faces[lf][1]],
1386 tconn[m_local_faces[lf][2]]}};
1387 // The face's other incident tet, if any.
1388 auto common = set_intersection(
1389 m_vertex_connectivity[vids[0]].m_conn_tets,
1390 m_vertex_connectivity[vids[1]].m_conn_tets);
1391 common =
1392 set_intersection(common, m_vertex_connectivity[vids[2]].m_conn_tets);
1393 size_t other = std::numeric_limits<size_t>::max();
1394 for (const size_t t : common) {
1395 if (t != tid) {
1396 other = t;
1397 }
1398 }
1399 // Migrate only when this tet was the canonical owner (tid < other)
1400 // and the other side survives; other < tid means the attribute
1401 // already lives on the survivor.
1402 if (other == std::numeric_limits<size_t>::max() || removed[other] ||
1403 other < tid) {
1404 continue;
1405 }
1406 std::array<size_t, 3> sorted = vids;
1407 std::sort(sorted.begin(), sorted.end());
1408 for (int lo = 0; lo < 4; ++lo) {
1409 std::array<size_t, 3> ovids = {
1410 {m_tet_connectivity[other][m_local_faces[lo][0]],
1411 m_tet_connectivity[other][m_local_faces[lo][1]],
1412 m_tet_connectivity[other][m_local_faces[lo][2]]}};
1413 std::sort(ovids.begin(), ovids.end());
1414 if (ovids == sorted) {
1415 p_face_attrs->move(tid * 4 + lf, other * 4 + lo);
1416 break;
1417 }
1418 }
1419 }
1420 }
1421 if (p_edge_attrs != nullptr) {
1422 for (int le = 0; le < 6; ++le) {
1423 const size_t v0 = tconn[m_local_edges[le][0]];
1424 const size_t v1 = tconn[m_local_edges[le][1]];
1425 const auto fan = set_intersection(
1426 m_vertex_connectivity[v0].m_conn_tets,
1427 m_vertex_connectivity[v1].m_conn_tets);
1428 // Only the canonical owner migrates, once, to the lowest survivor.
1429 if (fan.empty() || fan.front() != tid) {
1430 continue;
1431 }
1432 size_t new_owner = std::numeric_limits<size_t>::max();
1433 for (const size_t t : fan) {
1434 if (!removed[t]) {
1435 new_owner = t; // fan is sorted ascending; first kept wins
1436 break;
1437 }
1438 }
1439 if (new_owner == std::numeric_limits<size_t>::max()) {
1440 continue; // the edge dies with its whole fan
1441 }
1442 for (int lo = 0; lo < 6; ++lo) {
1443 const size_t w0 = m_tet_connectivity[new_owner][m_local_edges[lo][0]];
1444 const size_t w1 = m_tet_connectivity[new_owner][m_local_edges[lo][1]];
1445 if ((w0 == v0 && w1 == v1) || (w0 == v1 && w1 == v0)) {
1446 p_edge_attrs->move(tid * 6 + le, new_owner * 6 + lo);
1447 break;
1448 }
1449 }
1450 }
1451 }
1452 }
1453 }
1454
1455 for (size_t tid : tids) {
1456 m_tet_connectivity[tid].m_is_removed = true;
1457 for (int j = 0; j < 4; j++)
1458 vector_erase(m_vertex_connectivity[m_tet_connectivity[tid][j]].m_conn_tets, tid);
1459 }
1460 for (auto& v : m_vertex_connectivity) {
1461 if (v.m_is_removed) continue;
1462 if (v.m_conn_tets.empty()) v.m_is_removed = true;
1463 }
