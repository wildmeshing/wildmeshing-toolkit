139 void init(const VectorXd& min_, const VectorXd& max_)
140 {
141 box_min = min_;
142 box_max = max_;
143
144 // Not a user knob. A TOPOLOGICAL offset is defined by preserving the topology of the
145 // region it wraps, so the shared collapse must always apply the substructure link
146 // condition -- without it a collapse across a thin offset band pinches the two sides
147 // together and the region stops being manifold. The offset's own collapse applied this
148 // unconditionally before it moved onto the shared engine, where it is gated on this
149 // flag; tetwild and simwild leave the flag off, which is why it is set here and not
150 // changed in wmtk.
151 preserve_topology = true;
152
153 // Fills diag_l, l/lr and splitting_l2 / collapsing_l2 -- the same 16/9 and 16/25
154 // factors this used to spell out itself. It also derives eps from epsr, which the
155 // offset never reads: its envelope tolerance is m_envelope_eps, set on the mesh.
156 init_lengths_from_diagonal((max_ - min_).norm());
157
158 if (target_distance > 0) {
159 target_distance_rel = target_distance / diag_l;
160 } else {
161 target_distance = target_distance_rel * diag_l;
162 }
163 }
