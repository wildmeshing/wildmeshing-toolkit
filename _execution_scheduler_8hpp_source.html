53template <class Tuple>
54class OpList
55{
56public:
57 void emplace_back(const Op& op, const Tuple& t) { m_items.emplace_back(intern(op), t); }
58 size_t size() const { return m_items.size(); }
59 bool empty() const { return m_items.empty(); }
60 void clear() { m_items.clear(); }
61
63 template <class F>
64 void consume(F&& f)
65 {
66 while (!m_items.empty()) {
67 const auto& [id, t] = m_items.front();
68 f(m_names[id], t);
69 m_items.pop_front();
70 }
71 m_items.shrink_to_fit();
72 }
73
75 void append(OpList&& other)
76 {
77 if (m_items.empty() && m_names.empty()) {
78 *this = std::move(other);
79 return;
80 }
81 std::vector<uint32_t> remap(other.m_names.size());
82 for (size_t i = 0; i < remap.size(); ++i) remap[i] = intern(other.m_names[i]);
83 while (!other.m_items.empty()) {
84 m_items.emplace_back(remap[other.m_items.front().first], other.m_items.front().second);
85 other.m_items.pop_front();
86 }
87 }
88
89private:
90 uint32_t intern(const Op& op)
91 {
92 // A list sees one to three distinct names; a scan is cheaper than any map.
93 for (size_t i = 0; i < m_names.size(); ++i) {
94 if (m_names[i] == op) return static_cast<uint32_t>(i);
95 }
96 m_names.push_back(op);
97 return static_cast<uint32_t>(m_names.size() - 1);
98 }
99
100 std::vector<Op> m_names;
101 std::deque<std::pair<uint32_t, Tuple>> m_items;
102};

106inline uint64_t edge_queue_key(size_t a, size_t b)
107{
108 if (a > b) std::swap(a, b);
109 if (b >> 32) {
110 log_and_throw_error("edge_queue_key: vertex id {} does not fit in 32 bits", b);
111 }
112 return (uint64_t(a) << 32) | uint64_t(b);
113}

471 bool execute(AppMesh& m, const std::function<void(const Push&)>& seed)
472 {
473 // The queue holds an operation's INDEX rather than its name. edit_operation_maps is a
474 // std::map, so iterating it yields the names in lexicographic order and an index is
475 // exactly a name's rank among them -- comparing indices is therefore identical to
476 // comparing the strings, which is what keeps the queue order, and so the output,
477 // bit-for-bit unchanged. What it buys is that a queue element is now trivially
478 // copyable: a heap sift moves 8 bytes instead of a std::string, and a tie on the
479 // priority is an integer compare instead of a string compare. It also turns the
480 // per-operation dispatch from a string-keyed std::map lookup into an index.
481 using OpId = uint32_t;
482 // priority, op index, #retries, tuple, queue_key. A struct rather than a std::tuple so the
483 // two 32-bit fields share a word: std::tuple lays its members out in reverse and pads each
484 // one to the Tuple's alignment, which costs 8 bytes per queued operation. The ordering
485 // is the one the std::tuple had -- (priority, op, tuple, retry, key), lexicographic -- so
486 // the pop order is unchanged. The key compares last, after the tuple, and two elements
487 // with the same tuple have the same key.
488 // What a parallel task may still take from the storage this round.
489 struct SlotBudget
490 {
491 size_t cells = 0;
492 size_t verts = 0;
493 bool stopped = false; // ended its round for want of slots, queue not drained
494 bool progressed = false; // ran at least one operation this round
495 };
496 struct Elem
497 {
498 double weight = 0;
499 OpId op = 0;
500 uint32_t retry = 0;
501 Tuple tup;
502 uint64_t key = 0; // queue_key, 0 when untracked
503
504 Elem() = default;
505 Elem(double w, OpId o, const Tuple& t, uint32_t r, uint64_t k)
506 : weight(w)
507 , op(o)
508 , retry(r)
509 , tup(t)
510 , key(k)
511 {}
512
513 // A member, not a hidden friend: a local class cannot define friend functions.
514 bool operator<(const Elem& b) const
515 {
516 return std::tie(weight, op, tup, retry, key) <
517 std::tie(b.weight, b.op, b.tup, b.retry, b.key);
518 }
519 };
520 // Each task owns its queue outright -- it is seeded before any thread starts, and the
521 // task both pops from it and pushes its renewed operations back into it -- so those need
522 // no lock. `final_queue` is the one that genuinely crosses threads: tasks push retry
523 // overflow into it while running, and it is drained after the barrier. Both are
524 // std::priority_queue with the same comparator underneath, so pop order is unchanged.
525 using LocalQueue = wmtk::threading::serial_priority_queue<Elem>;
526 using SharedQueue = wmtk::threading::concurrent_priority_queue<Elem>;
527
528 std::vector<const Op*> op_name;
529 std::vector<std::function<std::optional<std::vector<Tuple>>(AppMesh&, const Tuple&)>*>
530 op_fn;
531 std::map<Op, OpId> op_id;
532 op_name.reserve(edit_operation_maps.size());
533 op_fn.reserve(edit_operation_maps.size());
534 for (auto& kv : edit_operation_maps) {
535 op_id.emplace(kv.first, OpId(op_name.size()));
536 op_name.push_back(&kv.first);
537 op_fn.push_back(&kv.second);
538 }
539 // An operation with no entry here was previously default-constructed into the map by
540 // operator[] and then called, which throws bad_function_call -- so it cannot occur in
541 // any working configuration. Say so plainly rather than ordering it arbitrarily.
542 const auto id_of = [&op_id](const Op& name) {
543 const auto it = op_id.find(name);
544 if (it == op_id.end()) {
545 log_and_throw_error("No operation registered under the name '{}'.", name);
546 }
547 return it->second;
548 };

550 std::atomic<bool> stop(false);
551 cnt_success = 0;
552 cnt_fail = 0;

554 // queue_key bookkeeping (see must_wait). `track` when an element is queued, `done` when
555 // it is tried or dropped; both no-ops for the untracked key 0.
556 m_queued.clear();
557 std::atomic<size_t> untracked_done(0);
558 const auto key_of = [&](const Op& op, const Tuple& e) -> uint64_t {
559 return queue_key ? queue_key(m, op, e) : 0;
560 };
561 const auto track = [&](const uint64_t key) {
562 if (key != 0) m_queued.add(key);
563 };
564 const auto done = [&](const uint64_t key) {
565 if (key != 0 && !m_queued.remove(key)) {
566 untracked_done.fetch_add(1, std::memory_order_relaxed);
567 }
568 };

570 // Whether anything actually watches the success count *while the pass runs*. When no
571 // stopping criterion is configured -- the case for every tetwild/triwild/simwild pass --
572 // nobody does, and the counters can be accumulated per task and folded in at the end
573 // instead of hammering one shared cache line from every thread on every operation.
574 const bool track_live_success =
575 stopping_criterion_checking_frequency != std::numeric_limits<size_t>::max();
576 std::atomic<size_t> live_success(0);

581 // Contention accounting. Everything here is either a per-task local folded in once or a
582 // write to the task's own slot, so it adds nothing to the inner loop. It answers the
583 // two questions the pass could not previously be asked: how often ring acquisition
584 // loses a race, and how much of a "parallel" pass is really the serial drain.
585 m_stats = PassStats{};
586 std::atomic<size_t> lock_failures(0);
587 std::atomic<size_t> overflowed(0);
588 std::vector<double> task_seconds(queues.size(), 0.);

590 // Per-task tallies folded into the shared counters once, on the way out. The guard is
591 // RAII rather than a line at the bottom because run_single_queue has early returns.
592 //
593 // DECLARED HERE, NOT INSIDE THE LAMBDA, to work around an Apple clang 17 codegen bug.
594 // A local class declared inside a GENERIC lambda (one with an `auto` parameter, so its
595 // operator() is a template) gets its destructor emitted as an undefined reference and
596 // never defined, at every optimization level including -O0. The link then fails with
597 // Undefined symbols: ... ::'lambda'(auto&, int)::operator()<...>
598 // ::CountFlusher::~CountFlusher()
599 // referenced from every translation unit that instantiates a pass -- 26 references
600 // across 6 archives, defined nowhere. Homebrew clang 22 compiles the same code
601 // correctly, which is why CI does not see this.
602 //
603 // Hoisting the class one scope out is enough: it is still local to operator(), which is
604 // itself a member of a class template, and that instantiates fine. Nothing else changes
605 // -- `counts` is still constructed per task inside the lambda, so the RAII flush and its
606 // ordering are identical.
607 struct CountFlusher
608 {
609 std::atomic_int& success_total;
610 std::atomic_int& fail_total;
611 std::atomic<size_t>& lock_failure_total;
612 std::atomic<size_t>& overflow_total;
613 int success = 0;
614 int fail = 0;
615 size_t lock_failure = 0;
616 size_t overflow = 0;
617 ~CountFlusher()
618 {
619 success_total.fetch_add(success, std::memory_order_relaxed);
620 fail_total.fetch_add(fail, std::memory_order_relaxed);
621 lock_failure_total.fetch_add(lock_failure, std::memory_order_relaxed);
622 overflow_total.fetch_add(overflow, std::memory_order_relaxed);
623 }
624 };

626 // `serial`: no other thread is running operations on the mesh, and Q is popped by this
627 // task alone and in priority order -- the serial policy, and the serial drain after a
628 // parallel pass. The storage may then grow between operations (see
629 // TetMesh::reserve_free_slots), never inside a parallel task; and a must_wait that says
630 // true is a defect there.
631 //
632 // `budget`: the slots a parallel task may still take this round (see the round loop
633 // below); null when the task may grow the storage instead.
634 auto run_single_queue = [&](auto& Q, int task_id, const bool serial, SlotBudget* budget) {
635 CountFlusher counts{cnt_success, cnt_fail, lock_failures, overflowed};
636
637 Elem ele_in_queue;
638 // Operations that lost a race for their ring wait here rather than going straight
639 // back into the live queue. Pushing them back immediately is a spin: the element was
640 // just popped as the queue's maximum, `retry` is the last tie-break key in Elem, so
641 // the requeued copy compares strictly greater than what was popped and nothing else
642 // was added -- it comes right back off the top and is retried against a conflict that
643 // has had no time to clear. Deferring lets every other operation this task owns run
644 // first, which is both useful work and the delay the conflict needs.
645 std::vector<Elem> second_chance;
646 // Operations popped since the deferred list was last given back to the queue.
647 // Waiting for the queue to drain completely can mean thousands of operations, by
648 // which time the mesh around a deferred operation has moved and its work has to be
649 // rediscovered. A bounded window gives the conflict time to clear without letting
650 // the operation go stale. 0 = only when the queue empties.
651 size_t since_refill = 0;
652 const auto refill = [&] {
653 for (auto& e : second_chance) {
654 Q.emplace(std::move(e));
655 }
656 second_chance.clear();
657 since_refill = 0;
658 };
659 for (;;) {
660 if (!second_chance.empty() && deferral_window > 0 &&
661 since_refill >= deferral_window) {
662 refill();
663 }
664 if (!Q.try_pop(ele_in_queue)) {
665 if (second_chance.empty()) {
666 break;
667 }
668 // Queue exhausted: the deferred operations have now had everything else
669 // run ahead of them, so give them another go. `retry` still increments on
670 // each attempt and still overflows to final_queue at max_retry_limit, so
671 // this terminates after at most that many rounds.
672 refill();
673 std::this_thread::yield();
674 continue;
675 }
676 ++since_refill;
677 auto& [weight, op, retry, tup, key] = ele_in_queue;
678 if (!tup.is_valid(m)) {
679 done(key);
680 continue;
681 }
682
683 std::vector<Elem> renewed_elements;
684 {
685 auto locked_vid = lock_vertices(
686 m,
687 tup,
688 task_id); // Note that returning `Tuples` would be invalid.
689 if (!locked_vid) {
690 counts.lock_failure++;
691 retry++;
692 if (retry < max_retry_limit) {
693 second_chance.push_back(ele_in_queue);
694 } else {
695 retry = 0;
696 counts.overflow++;
697 final_queue.emplace(ele_in_queue);
698 }
699 continue;
700 }
701 if (tup.is_valid(m)) {
702 const Op& op_str = *op_name[op];
703 if (!is_weight_up_to_date(
704 m,
705 std::tuple<double, Op, Tuple>(weight, op_str, tup))) {
706 done(key);
707 operation_cleanup(m);
708 continue;
709 } // this can encode, in qslim, recompute(energy) == weight.
710 if (must_wait && must_wait(m, op_str, tup)) {
711 if (serial) {
712 m_wait_defects.fetch_add(1, std::memory_order_relaxed);
713 done(key);
714 on_fail(m, op_str, tup);
715 counts.fail++;
716 operation_cleanup(m);
717 continue;
718 }
719 // Set aside exactly like a lost lock race, above.
720 operation_cleanup(m);
721 m_waits.fetch_add(1, std::memory_order_relaxed);
722 retry++;
723 if (retry < max_retry_limit) {
724 second_chance.push_back(ele_in_queue);
725 } else {
726 retry = 0;
727 counts.overflow++;
728 final_queue.emplace(ele_in_queue);
729 }
730 continue;
731 }
732 if (serial) {
733 m.reserve_free_slots(m.cell_slot_bound(tup), 1);
734 } else if (budget != nullptr) {
735 // The ring is locked, so the stars the bound reads are stable.
736 if (m.cell_slot_bound(tup) > budget->cells || budget->verts < 1) {
737 // Out of this round's slots: hand the operation back untouched,
738 // with everything deferred, and end the task's round.
739 operation_cleanup(m);
740 Q.emplace(ele_in_queue);
741 refill();
742 budget->stopped = true;
743 return;
744 }
745 }
746 const auto usage_before = AppMesh::slot_usage_of_this_thread();
747 auto newtup = (*op_fn[op])(m, tup);
748 done(key);
749 if (budget != nullptr) {
750 const auto usage = AppMesh::slot_usage_of_this_thread();
751 budget->cells -= usage.cells - usage_before.cells;
752 budget->verts -= usage.verts - usage_before.verts;
753 budget->progressed = true;
754 }
755 std::vector<std::pair<Op, Tuple>> renewed_tuples;
756 if (newtup) {
757 renewed_tuples = renew_neighbor_tuples(m, op_str, newtup.value());
758 counts.success++;
759 if (track_live_success) {
760 live_success.fetch_add(1, std::memory_order_relaxed);
761 }
762 } else {
763 on_fail(m, op_str, tup);
764 counts.fail++;
765 }
766 // Tracked here, with the ring still locked, although the elements are
767 // queued only after it is released: in between they count as queued.
768 for (const auto& [o, e] : renewed_tuples) {
769 auto val = priority(m, o, e);
770 if (should_renew(val)) {
771 const uint64_t k = key_of(o, e);
772 track(k);
773 renewed_elements.emplace_back(val, id_of(o), e, 0, k);
774 }
775 }
776 } else {
777 done(key);
778 }
779 operation_cleanup(m); // Maybe use RAII
780 }
781 for (auto& e : renewed_elements) {
782 Q.emplace(e);
783 }
784
785 if (stop.load(std::memory_order_acquire)) {
786 return;
787 }
788 if (track_live_success && live_success.load(std::memory_order_relaxed) >
789 stopping_criterion_checking_frequency) {
790 if (stopping_criterion(m)) {
791 stop.store(true);
792 return;
793 }
794 }
795 }
796 };
797
798 if (policy == ExecutionPolicy::kSeq) {
799 seed([&](const Op& op, const Tuple& e) {
800 if (!e.is_valid(m)) {
801 return;
802 }
803 const uint64_t k = key_of(op, e);
804 track(k);
805 final_queue.emplace(priority(m, op, e), id_of(op), e, 0, k);
806 });
807 run_single_queue(final_queue, 0, /*serial=*/true, nullptr);
808 } else {
809 seed([&](const Op& op, const Tuple& e) {
810 if (!e.is_valid(m)) {
811 return;
812 }
813 const uint64_t k = key_of(op, e);
814 track(k);
815 queues[get_partition_id(m, e)].emplace(priority(m, op, e), id_of(op), e, 0, k);
816 });
817 // Comment out parallel: work on serial first.
818 using clock = std::chrono::steady_clock;
819 const auto t_parallel = clock::now();
820 // Rounds. The storage cannot grow while the tasks run, so each round starts at a
821 // serial point that grows it by a headroom split into per-task budgets; a task runs
822 // an operation only if its budget covers the operation's slot bound, and is charged
823 // what the operation actually took. A task that cannot afford its next operation
824 // hands it back and ends its round. No operation is ever refused for want of slots,
825 // and the storage only ever needs to be a quarter larger than the mesh.
826 //
827 // A task that ends a round without having run anything had a single operation
828 // larger than its share; its share doubles until it fits, so every round but the
829 // last makes progress and the loop terminates.
830 const size_t n_tasks = queues.size();
831 std::vector<SlotBudget> budgets(n_tasks);
832 std::vector<unsigned> boost(n_tasks, 0);
833 wmtk::threading::task_group tg;
834 for (size_t round = 0;; ++round) {
835 // A quarter of the mesh per round, or whatever the storage already has free if
836 // that is more -- a mesh that still preallocates never runs more rounds than
837 // its headroom forces.
838 const size_t cell_room =
839 std::max(m.cell_capacity() / 4, m.cell_storage_capacity() - m.cell_capacity());
840 const size_t vert_room =
841 std::max(m.vert_capacity() / 4, m.vert_storage_capacity() - m.vert_capacity());
842 const size_t cell_share = std::max<size_t>(4096, cell_room / n_tasks);
843 const size_t vert_share = std::max<size_t>(1024, vert_room / n_tasks);
844 size_t cells = 0, verts = 0;
845 for (size_t t = 0; t < n_tasks; ++t) {
846 budgets[t] = SlotBudget{cell_share << boost[t], vert_share << boost[t]};
847 cells += budgets[t].cells;
848 verts += budgets[t].verts;
849 }
850 m.reserve_free_slots(cells, verts);
851 for (int task_id = 0; task_id < n_tasks; task_id++) {
852 tg.run([&run_single_queue, &queues, &task_seconds, &budgets, task_id] {
853 const auto t0 = clock::now();
854 run_single_queue(
855 queues[task_id],
856 task_id,
857 /*serial=*/false,
858 &budgets[task_id]);
859 // Each task writes only its own slot.
860 task_seconds[task_id] +=
861 std::chrono::duration<double>(clock::now() - t0).count();
862 });
863 }
864 tg.wait();
865 bool again = false;
866 for (size_t t = 0; t < n_tasks; ++t) {
867 if (!budgets[t].stopped) continue;
868 again = true;
869 if (!budgets[t].progressed) boost[t] = std::min(boost[t] + 1, 24u);
870 }
871 if (!again) break;
872 m_stats.rounds = round + 2;
873 }
874 m_stats.parallel_seconds =
875 std::chrono::duration<double>(clock::now() - t_parallel).count();
876 m_stats.final_queue_size = final_queue.size();
877
878 logger().debug("Parallel Complete, remains element {}", final_queue.size());
879
880 const auto t_tail = clock::now();
881 run_single_queue(final_queue, 0, /*serial=*/true, nullptr);
882 m_stats.serial_tail_seconds =
883 std::chrono::duration<double>(clock::now() - t_tail).count();
884 }
885
886 m_stats.lock_failures = lock_failures.load(std::memory_order_relaxed);
887 m_stats.overflowed = overflowed.load(std::memory_order_relaxed);
888 if (!task_seconds.empty()) {
889 const auto mm = std::minmax_element(task_seconds.begin(), task_seconds.end());
890 m_stats.idlest_task_seconds = *mm.first;
891 m_stats.busiest_task_seconds = *mm.second;
892 }
893
894 logger().info(
895 "executed: {} | success / fail: {} / {}",
896 (int)cnt_success + (int)cnt_fail,
897 (int)cnt_success,
898 (int)cnt_fail);
899 if (m_stats.rounds > 1) {
900 logger().info(
901 " parallel region ran in {} rounds (storage grown between them)",
902 m_stats.rounds);
903 }
904 log_contention();
905 // Every tracked element queued in this call was tried or dropped, unless the stopping
906 // criterion ended the call early. Anything else is a bookkeeping defect, and is_queued()
907 // may have answered wrongly during the call.
908 if (queue_key && !stop.load()) {
909 const size_t left = m_queued.size();
910 const size_t untracked = untracked_done.load();
911 if (left > 0 || untracked > 0) {
912 logger().warn(
913 "queue_key bookkeeping: {} keys still counted as queued after the pass, {} "
914 "finished elements were never counted",
915 left,
916 untracked);
917 }
918 }
919 m_queued.clear();
920 return true;
921 }
