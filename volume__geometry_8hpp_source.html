76template <typename Accept>
77std::optional<Location>
78locate_in(const Tets& T, const Positions& X, const Eigen::Vector3d& p, const Accept& accept)
79{
80 if (!p.allFinite()) return std::nullopt;
81 std::vector<size_t> within_rounding;
82 for (size_t k = 0; k < T.size(); ++k) {
83 if (!accept(k)) continue;
84 const auto& t = T[k];
85 // only a tet whose box (grown by a rounding) holds p can hold it or be a rounding off
86 Eigen::Vector3d lo = X[t[0]], hi = lo;
87 for (int c = 1; c < 4; ++c) {
88 lo = lo.cwiseMin(X[t[c]]);
89 hi = hi.cwiseMax(X[t[c]]);
90 }
91 const double size = (hi - lo).norm(), eps = 1e-9 * size;
92 if (((p.array() < lo.array() - eps) || (p.array() > hi.array() + eps)).any()) continue;
93 const Outside o = outside_of(p, X[t[0]], X[t[1]], X[t[2]], X[t[3]]);
94 if (o.faces == 0) {
95 Eigen::Vector4d b = barycentric(p, X[t[0]], X[t[1]], X[t[2]], X[t[3]]);
96 if (!(b.minCoeff() >= -1e-9 && b.maxCoeff() <= 1 + 1e-9)) {
97 b = exact_barycentric(p, X[t[0]], X[t[1]], X[t[2]], X[t[3]]);
98 }
99 b = b.cwiseMax(0.);
100 return Location{k, b / b.sum(), 0.};
101 }
102 if (o.distance <= eps) within_rounding.push_back(k);
103 }
104 std::optional<Location> out;
105 if (!within_rounding.empty()) {
106 for (const size_t k : within_rounding) {
107 const auto& t = T[k];
108 const Eigen::Vector4d b = exact_barycentric(p, X[t[0]], X[t[1]], X[t[2]], X[t[3]]);
109 if (!out || b.minCoeff() > out->min_coordinate) out = Location{k, b, b.minCoeff()};
110 }
111 }
112 if (out) {
113 out->b = out->b.cwiseMax(0.);
114 out->b /= out->b.sum();
115 }
116 return out;
117}
