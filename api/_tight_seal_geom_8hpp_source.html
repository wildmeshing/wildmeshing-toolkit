54template <typename V>
55std::optional<V> level_crossing(
56 const std::function<double(const V&)>& f,
57 const V& a,
58 const V& b,
59 const double level,
60 const double step_floor,
61 const bool from_below)
62{
63 const auto before = [&](const double v) { return from_below ? v < level : v > level; };
64 double v = f(a);
65 if (!before(v)) {
66 return std::nullopt;
67 }
68 const double L = (b - a).norm();
69 const V dir = (b - a) / L;
70 double t_prev = 0.;
71 double t = 0.;
72 while (before(v)) {
73 t_prev = t;
74 t += std::abs(v - level) + step_floor;
75 if (t >= L) {
76 t = L;
77 v = f(b);
78 if (before(v)) {
79 return std::nullopt;
80 }
81 break;
82 }
83 v = f(a + t * dir);
84 }
85 // f(t_prev) before the level, f(t) at it or past it
86 double lo = t_prev;
87 double hi = t;
88 for (int i = 0; i < 50; ++i) {
89 const double m = (lo + hi) / 2;
90 if (before(f(a + m * dir))) {
91 lo = m;
92 } else {
93 hi = m;
94 }
95 }
96 return V(a + ((lo + hi) / 2) * dir);
97}
