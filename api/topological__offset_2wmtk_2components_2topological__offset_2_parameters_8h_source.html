203 // Sampling density of THE measure -- both the criterion's and the energy's, in 2D and 3D --
204 // and of the residual diagnostics that share the lattice. Order k puts these points on a
205 // face (3D):
206 // 0 -> 3, the CORNERS alone, so the measure is exactly the three vertices' placement error;
207 // k >= 1 -> the vertices of the triangle subdivided k-1 times by 4-way midpoint refinement
208 // plus the centroid of each of its 4^(k-1) sub-triangles, i.e. 4, 10, 31, 109, ...
209 // and on a front chord (2D), the same rule one dimension down:
210 // 0 -> 2, the ENDS alone;
211 // k >= 1 -> the vertices of the chord cut into 2^(k-1) pieces plus the midpoint of each
212 // piece, i.e. 3, 5, 9, 17, ...
213 // The corners are IN the stencil, unlike the strictly interior lattice this replaces, because
214 // the quantity measured is a distance to the level set rather than an interpolation error and
215 // so is not identically zero there. Raising it costs a Phi value per sample in
216 // energy_criterion() and the frames, and in 2D also in every front solve (its stencil energy,
217 // until the 2D smoother is ported); neither energy has a face or chord term.
218 // See TopoOffsetTetMesh::for_each_face_sample, TopoOffsetTriMesh::for_each_edge_sample.
219 // That is EXPERIMENTAL_stencil_type "subdivision", the default; the other types (3D) have
220 // their own lattices, below.

731 // before the first smoothing pass of the next turn; the other path has no such point.
732 if (settle_mark == "vertex_split" && !front_pocket_split) {
733 log_and_throw_error(
734 "EXPERIMENTAL_settle_mark \"vertex_split\" needs EXPERIMENTAL_front_pocket_split "
735 "true: its splits are made right before the first smoothing pass of the next turn");
736 }
737 sizing_collapse_min = json_params["sizing_collapse_min"];
738 use_rest_pose = json_params["use_rest_pose"];
739 max_rounds = json_params["max_rounds"];
740 w_amips = json_params["w_amips"];
741 smoothing_mode = json_params["smoothing_mode"];
742 project_line_search_steps = json_params["project_line_search_steps"];
743 project_line_search_nested_steps = json_params["project_line_search_nested_steps"];
744 w_envelope = 1. - w_amips;
745 perform_sanity_checks = json_params["perform_sanity_checks"];
746 }
747
748 void init(const VectorXd& min_, const VectorXd& max_)
749 {
750 box_min = min_;
751 box_max = max_;
752
753 // Not a user knob: a topological offset preserves the topology of the region it wraps, so
754 // the shared collapse must always apply the substructure link condition, or a collapse
755 // across a thin band pinches the two sides together and the region stops being manifold.
756 // Set here because tetwild and simwild leave the flag off.
757 preserve_topology = true;
758
759 // Fills diag_l, l/lr and splitting_l2 / collapsing_l2. It also derives eps from epsr,
760 // which the offset never reads: its envelope tolerance is m_envelope_eps, set on the mesh.
761 init_lengths_from_diagonal((max_ - min_).norm());
762
763 // Each entry's own distance, absolute or relative; the run's is the smallest of them. A
764 // Parameters built by hand (the unit tests) has no entry and sets the run's directly.
765 for (size_t k = 0; k < offsets.size(); ++k) {
766 Offset& o = offsets[k];
767 if (o.target_distance > 0) {
768 o.target_distance_rel = o.target_distance / diag_l;
769 } else {
770 o.target_distance = o.target_distance_rel * diag_l;
771 }
772 if (!(o.target_distance > 0)) {
773 log_and_throw_error(
774 "offsets[{}]: target_distance {} (target_distance_rel {}) is not positive",
775 k,
776 o.target_distance,
777 o.target_distance_rel);
778 }
779 if (k == 0 || o.target_distance < target_distance) target_distance = o.target_distance;
780 }
781 if (target_distance > 0) {
782 target_distance_rel = target_distance / diag_l;
783 } else {
784 target_distance = target_distance_rel * diag_l;
785 }
786
787 // An ordinary relative length: it bounds how far a region boundary may drift in space, so
788 // the bounding box is the right reference.
789 if (envelope_size > 0) {
790 envelope_size_rel = envelope_size / diag_l;
791 } else {
792 envelope_size = envelope_size_rel * diag_l;
793 }
794
795 // The convergence epsilon, absolute or relative; relative to target_distance (resolved
796 // above), the offset it measures. A growth stage changes target_distance later, never
797 // this.
798 // Several entries of offsets: each piece's bar is front_conv_rel of its own distance. An
799 // absolute front_conv is one length for every piece, which entries of different distances
800 // would each need read in their own units; refused for those until it is (a global
801 // tolerance key, Uday 2026-11-09: "once things are working").
802 if (front_conv > 0) {
803 for (const Offset& o : offsets) {
804 if (o.target_distance != target_distance) {
805 log_and_throw_error(
806 "front_conv {} (absolute) with offsets of different target distances ({} "
807 "and {}): give front_conv_rel, each offset's bar that fraction of its own "
808 "distance",
809 front_conv,
810 target_distance,
811 o.target_distance);
812 }
813 }
814 front_conv_rel = front_conv / target_distance;
815 } else {
816 front_conv = front_conv_rel * target_distance;
817 }
818
819 // An epsilon coarser than the offset it measures decides nothing: every front vertex is
820 // "placed" and every face "resolved" from the first turn, whatever the offset looks like.
821 // Checked on the resolved ABSOLUTE values, so it catches the mistake whichever of the two
822 // forms the config used to state it.
823 if (front_conv > target_distance) {
824 log_and_throw_error(
825 "front_conv {} must be <= target_distance {}: the convergence epsilon cannot be "
826 "coarser than the offset distance it measures, or every front face reads as "
827 "resolved from the first turn",
828 front_conv,
829 target_distance);
830 }
831
832 // The held surfaces -- the input complex, the region boundaries and the box -- may move
833 // within envelope_size, and the offset is measured to front_conv: an envelope wider than
834 // the bar lets the input the band is grown from drift farther than the offset's own
835 // accuracy, so the bar cannot be met where the offset has detail at that scale. Refused
836 // rather than narrowed (Uday 2026-11-12): the config states the envelope. front_conv is
837 // the smallest bar (several entries: front_conv_rel of the smallest distance). MEASURED
838 // before (triwild20k_172573 at 0.9 delta_1, delta 0.414, front_conv 0.041, envelope_size
839 // 1.76 from the default envelope_size_rel 1e-3): the input curves, held to 4.3 delta,
840 // became chords across a hairpin's bend inside a 0.09-wide gap of the offset, and the run
841 // stuck there for 80 turns; at envelope_size 0.035 it converged in 18. The two are
842 // compared up to the rounding of the products that resolve them (a config giving them
843 // equal must not be refused by an ulp).
844 if (envelope_size > front_conv * (1. + 1e-12)) {
845 log_and_throw_error(
846 "envelope_size {} ({} x the bounding box diagonal {}) is wider than the bar "
847 "front_conv {} ({} x target_distance {}): the input complex and the other held "
848 "surfaces could move farther than the offset is measured to. Set envelope_size_rel "
849 "<= {} (or envelope_size <= {}), or a coarser front_conv_rel",
850 envelope_size,
851 envelope_size_rel,
852 diag_l,
853 front_conv,
854 front_conv_rel,
855 target_distance,
856 front_conv / diag_l,
857 front_conv);
858 }
859
860 // The offset surface's envelope is envelope_size's (no key of its own since 2026-10-20);
861 // the two fields only for the 2D code, which still reads them.
862 offset_envelope = envelope_size;
863 offset_envelope_rel = envelope_size_rel;
864
865 // l_min is relative to the offset distance rather than the bounding box: it is the offset
866 // that has to be resolved. See the declaration.
867 if (min_edge_length_rel < 0) {
868 // The envelope eps expressed as a multiple of target_distance, which is what this
869 // wants: envelope_size, the offset surface's envelope too since offset_envelope went
870 // (2026-10-20), resolved to an absolute length above.
871 min_edge_length_rel = std::max(envelope_size / std::max(target_distance, 1e-16), 1e-12);
872 }
873 if (min_edge_length < 0) {
874 min_edge_length = min_edge_length_rel * target_distance;
875 } else {
876 min_edge_length_rel = min_edge_length / std::max(target_distance, 1e-16);
877 }
878 }
