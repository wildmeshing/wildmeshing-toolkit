195 // Sampling density of THE measure -- both the criterion's and the energy's, in 2D and 3D --
196 // and of the residual diagnostics that share the lattice. Order k puts these points on a
197 // face (3D):
198 // 0 -> 3, the CORNERS alone, so the measure is exactly the three vertices' placement error;
199 // k >= 1 -> the vertices of the triangle subdivided k-1 times by 4-way midpoint refinement
200 // plus the centroid of each of its 4^(k-1) sub-triangles, i.e. 4, 10, 31, 109, ...
201 // and on a front chord (2D), the same rule one dimension down:
202 // 0 -> 2, the ENDS alone;
203 // k >= 1 -> the vertices of the chord cut into 2^(k-1) pieces plus the midpoint of each
204 // piece, i.e. 3, 5, 9, 17, ...
205 // The corners are IN the stencil, unlike the strictly interior lattice this replaces, because
206 // the quantity measured is a distance to the level set rather than an interpolation error and
207 // so is not identically zero there. Raising it costs a Phi value per sample in
208 // energy_criterion() and the frames, and in 2D also in the operations' energy rules
209 // (TopoOffsetTriMesh::tri_energy()) and every front solve; the 3D energy has no face term.
210 // See TopoOffsetTetMesh::for_each_face_sample, TopoOffsetTriMesh::for_each_edge_sample.
211 // That is EXPERIMENTAL_stencil_type "subdivision", the default; the other types (3D) have
212 // their own lattices, below.

705 "of each operation group",
706 label_cleanup);
707 }
708 // "vertex_split" splits its edges in loop_group()'s EXPERIMENTAL_front_pocket_split path,
709 // before the first smoothing pass of the next turn; the other path has no such point.
710 if (settle_mark == "vertex_split" && !front_pocket_split) {
711 log_and_throw_error(
712 "EXPERIMENTAL_settle_mark \"vertex_split\" needs EXPERIMENTAL_front_pocket_split "
713 "true: its splits are made right before the first smoothing pass of the next turn");
714 }
715 sizing_collapse_min = json_params["sizing_collapse_min"];
716 use_rest_pose = json_params["use_rest_pose"];
717 max_rounds = json_params["max_rounds"];
718 w_amips = json_params["w_amips"];
719 smoothing_mode = json_params["smoothing_mode"];
720 project_line_search_steps = json_params["project_line_search_steps"];
721 project_line_search_nested_steps = json_params["project_line_search_nested_steps"];
722 w_envelope = 1. - w_amips;
723 perform_sanity_checks = json_params["perform_sanity_checks"];

726 void init(const VectorXd& min_, const VectorXd& max_)
727 {
728 box_min = min_;
729 box_max = max_;
730
731 // Not a user knob: a topological offset preserves the topology of the region it wraps, so
732 // the shared collapse must always apply the substructure link condition, or a collapse
733 // across a thin band pinches the two sides together and the region stops being manifold.
734 // Set here because tetwild and simwild leave the flag off.
735 preserve_topology = true;
736
737 // Fills diag_l, l/lr and splitting_l2 / collapsing_l2. It also derives eps from epsr,
738 // which the offset never reads: its envelope tolerance is m_envelope_eps, set on the mesh.
739 init_lengths_from_diagonal((max_ - min_).norm());
740
741 // Each entry's own distance, absolute or relative; the run's is the smallest of them. A
742 // Parameters built by hand (the unit tests) has no entry and sets the run's directly.
743 for (size_t k = 0; k < offsets.size(); ++k) {
744 Offset& o = offsets[k];
745 if (o.target_distance > 0) {
746 o.target_distance_rel = o.target_distance / diag_l;
747 } else {
748 o.target_distance = o.target_distance_rel * diag_l;
749 }
750 if (!(o.target_distance > 0)) {
751 log_and_throw_error(
752 "offsets[{}]: target_distance {} (target_distance_rel {}) is not positive",
753 k,
754 o.target_distance,
755 o.target_distance_rel);
756 }
757 if (k == 0 || o.target_distance < target_distance) target_distance = o.target_distance;
758 }
759 if (target_distance > 0) {
760 target_distance_rel = target_distance / diag_l;
761 } else {
762 target_distance = target_distance_rel * diag_l;
763 }
764
765 // An ordinary relative length: it bounds how far a region boundary may drift in space, so
766 // the bounding box is the right reference.
767 if (envelope_size > 0) {
768 envelope_size_rel = envelope_size / diag_l;
769 } else {
770 envelope_size = envelope_size_rel * diag_l;
771 }
772
773 // The convergence epsilon, absolute or relative; relative to target_distance (resolved
774 // above), the offset it measures. A growth stage changes target_distance later, never
775 // this.
776 // Several entries of offsets: each piece's bar is front_conv_rel of its own distance. An
777 // absolute front_conv is one length for every piece, which entries of different distances
778 // would each need read in their own units; refused for those until it is (a global
779 // tolerance key, Uday 2026-11-09: "once things are working").
780 if (front_conv > 0) {
781 for (const Offset& o : offsets) {
782 if (o.target_distance != target_distance) {
783 log_and_throw_error(
784 "front_conv {} (absolute) with offsets of different target distances ({} "
785 "and {}): give front_conv_rel, each offset's bar that fraction of its own "
786 "distance",
787 front_conv,
788 target_distance,
789 o.target_distance);
790 }
791 }
792 front_conv_rel = front_conv / target_distance;
793 } else {
794 front_conv = front_conv_rel * target_distance;
795 }
796
797 // An epsilon coarser than the offset it measures decides nothing: every front vertex is
798 // "placed" and every face "resolved" from the first turn, whatever the offset looks like.
799 // Checked on the resolved ABSOLUTE values, so it catches the mistake whichever of the two
800 // forms the config used to state it.
801 if (front_conv > target_distance) {
802 log_and_throw_error(
803 "front_conv {} must be <= target_distance {}: the convergence epsilon cannot be "
804 "coarser than the offset distance it measures, or every front face reads as "
805 "resolved from the first turn",
806 front_conv,
807 target_distance);
808 }
809
810 // The offset surface's envelope is envelope_size's (no key of its own since 2026-10-20);
811 // the two fields only for the 2D code, which still reads them.
812 offset_envelope = envelope_size;
813 offset_envelope_rel = envelope_size_rel;
814
815 // l_min is relative to the offset distance rather than the bounding box: it is the offset
816 // that has to be resolved. See the declaration.
817 if (min_edge_length_rel < 0) {
818 // The envelope eps expressed as a multiple of target_distance, which is what this
819 // wants: envelope_size, the offset surface's envelope too since offset_envelope went
820 // (2026-10-20), resolved to an absolute length above.
821 min_edge_length_rel = std::max(envelope_size / std::max(target_distance, 1e-16), 1e-12);
822 }
823 if (min_edge_length < 0) {
824 min_edge_length = min_edge_length_rel * target_distance;
825 } else {
826 min_edge_length_rel = min_edge_length / std::max(target_distance, 1e-16);
827 }
828 }
