128 // Sampling density of THE measure -- both the criterion's and the energy's, in 2D and 3D --
129 // and of the residual diagnostics that share the lattice. Order k puts these points on a
130 // face (3D):
131 // 0 -> 3, the CORNERS alone, so the measure is exactly the three vertices' placement error;
132 // k >= 1 -> the vertices of the triangle subdivided k-1 times by 4-way midpoint refinement
133 // plus the centroid of each of its 4^(k-1) sub-triangles, i.e. 4, 10, 31, 109, ...
134 // and on a front chord (2D), the same rule one dimension down:
135 // 0 -> 2, the ENDS alone;
136 // k >= 1 -> the vertices of the chord cut into 2^(k-1) pieces plus the midpoint of each
137 // piece, i.e. 3, 5, 9, 17, ...
138 // The corners are IN the stencil, unlike the strictly interior lattice this replaces, because
139 // the quantity measured is a distance to the level set rather than an interpolation error and

142 // (TopoOffsetTriMesh::tri_energy()) and every front solve; the 3D energy has no face term.
143 // See TopoOffsetTetMesh::for_each_face_sample, TopoOffsetTriMesh::for_each_edge_sample.

658 void init(const VectorXd& min_, const VectorXd& max_)
659 {
660 box_min = min_;
661 box_max = max_;
662
663 // Not a user knob: a topological offset preserves the topology of the region it wraps, so
664 // the shared collapse must always apply the substructure link condition, or a collapse
665 // across a thin band pinches the two sides together and the region stops being manifold.
666 // Set here because tetwild and simwild leave the flag off.
667 preserve_topology = true;
668
669 // Fills diag_l, l/lr and splitting_l2 / collapsing_l2. It also derives eps from epsr,
670 // which the offset never reads: its envelope tolerance is m_envelope_eps, set on the mesh.
671 init_lengths_from_diagonal((max_ - min_).norm());
672
673 if (target_distance > 0) {
674 target_distance_rel = target_distance / diag_l;
675 } else {
676 target_distance = target_distance_rel * diag_l;
677 }
678
679 // An ordinary relative length: it bounds how far a region boundary may drift in space, so
680 // the bounding box is the right reference.
681 if (envelope_size > 0) {
682 envelope_size_rel = envelope_size / diag_l;
683 } else {
684 envelope_size = envelope_size_rel * diag_l;
685 }
686
687 // The convergence epsilon, absolute or relative; relative to target_distance (resolved
688 // above), the offset it measures. A growth stage changes target_distance later, never
689 // this.
690 if (front_conv > 0) {
691 front_conv_rel = front_conv / target_distance;
692 } else {
693 front_conv = front_conv_rel * target_distance;
694 }
695
696 // An epsilon coarser than the offset it measures decides nothing: every front vertex is
697 // "placed" and every face "resolved" from the first turn, whatever the offset looks like.
698 // Checked on the resolved ABSOLUTE values, so it catches the mistake whichever of the two
699 // forms the config used to state it.
700 if (front_conv > target_distance) {
701 log_and_throw_error(
702 "front_conv {} must be <= target_distance {}: the convergence epsilon cannot be "
703 "coarser than the offset distance it measures, or every front face reads as "
704 "resolved from the first turn",
705 front_conv,
706 target_distance);
707 }
708
709 // The offset surface's envelope is envelope_size's (no key of its own since 2026-10-20);
710 // the two fields only for the 2D code, which still reads them.
711 offset_envelope = envelope_size;
712 offset_envelope_rel = envelope_size_rel;
713
714 // l_min is relative to the offset distance rather than the bounding box: it is the offset
715 // that has to be resolved. See the declaration.
716 if (min_edge_length_rel < 0) {
717 // The envelope eps expressed as a multiple of target_distance, which is what this
718 // wants: envelope_size, the offset surface's envelope too since offset_envelope went
719 // (2026-10-20), resolved to an absolute length above.
720 min_edge_length_rel = std::max(envelope_size / std::max(target_distance, 1e-16), 1e-12);
721 }
722 if (min_edge_length < 0) {
723 min_edge_length = min_edge_length_rel * target_distance;
724 } else {
725 min_edge_length_rel = min_edge_length / std::max(target_distance, 1e-16);
726 }
727 }
728};
