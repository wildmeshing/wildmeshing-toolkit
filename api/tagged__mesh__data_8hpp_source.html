34template <typename Point>
35MeshData feature_data(
36 const std::vector<std::array<Point, 2>>& segments,
37 const std::vector<Point>& anchors)
38{
39 std::map<std::array<double, 3>, int64_t> rows;
40 MeshData data;
41 std::vector<std::array<double, 3>> vertices;
42 auto vertex = [&](const Point& p) {
43 std::array<double, 3> key{{0, 0, 0}};
44 for (int i = 0; i < p.size(); ++i) key[i] = p[i];
45 auto [it, added] = rows.emplace(key, rows.size());
46 if (added) vertices.push_back(key);
47 return it->second;
48 };
49 auto& E = data.cells[CellType::Edge].connectivity;
50 E.resize(segments.size(), 2);
51 for (size_t i = 0; i < segments.size(); ++i) {
52 E(i, 0) = vertex(segments[i][0]);
53 E(i, 1) = vertex(segments[i][1]);
54 }
55 auto& P = data.cells[CellType::Point].connectivity;
56 P.resize(anchors.size(), 1);
57 for (size_t i = 0; i < anchors.size(); ++i) P(i, 0) = vertex(anchors[i]);
58 data.V.resize(vertices.size(), 3);
59 for (size_t i = 0; i < vertices.size(); ++i)
60 for (int j = 0; j < 3; ++j) data.V(i, j) = vertices[i][j];
61 data.validate();
62 return data;
63}
64// The envelope a component hands on -- outputs["envelope"] on the memory route,
65// <output>_envelope.msh on the file route -- is the REFERENCE it held its tags to: the geometry its
66// envelopes, tubes and anchors were built around (the input, or the simplified input, per
67// envelope_reference), labelled as its output mesh labels what it held there. In 3D a triangle
68// block (an input surface's face tag, a region's tet tag) and an edge block (the feature curves,
69// feature_<i>), in 2D an edge block, each with one `tags` column per label; the input points as a
70// point block without tags, since the mesh writes nothing for them. No eps: a receiver builds its
71// own envelopes around the reference at its own eps, label by label, so that every stage of a
72// chain holds its tags within one eps of the first component's input. Rebuilt from the mesh it
73// receives, the bound would grow by one eps per stage.
74//
75// append_envelope_block() adds one kind of it: the `type` cells of `block` (geometry_data() of
76// the reference's cells), on vertices of their own after the envelope's, row r labelled by the
77// bits of sets[r] (bit j: names[j]); without names, unlabelled. A block without cells adds
78// nothing. One block per type.
79inline void append_envelope_block(
80 MeshData& envelope,
81 const MeshData& block,
82 const CellType type,
83 const std::vector<InputSet>& sets,
84 const std::vector<std::string>& names)
85{
86 const auto& cells = block.cell(type).connectivity;
87 if (cells.rows() == 0) return;
88 if (envelope.cells.count(type))
89 throw std::runtime_error("An envelope has one block of each cell type");
90 if (!names.empty() && sets.size() != size_t(cells.rows()))
91 throw std::runtime_error("An envelope needs one label set per cell");
92 const Eigen::Index base = envelope.V.rows();
93 envelope.V.conservativeResize(base + block.V.rows(), 3);
94 envelope.V.bottomRows(block.V.rows()) = block.V;
95 auto& out = envelope.cells[type];
96 out.connectivity = (cells.array() + int64_t(base)).matrix();
97 if (names.empty()) return;
98 DataMatrix<uint8_t> member = DataMatrix<uint8_t>::Zero(cells.rows(), names.size());
99 for (Eigen::Index r = 0; r < cells.rows(); ++r)
100 for (size_t j = 0; j < names.size() && j < size_t(SurfaceTagAttributes::max_inputs); ++j)
101 member(r, j) = (sets[r] >> j) & 1;
102 out.attributes["tags"] = {std::move(member), names};
103}

117// Whether `envelope` is a labelled reference (append_envelope_block()), as producers write it,
118// rather than bare geometry: its triangles (3D) or edges (2D) carry `tags`. Bare geometry, an
119// `envelope` the caller built, is held to as a whole, as before the labels.
120inline bool is_labelled_envelope(const MeshData& envelope, const bool is_3d)
121{
122 const auto it = envelope.cells.find(is_3d ? CellType::Triangle : CellType::Edge);
123 return it != envelope.cells.end() && it->second.attributes.count("tags");
124}
125// The labels of tags 0 to count - 1, named by `names` (id -> name): a tag's name, or tag_<id>
126// without one, unless that name is taken, in which case '_' is appended as SimWildMesh::write_msh
127// does; labels must be unique.
128inline std::vector<std::string> tag_labels(
129 const std::map<int64_t, std::string>& names,
130 const size_t count)
131{
132 std::set<std::string> named, labelled;
133 for (const auto& [id, name] : names) named.insert(name);
134 std::vector<std::string> labels;
135 for (size_t j = 0; j < count; ++j) {
136 const auto it = names.find(int64_t(j));
137 const bool unnamed = it == names.end();
138 std::string label = unnamed ? "tag_" + std::to_string(j) : it->second;
139 while (labelled.count(label) || (unnamed && named.count(label))) label += "_";
140 labelled.insert(label);
141 labels.push_back(label);
142 }
143 return labels;
144}
145// The labels of an application's inputs, one per input: input_names, the rest tag_<i>, by the
146// rule of tag_labels(). What tetwild's face tags and triwild's edge tags name the inputs by, so
147// that an input's tags are named alike in every component.
148inline std::vector<std::string> input_labels(
149 const std::vector<std::string>& input_names,
150 const size_t n_inputs)
151{
152 std::map<int64_t, std::string> names;
153 for (size_t i = 0; i < input_names.size() && i < n_inputs; ++i)
154 names[int64_t(i)] = input_names[i];
155 return tag_labels(names, n_inputs);
156}
157template <typename Mesh, typename TupleRange, typename Tags>
158void collect_tags(Mesh& mesh, MeshMap& outputs, CellType type, const TupleRange& cells, Tags tags)
159{
160 // A result without cells has no block yet: the exporter only creates the ones it fills.
161 auto& block = outputs.at("mesh").cells[type];
162 if (block.connectivity.cols() != cell_size(type)) block.connectivity.resize(0, cell_size(type));
163 std::vector<std::string> labels = tag_labels(mesh.m_tag_id_to_name, mesh.m_tags_count);
164 DataMatrix<uint8_t> values = DataMatrix<uint8_t>::Zero(cells.size(), labels.size());
165 for (size_t i = 0; i < cells.size(); ++i)
166 for (size_t j = 0; j < labels.size(); ++j) values(i, j) = tags(cells[i]).count(j) ? 1 : 0;
167 // The fixed tags field replaces per-group named scalar arrays.
168 for (size_t j = 0; j < labels.size(); ++j) {
169 block.attributes.erase(labels[j]);
170 block.attributes.erase("tag_" + std::to_string(j));
171 }
172 // Cells without a tag are the background the component built. They are labelled
173 // explicitly: the IO infers nothing from an empty row.
174 if (std::find(labels.begin(), labels.end(), "ambient") == labels.end()) {
175 const auto empty = (values.template cast<int>().rowwise().sum().array() == 0).eval();
176 if (labels.empty() || empty.any()) {
177 values.conservativeResize(Eigen::NoChange, values.cols() + 1);
178 values.col(values.cols() - 1) = empty.template cast<uint8_t>();
179 labels.push_back("ambient");
180 }
181 }
182 block.attributes["tags"] = {std::move(values), std::move(labels)};
183}
184// The tracked simplices -- `type` Triangle: a tet mesh's faces; Edge: a triangle mesh's edges --
185// whose input set is not empty, as the `type` block of outputs["mesh"]: rows of its V (the
186// vertices in get_vertices() order, as compact_slot_outputs leaves them), one tags column per
187// name (bit j of the input set is names[j]), each row oriented as the simplex's input was.
188// Nothing is written without names.
189template <CellType type, typename Mesh>
190void collect_input_set_tags(
191 const Mesh& mesh,
192 MeshMap& outputs,
193 const std::vector<std::string>& names)
194{
195 static_assert(type == CellType::Triangle || type == CellType::Edge);
196 constexpr size_t N = type == CellType::Triangle ? 3 : 2;
197 if (names.empty()) return;
198 std::vector<int64_t> row(mesh.vert_capacity(), -1);
199 int64_t n = 0;
200 for (const auto& v : mesh.get_vertices()) row[v.vid(mesh)] = n++;
201 std::vector<std::array<int64_t, N>> simplices;
202 std::vector<InputSet> sets;
203 const InputSet mask = names.size() >= size_t(SurfaceTagAttributes::max_inputs)
204 ? ~InputSet(0)
205 : (InputSet(1) << names.size()) - 1;
206 const auto add = [&](const SurfaceTagAttributes& attr, std::array<size_t, N> vids) {
207 // Bits past the names are not these tags -- simwild's region labels -- and are not
208 // written.
209 const InputSet set = attr.input_set() & mask;
210 if (set == 0) return;
211 std::sort(vids.begin(), vids.end());
212 if (attr.orientation_along(vids) < 0) std::swap(vids[N - 2], vids[N - 1]);
213 std::array<int64_t, N> r;
214 for (size_t k = 0; k < N; ++k) r[k] = row[vids[k]];
215 simplices.push_back(r);
216 sets.push_back(set);
217 };
218 if constexpr (type == CellType::Triangle) {
219 for (const auto& f : mesh.get_faces())
220 add(mesh.m_face_attribute[f.fid(mesh)], mesh.get_face_vids(f));
221 } else {
222 for (const auto& e : mesh.get_edges())
223 add(mesh.m_edge_attribute[e.eid(mesh)], mesh.get_edge_vids(e));
224 }
225 auto& block = outputs.at("mesh").cells[type];
226 block.connectivity.resize(simplices.size(), N);
227 DataMatrix<uint8_t> member = DataMatrix<uint8_t>::Zero(simplices.size(), names.size());
228 for (size_t i = 0; i < simplices.size(); ++i) {
229 for (size_t k = 0; k < N; ++k) block.connectivity(i, k) = simplices[i][k];
230 for (size_t j = 0; j < names.size(); ++j) member(i, j) = (sets[i] >> j) & 1;
231 }
232 block.attributes["tags"] = {std::move(member), names};
233}
234// The tet edges tagged `feature` (TetOptimizerMesh::FeatureEdgeAttributes: the feature curves the
235// application was given) whose feature input set is not empty, as the Edge block of
236// outputs["mesh"], on the rows collect_input_set_tags() uses: one tags column per name (bit j of
237// the set is names[j], the input_edges file j). A curve carries no orientation, so a row is its
238// edge's vertices in ascending order. The order2 kind is not written: a reader derives it again
239// from the faces (tag_order2_edges()). Nothing is written without names or edge tracking.
240template <typename Mesh>
241void collect_feature_edge_tags(
242 const Mesh& mesh,
243 MeshMap& outputs,
244 const std::vector<std::string>& names)
245{
246 using EdgeTags = typename Mesh::FeatureEdgeAttributes;
247 if (names.empty() || !mesh.m_track_feature_edges) return;
248 std::vector<int64_t> row(mesh.vert_capacity(), -1);
249 int64_t n = 0;
250 for (const auto& v : mesh.get_vertices()) row[v.vid(mesh)] = n++;
251 const InputSet mask = names.size() >= size_t(SurfaceTagAttributes::max_inputs)
252 ? ~InputSet(0)
253 : (InputSet(1) << names.size()) - 1;
254 std::vector<std::array<int64_t, 2>> edges;
255 std::vector<InputSet> sets;
256 for (const auto& e : mesh.get_edges()) {
257 const auto& attr = mesh.m_feature_edge_attribute[e.eid(mesh)];
258 if (!attr.has(EdgeTags::feature)) continue;
259 const InputSet set = attr.inputs(EdgeTags::feature) & mask;
260 if (set == 0) continue;
261 const size_t a = e.vid(mesh), b = e.switch_vertex(mesh).vid(mesh);
262 edges.push_back({{row[std::min(a, b)], row[std::max(a, b)]}});
263 sets.push_back(set);
264 }
265 auto& block = outputs.at("mesh").cells[CellType::Edge];
266 block.connectivity.resize(edges.size(), 2);
267 DataMatrix<uint8_t> member = DataMatrix<uint8_t>::Zero(edges.size(), names.size());
268 for (size_t i = 0; i < edges.size(); ++i) {
269 for (size_t k = 0; k < 2; ++k) block.connectivity(i, k) = edges[i][k];
270 for (size_t j = 0; j < names.size(); ++j) member(i, j) = (sets[i] >> j) & 1;
271 }
272 block.attributes["tags"] = {std::move(member), names};
273}
