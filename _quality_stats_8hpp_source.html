65template <class ValidFn, class TagsFn, class QualityFn>
66QualityBreakdown collect_quality_breakdown(
67 const size_t capacity,
68 const std::vector<std::tuple<expression_parser::ExpressionPtr, double>>& quality_field,
69 const double base_target,
70 ValidFn&& is_valid,
71 TagsFn&& tags_of,
72 QualityFn&& quality_of)
73{
74 QualityBreakdown out;
75 out.groups.resize(quality_field.size() + 1);
76 out.groups[0].label = "(default)";
77 out.groups[0].target = base_target;
78 for (size_t i = 0; i < quality_field.size(); ++i) {
79 const auto& [expr, target] = quality_field[i];
80 out.groups[i + 1].label = expr->to_string();
81 out.groups[i + 1].target = target;
82 }
83
84 for (size_t cid = 0; cid < capacity; ++cid) {
85 if (!is_valid(cid)) {
86 continue;
87 }
88 const auto& tags = tags_of(cid);
89
90 // Last match wins, mirroring target_quality. Scanning backwards finds it first, and the
91 // common case -- a cell matching nothing -- still costs the whole field either way.
92 size_t group = 0;
93 for (size_t i = quality_field.size(); i-- > 0;) {
94 if (std::get<0>(quality_field[i])->eval(tags)) {
95 group = i + 1;
96 break;
97 }
98 }
99
100 const double quality = quality_of(cid);
101 QualityGroup& g = out.groups[group];
102 g.max_quality = std::max(g.max_quality, quality);
103 g.sum_quality += quality;
104 ++g.count;
105
106 const double relative = quality / g.target;
107 out.max_relative = std::max(out.max_relative, relative);
108 out.avg_relative += relative;
109 ++out.count;
110 }
111
112 if (out.count > 0) {
113 out.avg_relative /= out.count;
114 }
115 return out;
116}

126inline void log_quality_breakdown(const QualityBreakdown& breakdown)
127{
128 std::vector<const QualityGroup*> order;
129 order.reserve(breakdown.groups.size());
130 for (const auto& g : breakdown.groups) {
131 order.push_back(&g);
132 }
133 std::sort(order.begin(), order.end(), [](const QualityGroup* a, const QualityGroup* b) {
134 return a->max_relative() > b->max_relative();
135 });
136
137 logger().info(
138 "Quality by tag ({} cells, max rel = {:.4}, avg rel = {:.4}):",
139 breakdown.count,
140 breakdown.max_relative,
141 breakdown.avg_relative);
142 for (const QualityGroup* g : order) {
143 if (g->count == 0) {
144 logger().info(" {:<20} no cells", g->label);
145 continue;
146 }
147 logger().info(
148 " {:<20} n = {:>8}, max = {:>8.6}, avg = {:>8.6}, target = {:>6.6}, max rel = {:6>.4}",
149 g->label,
150 g->count,
151 g->max_quality,
152 g->avg_quality(),
153 g->target,
154 g->max_relative());
155 }
