568 if (!m_high_valence_claim) return;
569 const size_t n = std::max(vert_capacity(), m_vertex_attribute.size());
570 if (n <= m_high_valence_claim_size) return;

572 for (size_t i = 0; i < m_high_valence_claim_size; ++i) {
573 grown[i].store(m_high_valence_claim[i].load(std::memory_order_relaxed));
574 }
575 m_high_valence_claim = std::move(grown);
