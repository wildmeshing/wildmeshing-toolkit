979 // ---- OPERATION ACCOUNTING (counting only) ----
980 // Every call into split_edge, collapse_edge, swap_edge, swap_edge_44, swap_edge_56 and
981 // swap_face reports `attempt` on entry and exactly ONE other event on the way out: `committed`,
982 // or the place that refused it. The application's hooks are two of those places (before_hook,
983 // after_hook); why a hook said no is the hook's own to count. So for each kind, attempt equals
984 // the sum of the other events, and a subclass that counts can check both that identity and a
985 // hook's refusals against the reasons the hook counted. Added because the refusals made here,
986 // below every hook, were visible only as a difference: on the deliverable cube at
987 // target_distance_rel 1e-2 / front_conv_rel 1e-4, 521142 of the 542123 edge swaps that
988 // passed every before-hook in turn 5 were refused inside this class with no reason recorded.
989 //
990 // The default does nothing and no site reads a count: the mesh behaves identically with or
991 // without an override.
992 enum class OpKind : int { split, collapse, swap_32, swap_44, swap_56, swap_face, COUNT };
993 enum class OpEvent : int {
994 attempt,
995 before_hook, // the application's *_before said no
996 link_condition, // collapse: the classical link condition (m_collapse_check_link_condition)
997 no_return_tet, // collapse: both probes for the return tuple hit a boundary face
998 duplicate_tet, // a new tet repeats an existing one (collapse manifold check, 4-4, 5-6)
999 topology, // collapse: m_collapse_check_topology found a face missing
1000 valence, // edge swap: the edge does not have this swap's number of tets around it
1001 boundary, // edge swap: the edge's ring of tets is open; face swap: a boundary face
1002 exists, // 3-2: the new face already exists; face swap: the new edge already exists
1003 no_case_allowed, // 4-4 / 5-6: every candidate was vetoed by *_accept_case
1004 no_better_case, // 4-4 / 5-6: no allowed candidate scored below the current cells
1005 out_of_slots, // the preallocated vertex or tet slots ran out
1006 after_hook, // the application's *_after said no, so the operation was rolled back
1007 invariants, // invariants() said no after the after-hook passed; rolled back
1008 committed,
1009 screened, // a dry run passed every check that precedes the change (see dry_run())
1010 COUNT
1011 };

1482 void remove_tets_by_ids(const std::vector<size_t>& tids)
1483 {
1484 // A face or edge attribute lives at ONE canonical slot, (lowest incident tet) * 4
1485 // or 6 + local index. Deleting a tet that is a simplex's canonical owner silently
1486 // re-canonicalizes the simplex onto a surviving tet whose cell was never
1487 // maintained -- the operations only ever write the canonical slot -- so the
1488 // surviving cell holds stale garbage. Measured: after filtering, interior faces
1489 // read back m_is_surface_fs = true 0.12 (eps 0.09) from the real surface, and the
1490 // extracted "tracked surface" grows phantom triangles. Migrate every attribute
1491 // whose owner dies to the simplex's new owner BEFORE touching the connectivity.
1492 if (p_face_attrs != nullptr || p_edge_attrs != nullptr) {
1493 std::vector<bool> removed(tet_capacity(), false);
1494 for (const size_t tid : tids) {
1495 removed[tid] = true;
1496 }
1497 for (const size_t tid : tids) {
1498 const auto& tconn = m_tet_connectivity[tid];
1499 if (p_face_attrs != nullptr) {
1500 for (int lf = 0; lf < 4; ++lf) {
1501 std::array<size_t, 3> vids = {
1502 {tconn[m_local_faces[lf][0]],
1503 tconn[m_local_faces[lf][1]],
1504 tconn[m_local_faces[lf][2]]}};
1505 // The face's other incident tet, if any.
1506 auto common = set_intersection(
1507 m_vertex_connectivity[vids[0]].m_conn_tets,
1508 m_vertex_connectivity[vids[1]].m_conn_tets);
1509 common =
1510 set_intersection(common, m_vertex_connectivity[vids[2]].m_conn_tets);
1511 size_t other = std::numeric_limits<size_t>::max();
1512 for (const size_t t : common) {
1513 if (t != tid) {
1514 other = t;
1515 }
1516 }
1517 // Migrate only when this tet was the canonical owner (tid < other)
1518 // and the other side survives; other < tid means the attribute
1519 // already lives on the survivor.
1520 if (other == std::numeric_limits<size_t>::max() || removed[other] ||
1521 other < tid) {
1522 continue;
1523 }
1524 std::array<size_t, 3> sorted = vids;
1525 std::sort(sorted.begin(), sorted.end());
1526 for (int lo = 0; lo < 4; ++lo) {
1527 std::array<size_t, 3> ovids = {
1528 {m_tet_connectivity[other][m_local_faces[lo][0]],
1529 m_tet_connectivity[other][m_local_faces[lo][1]],
1530 m_tet_connectivity[other][m_local_faces[lo][2]]}};
1531 std::sort(ovids.begin(), ovids.end());
1532 if (ovids == sorted) {
1533 p_face_attrs->move(tid * 4 + lf, other * 4 + lo);
1534 break;
1535 }
1536 }
1537 }
1538 }
1539 if (p_edge_attrs != nullptr) {
1540 for (int le = 0; le < 6; ++le) {
1541 const size_t v0 = tconn[m_local_edges[le][0]];
1542 const size_t v1 = tconn[m_local_edges[le][1]];
1543 const auto fan = set_intersection(
1544 m_vertex_connectivity[v0].m_conn_tets,
1545 m_vertex_connectivity[v1].m_conn_tets);
1546 // Only the canonical owner migrates, once, to the lowest survivor.
1547 if (fan.empty() || fan.front() != tid) {
1548 continue;
1549 }
1550 size_t new_owner = std::numeric_limits<size_t>::max();
1551 for (const size_t t : fan) {
1552 if (!removed[t]) {
1553 new_owner = t; // fan is sorted ascending; first kept wins
1554 break;
1555 }
1556 }
1557 if (new_owner == std::numeric_limits<size_t>::max()) {
1558 continue; // the edge dies with its whole fan
1559 }
1560 for (int lo = 0; lo < 6; ++lo) {
1561 const size_t w0 = m_tet_connectivity[new_owner][m_local_edges[lo][0]];
1562 const size_t w1 = m_tet_connectivity[new_owner][m_local_edges[lo][1]];
1563 if ((w0 == v0 && w1 == v1) || (w0 == v1 && w1 == v0)) {
1564 p_edge_attrs->move(tid * 6 + le, new_owner * 6 + lo);
1565 break;
1566 }
1567 }
1568 }
1569 }
1570 }
1571 }
1572
1573 for (size_t tid : tids) {
1574 m_tet_connectivity[tid].m_is_removed = true;
1575 for (int j = 0; j < 4; j++)
1576 vector_erase(m_vertex_connectivity[m_tet_connectivity[tid][j]].m_conn_tets, tid);
1577 }
1578 for (auto& v : m_vertex_connectivity) {
1579 if (v.m_is_removed) continue;
1580 if (v.m_conn_tets.empty()) v.m_is_removed = true;
1581 }
1582 if (m_surface_history) resync_surface_history();
