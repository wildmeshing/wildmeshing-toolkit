2229 const std::array<size_t, 2>& vids,
2230 const EdgeAttributes& /*attr*/) const override
2231 {
2232 // Boundary geometry first: a segment on any tag-region boundary may not
2233 // drift out of that boundary's tube, and one on several boundaries -- a junction -- is
2234 // held in their intersection. The mask carries the input complex too, since every complex
2235 // simplex lies on tag boundaries, and E_t is built from the input mesh before construction
2236 // touches it, so the per-tag tubes hold the as-loaded geometry.
2237 //
2238 // Keyed on the vertices, not the edge: every caller is an operation asking about a segment
2239 // it is about to create or has just created, whose own edge attributes are not written
2240 // yet. The endpoints' masks are, maintained by the operations themselves (AND at a split,
2241 // OR at a collapse).
2242 uint64_t mask = edge_mask(vids);
2243 bool all_offset = true;
2244 for (const size_t v : vids) {
2245 all_offset = all_offset && m_vertex_extra[v].m_is_on_offset;
2246 }
2247
2248 // The ambiguous case: both endpoints can be on region boundaries and on the offset front
2249 // at once -- a real state, and the point of tracking the two families separately. The
2250 // endpoint-mask AND is then necessary but not sufficient for the segment lying on a shared
2251 // boundary: two vertices on different junctions can share a tag bit by coincidence and be
2252 // joined by an offset chord nowhere near that tag's curve, which a container it was never
2253 // meant to satisfy then refuses to refine. Ask the edge's own class, the only record that
2254 // distinguishes a chord from a boundary.
2255 //
2256 // Reading the slot is safe here and would not be unconditionally: the base checks a
2257 // split's two child segments before writing their attributes, so their slots still hold a
2258 // recycled edge's class. A split child never reaches this branch -- its mask and its
2259 // offset flag are mutually exclusive by construction, so one of `mask` and `all_offset` is
2260 // always empty -- and the m_is_surface_fs guard leaves both constraints standing rather
2261 // than dropping one when a slot is illegible.
2262 if (mask != 0 && all_offset) {
2263 if (const auto found = try_tuple_from_edge(vids)) {
2264 const size_t eid = std::get<1>(*found);
2265 if (m_edge_attribute[eid].m_is_surface_fs) {
2266 if (edge_is_offset(eid)) {
2267 mask = 0; // an offset edge lies on no region boundary
2268 } else {
2269 all_offset = false; // a region edge is not the offset front
2270 }
2271 }
2272 }
2273 }
2274 // Both families compose: whatever holds this segment holds it at once. The offset side
2275 // holds only in the frozen-front final pass (see containment_for()); in the loop the
2276 // front is what the optimization moves, so it contributes nothing there.
2277 const std::shared_ptr<Envelope> base = containment_for(mask, all_offset);
2278 if (base || m_deform_tags.empty()) return base;
2279 // deform_others' ops-only tube: a released boundary is held by no mask -- its vertices
2280 // were freed so smoothing can carry the object -- which would leave the operations free
2281 // to decimate and reposition it. A segment the masks and the offset class do not claim,
2282 // but which lies on a released boundary by its incident faces' current tags, is held to
2283 // the tube around the boundary's current shape. A fresh split child can misread its
2284 // recycled face slots for this one check, and is at worst skipped or over-held once.
2285 if (const auto found = try_tuple_from_edge(vids)) {
2286 if (edge_borders_released_boundary(std::get<0>(*found))) return released_envelope();
2287 }

2293 // of a complex with faces arises only where two or more selected tags meet (the selection
2294 // labels an isolated simplex only where its face star is tag-heterogeneous), so the edges at

2670 template <typename Visit>
2671 void for_each_edge_sample(const Vector2d& p0, const Vector2d& p1, Visit&& visit) const
2672 {
2673 const int k = m_offset_params.stencil_order;
2674 if (k < 0) return;
2675
2676 const auto emit = [&](const double wa, const double wb) {
2677 visit(Vector2d(wa * p0 + wb * p1), wa, wb);
2678 };
2679 // Order 0 is the two ENDS alone, each end's own placement error.
2680 if (k == 0) {
2681 emit(1., 0.);
2682 emit(0., 1.);
2683 return;
2684 }
2685 // Order k >= 1: the vertices of the chord cut into n = 2^(k-1) pieces, then the midpoint
2686 // of each piece -- 2n + 1 points: 3, 5, 9, 17, ...
2687 const int n = 1 << (k - 1);
2688 const double dn = double(n);
2689 for (int i = n; i >= 0; --i) {
2690 emit(double(i) / dn, double(n - i) / dn);
2691 }
2692 const double d2n = 2. * dn;
2693 for (int i = n - 1; i >= 0; --i) {
2694 emit((2. * i + 1.) / d2n, (2. * (n - 1 - i) + 1.) / d2n);
2695 }
2696 }
