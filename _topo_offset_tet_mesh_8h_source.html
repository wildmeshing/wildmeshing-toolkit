971 std::shared_ptr<SampleEnvelope> surface_envelope_for_face(
972 const std::array<size_t, 3>& vids) const override
973 {
974 uint64_t mask = face_mask(vids);
975 bool all_offset = true;
976 for (const size_t v : vids) {
977 all_offset = all_offset && m_vertex_extra[v].m_is_on_offset;
978 }
979 // The ambiguous case: all corners can be on region boundaries and on the offset surface
980 // at once. The corner-mask AND is then necessary but not sufficient for the face lying
981 // on a shared boundary; ask the face's own class, the only record that distinguishes a
982 // chord from a boundary. Reading the slot is safe here: a split child never reaches
983 // this branch, and the m_is_surface_fs guard leaves both constraints standing when a
984 // slot is illegible.
985 if (mask != 0 && all_offset) {
986 if (const auto found = try_tuple_from_face(vids)) {
987 const size_t fid = std::get<1>(*found);
988 if (m_face_attribute[fid].m_is_surface_fs) {
989 if (face_is_offset(fid)) {
990 mask = 0; // an offset face lies on no region boundary
991 } else {
992 all_offset = false; // a region face is not the offset surface
993 }
994 }
995 }
996 }
997 const std::shared_ptr<SampleEnvelope> base = containment_for(mask, all_offset);
998 if (base || m_deform_tags.empty()) return base;
999 // deform_others' ops-only tube: a released boundary is held by no mask -- its vertices
1000 // were freed so smoothing can carry the object -- which would leave the operations free
1001 // to decimate and reposition it. A face the masks and the offset class do not claim, but
1002 // which lies on a released boundary by its incident tets' current tags, is held to the
1003 // tube around the boundary's current shape.
1004 if (const auto found = try_tuple_from_face(vids)) {
1005 if (face_borders_released_boundary(std::get<0>(*found))) return released_envelope();
1006 }
1007 return nullptr;
1008 }
