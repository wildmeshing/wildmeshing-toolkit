1087 std::shared_ptr<SampleEnvelope> surface_envelope_for_edge(
1088 const std::array<size_t, 2>& vids) const override
1089 {
1090 // Boundary geometry first, in both phases: a segment on any tag-region boundary may not
1091 // drift out of that boundary's tube, and one on several boundaries -- a junction -- is
1092 // held in their intersection. The mask carries the input complex too, since every complex
1093 // simplex lies on tag boundaries, and E_t is built from the input mesh before construction
1094 // touches it, so the per-tag tubes hold the as-loaded geometry.
1095 //
1096 // Keyed on the vertices, not the edge: every caller is an operation asking about a segment
1097 // it is about to create or has just created, whose own edge attributes are not written
1098 // yet. The endpoints' masks are, maintained by the operations themselves (AND at a split,
1099 // OR at a collapse).
1100 uint64_t mask = edge_mask(vids);
1101 bool all_offset = true;
1102 for (const size_t v : vids) {
1103 all_offset = all_offset && m_vertex_extra[v].m_is_on_offset;
1104 }
1105
1106 // The ambiguous case: both endpoints can be on region boundaries and on the offset front
1107 // at once -- a real state, and the point of tracking the two families separately. The
1108 // endpoint-mask AND is then necessary but not sufficient for the segment lying on a shared
1109 // boundary: two vertices on different junctions can share a tag bit by coincidence and be
1110 // joined by an offset chord nowhere near that tag's curve, which a container it was never
1111 // meant to satisfy then refuses to refine. Ask the edge's own class, the only record that
1112 // distinguishes a chord from a boundary.
1113 //
1114 // Reading the slot is safe here and would not be unconditionally: the base checks a
1115 // split's two child segments before writing their attributes, so their slots still hold a
1116 // recycled edge's class. A split child never reaches this branch -- its mask and its
1117 // offset flag are mutually exclusive by construction, so one of `mask` and `all_offset` is
1118 // always empty -- and the m_is_surface_fs guard leaves both constraints standing rather
1119 // than dropping one when a slot is illegible.
1120 if (mask != 0 && all_offset) {
1121 if (const auto found = try_tuple_from_edge(vids)) {
1122 const size_t eid = std::get<1>(*found);
1123 if (m_edge_attribute[eid].m_is_surface_fs) {
1124 if (edge_is_offset(eid)) {
1125 mask = 0; // an offset edge lies on no region boundary
1126 } else {
1127 all_offset = false; // a region edge is not the offset front
1128 }
1129 }
1130 }
1131 }
1132 // Both families compose: whatever holds this segment holds it at once. Phase A holds the
1133 // offset where Phase B left it; Phase B is what moves it, so it contributes nothing there
1134 // -- and null before the offset exists at all, which is the pre-pass.
1135 const std::shared_ptr<SampleEnvelope> base = containment_for(mask, all_offset);
1136 if (base || m_deform_tags.empty()) return base;
1137 // deform_others' ops-only tube: a released boundary is held by no mask -- its vertices
1138 // were freed so smoothing can carry the object -- which would leave the operations free
1139 // to decimate and reposition it. A segment the masks and the offset class do not claim,
1140 // but which lies on a released boundary by its incident faces' current tags, is held to
1141 // the tube around the boundary's current shape. A fresh split child can misread its
1142 // recycled face slots for this one check, and is at worst skipped or over-held once.
1143 if (const auto found = try_tuple_from_edge(vids)) {
1144 if (edge_borders_released_boundary(std::get<0>(*found))) return released_envelope();
1145 }
1146 return nullptr;
1147 }

1179 std::shared_ptr<SampleEnvelope> smoothing_energy_envelope(const size_t vid) const override
1180 {
1181 if (m_vertex_extra[vid].m_is_on_offset && !vertex_is_on_region(vid)) {
1182 return nullptr;
1183 }
1184 const uint64_t mask = vertex_boundary_mask(vid);
1185 if (mask == 0) {
1186 // Reachable only for a construction artefact (a wall-chord midpoint flagged
1187 // on-surface with disjoint endpoint masks); its containment is vacuous too, so no
1188 // pull is behavior-neutral. Not an error.
1189 return nullptr;
1190 }
1191 std::shared_ptr<SampleEnvelope> best;
1192 double worst_d2 = -1.;
1193 for (const auto& [tag, env] : m_tag_envelopes) {
1194 const auto it = m_tag_bit.find(tag);
1195 if (it == m_tag_bit.end() || !(mask & (uint64_t(1) << it->second))) continue;
1196 if (!best) {
1197 best = env;
1198 if ((mask & (mask - 1)) == 0) break; // single bit: no violation contest to run
1199 worst_d2 = env->squared_distance(m_vertex_attribute[vid].m_posf);
1200 continue;
1201 }
1202 const double d2 = env->squared_distance(m_vertex_attribute[vid].m_posf);
1203 if (d2 > worst_d2) {
1204 worst_d2 = d2;
1205 best = env;
1206 }
1207 }
1208 return best;
1209 }
