30 // A history the component recorded (wmtk/history; its history_output), of a surface or of
31 // curves. The file route saves it to `path`. The memory route, which writes no file, returns
32 // the bytes of that file in the report, as "history" -- a JSON binary value, bytes in Python
33 // (take_result()).
34 template <int N>
35 void save_history(const history::History<N>& h, const std::string& path)
36 {

97 // Mesh fields come in two kinds. DATA is what a component reads: `tags`, the label columns of
98 // a cell block, on every route; the envelope mesh; and the named fields of data_fields. Every
99 // other field is an ANNOTATION, written for people -- tetwild's energy, triwild's quality,
100 // the inputs' winding numbers wn_<name>, the offset's diagnostics -- and every reader drops
101 // it: leaves it unread and names it once per read, at info (dropped_fields(), log_dropped()).
102 // Until 2026-11 each reader listed the fields it took and threw `Unknown mesh attribute` on
103 // any other, so a new field had to be added to every list or a chain broke: triwild's
104 // wn_<name> broke triwild -> simwild on the memory route.
105 //
106 // The named data fields, each with who reads it; a writer that trims its result to data
107 // (topological_offset's) keeps them. Besides them, a call may name fields it reads
108 // (isotropic_remeshing's feature_fields), tracking's transfer carries every triangle field,
109 // and simwild reads 2D label images, cell fields tag_<k> (tags_from_label_images()).
110 inline static const std::set<std::string> data_fields = {
111 "vid", // vertex: the ids corr_input_vid names: simwild, prismatic_mesh
112 "corr_input_vid", // vertex: an offset's correspondence: simwild, prismatic_mesh
113 "labels", // vertex: the offset's vertex labels: prismatic_mesh
114 "is_cone", // vertex: c1_simplification
115 "vertex_id", // vertex and point: isotropic_remeshing's features
116 "offset_tag", // tet: the offset's band: prismatic_mesh
117 "tag_0", // tet: the input region as a field: prismatic_mesh
118 "edge_id", // edge: isotropic_remeshing's features
119 "dofs", // triangle: c1_simplification
120 "uv", // triangle, per corner: isotropic_remeshing's feature_fields, tracking
121 "normal", // triangle, per corner: as uv
122 };

128 [[nodiscard]] static std::set<std::string> dropped_fields(
129 const MeshAttributes& a,
130 const std::set<std::string>& read,
131 const std::set<std::string>& required = {})
132 {
133 for (const auto& name : required)
134 if (!a.count(name))
135 throw std::runtime_error("Missing required mesh attribute: " + name);
136 std::set<std::string> dropped;
137 for (const auto& [name, field] : a)
138 if (!read.count(name)) dropped.insert(name);

154 enum class InputKind { surface, curve, points };
155 InputKind input_kind(const std::string& name) const
156 {
157 if (memory()) {
158 const auto& data = mesh(name);
159 const auto has = [&](const CellType type) {
160 const auto it = data.cells.find(type);
161 return it != data.cells.end() && it->second.connectivity.rows() > 0;
162 };
163 if (has(CellType::Triangle)) return InputKind::surface;
164 if (has(CellType::Edge)) return InputKind::curve;
165 return InputKind::points;
166 }
167 return file_input_kind(name);
168 }
170 static InputKind file_input_kind(const std::string& name)
171 {
172 std::string extension = std::filesystem::path(name).extension().string();
173 for (char& c : extension) c = char(std::tolower(static_cast<unsigned char>(c)));
174 if (extension != ".obj") return InputKind::surface;
175 std::ifstream in(name);
176 if (!in) log_and_throw_error("Could not open input file {}", name);
177 bool has_l = false;
178 std::string line;
179 while (std::getline(in, line)) {
180 if (line.size() < 2 || (line[1] != ' ' && line[1] != '\t')) continue;
181 if (line[0] == 'f') return InputKind::surface;
182 if (line[0] == 'l') has_l = true;
183 }
184 return has_l ? InputKind::curve : InputKind::points;
185 }

200 // One named input, uncleaned: the memory twin of reading one file. Its fields are dropped
201 // (dropped_fields()) but `triangle_attributes`, which the component reads itself
202 // (isotropic_remeshing's feature_fields).
203 void raw_triangles(
204 const std::string& name,
205 MatrixXd& V,
206 MatrixXi& F,
207 const std::set<std::string>& triangle_attributes = {}) const
208 {
209 const auto& data = mesh(name);
210 std::set<std::string> dropped = dropped_fields(data.vertex_attributes, {});
211 for (const auto& [type, c] : data.cells) {
212 if (type != CellType::Triangle)
213 throw std::runtime_error("Expected triangle input: " + name);
214 dropped.merge(dropped_fields(c.attributes, triangle_attributes));
215 }
216 log_dropped("input " + name, dropped);
217 V = data.V;
218 F = indices(data.cell(CellType::Triangle).connectivity);
219 if (!F.rows()) throw std::runtime_error("Empty triangle input: " + name);
220 }

236 // Like read_triangle_mesh(paths): concatenate the raw inputs, then clean them once, so
237 // vertex merging sees the same vertices (isolated ones included) in the same order.
238 // `read`, as there: the inputs as given, concatenated, and where their vertices and faces
239 // went. `triangle_attributes`, as in raw_triangles.
240 void triangles(
241 const std::vector<std::string>& names,
242 MatrixXd& V,
243 MatrixXi& F,
244 double rel,
245 double abs,
246 std::vector<int>* provenance = nullptr,
247 ReadProvenance* read = nullptr,
248 const std::set<std::string>& triangle_attributes = {}) const
249 {
250 if (!memory()) {
251 read_triangle_mesh(names, V, F, rel, abs, provenance, read);
252 return;
253 }
254 V.resize(0, 3);
255 F.resize(0, 3);
256 if (provenance) provenance->clear();
257 for (size_t i = 0; i < names.size(); ++i) {
258 MatrixXd v;
259 MatrixXi f;
260 raw_triangles(names[i], v, f, triangle_attributes);
261 const auto nv = V.rows(), nf = F.rows();
262 f.array() += nv;
263 V.conservativeResize(nv + v.rows(), 3);
264 V.bottomRows(v.rows()) = v;
265 F.conservativeResize(nf + f.rows(), 3);
266 F.bottomRows(f.rows()) = f;
267 if (provenance) provenance->resize(F.rows(), int(i));
268 }
269 clean_triangle_mesh(V, F, rel, abs, provenance, read);
270 if (!F.rows()) throw std::runtime_error("No faces after cleaning");
271 }

320template <typename Mesh, typename Position>
321MeshData triangle_data(const Mesh& m, Position position)
322{
323 MeshData data;
324 const auto vs = m.get_vertices();
325 const auto fs = m.get_faces();
326 data.V.resize(vs.size(), 3);
327 std::map<size_t, int64_t> rows;
328 for (size_t i = 0; i < vs.size(); ++i) {
329 rows[vs[i].vid(m)] = i;
330 data.V.row(i) = position(vs[i].vid(m));
331 }
332 auto& c = data.cells[CellType::Triangle].connectivity;
333 c.resize(fs.size(), 3);
334 for (size_t i = 0; i < fs.size(); ++i) {
335 auto v = m.oriented_tri_vertices(fs[i]);
336 for (int j = 0; j < 3; ++j) c(i, j) = rows.at(v[j].vid(m));
337 }
338 data.validate();
339 return data;
340}
