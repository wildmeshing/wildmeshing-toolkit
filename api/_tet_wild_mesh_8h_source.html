150 // The open boundaries are held by the base: tagged at init (tag_order2_edges()), every
151 // tagged edge a collapse keeps, or a smoothing move touches, must stay in
152 // m_order2_edge_envelope, no swap removes one, and a collapse that leaves one with other
153 // than one face drops its tag. tetwild's own open-boundary machinery is gone with it: the
154 // per-vertex flags, their tube m_order2_envelope (the same tube as the base's), the
155 // collapse's point test (a flagged vertex collapses only onto a flagged vertex or into the
156 // tube, implied by the base's segment test whenever the vertex has a tagged edge) and
157 // is_open_boundary_edge(), which classified edges by the flags and the tube.

180 // only used with unit tests
181 void create_mesh_attributes(
182 const std::vector<VertexAttributes>& _vertex_attribute,
183 const std::vector<TetAttributes>& _tet_attribute)
184 {
185 const size_t n_tet = _tet_attribute.size();
186 m_vertex_attribute.resize(_vertex_attribute.size());
187 m_feature_point_attribute.resize(_vertex_attribute.size());
188 m_face_attribute.resize(4 * n_tet);
189 m_tet_attribute.resize(n_tet);
190
191 // new for edge
192 // m_edge_attribute.resize(6 * n_tet);
193
194 for (size_t i = 0; i < _vertex_attribute.size(); i++)
195 m_vertex_attribute[i] = _vertex_attribute[i];
196
197 // Keep whatever init() reserved. AttributeCollection::resize only ever grows, so the
198 // calls above cannot shrink a collection -- but assigning m_attributes directly does,
199 // and it used to drop the tet attributes to exactly n_tet. The attributes have to stay
200 // at least as large as the connectivity: an operation that creates a new element
201 // indexes them by its id, and a 5->6 swap creates one. That left m_tet_attribute one
202 // short of the tet the swap adds, and AttributeCollection::operator[] read past the end
203 // of it -- ASan reports a heap-buffer-overflow reached from swap_edge_56_after, and
204 // libstdc++ then aborts on the corrupted heap a few allocations later, while libc++
205 // carries on and the tests pass.
206 const size_t tcap = std::max(n_tet, m_tet_attribute.size());
207 m_tet_attribute.m_attributes = std::vector<TetAttributes>(tcap);
208 for (size_t i = 0; i < n_tet; i++) m_tet_attribute[i] = _tet_attribute[i];
209 for (size_t i = 0; i < n_tet; i++)
210 m_tet_attribute[i].m_quality = get_quality(tuple_from_tet(i));
211 }
