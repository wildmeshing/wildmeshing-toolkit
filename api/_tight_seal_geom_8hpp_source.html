44template <typename V>
45std::optional<V> level_crossing(
46 const std::function<double(const V&)>& f,
47 const V& a,
48 const V& b,
49 const double level,
50 const double step_floor,
51 const bool from_below)
52{
53 const auto before = [&](const double v) { return from_below ? v < level : v > level; };
54 double v = f(a);
55 if (!before(v)) {
56 return std::nullopt;
57 }
58 const double L = (b - a).norm();
59 const V dir = (b - a) / L;
60 double t_prev = 0.;
61 double t = 0.;
62 while (before(v)) {
63 t_prev = t;
64 t += std::abs(v - level) + step_floor;
65 if (t >= L) {
66 t = L;
67 v = f(b);
68 if (before(v)) {
69 return std::nullopt;
70 }
71 break;
72 }
73 v = f(a + t * dir);
74 }
75 // f(t_prev) before the level, f(t) at it or past it
76 double lo = t_prev;
77 double hi = t;
78 for (int i = 0; i < 50; ++i) {
79 const double m = (lo + hi) / 2;
80 if (before(f(a + m * dir))) {
81 lo = m;
82 } else {
83 hi = m;
84 }
85 }
86 return V(a + ((lo + hi) / 2) * dir);
87}
