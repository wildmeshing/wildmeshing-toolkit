23template <typename T>
24DataMatrix<T> matrix(const pybind11::array& value, bool allow_vector = false)
25{
26 if (value.ndim() != 2 && !(allow_vector && value.ndim() == 1))
27 throw pybind11::value_error(
28 "Mesh arrays must have two dimensions (attributes may be vectors)");
29 const auto contiguous =
30 pybind11::array_t<T, pybind11::array::c_style | pybind11::array::forcecast>::ensure(value);
31 if (!contiguous) throw pybind11::type_error("Cannot convert mesh array");
32 DataMatrix<T> result(value.shape(0), value.ndim() == 1 ? 1 : value.shape(1));
33 if (result.size())
34 std::memcpy(result.data(), contiguous.data(), size_t(result.size()) * sizeof(T));
35 return result;
36}
