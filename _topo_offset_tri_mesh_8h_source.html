1332 std::shared_ptr<Envelope> surface_envelope_for_edge(
1333 const std::array<size_t, 2>& vids,
1334 const EdgeAttributes& /*attr*/) const override
1335 {
1336 // Boundary geometry first: a segment on any tag-region boundary may not
1337 // drift out of that boundary's tube, and one on several boundaries -- a junction -- is
1338 // held in their intersection. The mask carries the input complex too, since every complex
1339 // simplex lies on tag boundaries, and E_t is built from the input mesh before construction
1340 // touches it, so the per-tag tubes hold the as-loaded geometry.
1341 //
1342 // Keyed on the vertices, not the edge: every caller is an operation asking about a segment
1343 // it is about to create or has just created, whose own edge attributes are not written
1344 // yet. The endpoints' masks are, maintained by the operations themselves (AND at a split,
1345 // OR at a collapse).
1346 uint64_t mask = edge_mask(vids);
1347 bool all_offset = true;
1348 for (const size_t v : vids) {
1349 all_offset = all_offset && m_vertex_extra[v].m_is_on_offset;
1350 }
1351
1352 // The ambiguous case: both endpoints can be on region boundaries and on the offset front
1353 // at once -- a real state, and the point of tracking the two families separately. The
1354 // endpoint-mask AND is then necessary but not sufficient for the segment lying on a shared
1355 // boundary: two vertices on different junctions can share a tag bit by coincidence and be
1356 // joined by an offset chord nowhere near that tag's curve, which a container it was never
1357 // meant to satisfy then refuses to refine. Ask the edge's own class, the only record that
1358 // distinguishes a chord from a boundary.
1359 //
1360 // Reading the slot is safe here and would not be unconditionally: the base checks a
1361 // split's two child segments before writing their attributes, so their slots still hold a
1362 // recycled edge's class. A split child never reaches this branch -- its mask and its
1363 // offset flag are mutually exclusive by construction, so one of `mask` and `all_offset` is
1364 // always empty -- and the m_is_surface_fs guard leaves both constraints standing rather
1365 // than dropping one when a slot is illegible.
1366 if (mask != 0 && all_offset) {
1367 if (const auto found = try_tuple_from_edge(vids)) {
1368 const size_t eid = std::get<1>(*found);
1369 if (m_edge_attribute[eid].m_is_surface_fs) {
1370 if (edge_is_offset(eid)) {
1371 mask = 0; // an offset edge lies on no region boundary
1372 } else {
1373 all_offset = false; // a region edge is not the offset front
1374 }
1375 }
1376 }
1377 }
1378 // Both families compose: whatever holds this segment holds it at once. The offset side
1379 // holds only in the frozen-front final pass (see containment_for()); in the loop the
1380 // front is what the optimization moves, so it contributes nothing there.
1381 const std::shared_ptr<Envelope> base = containment_for(mask, all_offset);
1382 if (base || m_deform_tags.empty()) return base;
1383 // deform_others' ops-only tube: a released boundary is held by no mask -- its vertices
1384 // were freed so smoothing can carry the object -- which would leave the operations free
1385 // to decimate and reposition it. A segment the masks and the offset class do not claim,
1386 // but which lies on a released boundary by its incident faces' current tags, is held to
1387 // the tube around the boundary's current shape. A fresh split child can misread its
1388 // recycled face slots for this one check, and is at worst skipped or over-held once.
1389 if (const auto found = try_tuple_from_edge(vids)) {
1390 if (edge_borders_released_boundary(std::get<0>(*found))) return released_envelope();
1391 }
1392 return nullptr;
1393 }

1755 template <typename Visit>
1756 void for_each_edge_sample(const Vector2d& p0, const Vector2d& p1, Visit&& visit) const
1757 {
1758 const int k = m_offset_params.stencil_order;
1759 if (k < 0) return;
1760
1761 const auto emit = [&](const double wa, const double wb) {
1762 visit(Vector2d(wa * p0 + wb * p1), wa, wb);
1763 };
1764 // Order 0 is the two ENDS alone, each end's own placement error.
1765 if (k == 0) {
1766 emit(1., 0.);
1767 emit(0., 1.);
1768 return;
1769 }
1770 // Order k >= 1: the vertices of the chord cut into n = 2^(k-1) pieces, then the midpoint
1771 // of each piece -- 2n + 1 points: 3, 5, 9, 17, ...
1772 const int n = 1 << (k - 1);
1773 const double dn = double(n);
1774 for (int i = n; i >= 0; --i) {
1775 emit(double(i) / dn, double(n - i) / dn);
1776 }
1777 const double d2n = 2. * dn;
1778 for (int i = n - 1; i >= 0; --i) {
1779 emit((2. * i + 1.) / d2n, (2. * (n - 1 - i) + 1.) / d2n);
1780 }
1781 }
