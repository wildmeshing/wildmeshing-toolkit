21inline size_t thread_stripe()
22{
23 static std::atomic<size_t> next{0};
24 thread_local const size_t id = next.fetch_add(1, std::memory_order_relaxed);
25 return id;
26}

62template <class Mesh>
63size_t run_localized_to_convergence(
64 Mesh& m,
65 ExecutePass<Mesh>& executor,
66 OpList<typename Mesh::Tuple> ops,
67 size_t max_passes = 0)
68{
69 using Tuple = typename Mesh::Tuple;
70
71 // vertex_epoch[v] = the last round in which v was in some successful operation's
72 // modified region. Capacity is fixed within a phase (storage is preallocated), and
73 // any vertex created during the phase (splits) has an id below this capacity.
74 std::vector<uint64_t> vertex_epoch(m.vert_capacity(), 0);
75 uint64_t round = 0;
76 // A failure is recorded by its operation's rank among the registered names rather than by
77 // the name itself: a pass can fail on most of its candidates, and a std::string per entry
78 // is 32 of the 72 bytes. edit_operation_maps is read-only while the pass runs.
79 std::vector<Op> op_names;
80 op_names.reserve(executor.edit_operation_maps.size());
81 for (const auto& kv : executor.edit_operation_maps) {
82 op_names.push_back(kv.first);
83 }
84 // The failures, recorded by on_fail on whichever thread fails an operation -- in a screened
85 // pass that is millions of refusals per round, reported in parallel. Each thread appends to
86 // a stripe of its own, so the lock it takes is uncontended; one list behind one mutex was
87 // taken for every refusal. A serial pass records on one stripe, in failure order. Deques
88 // rather than vectors: most candidates of a pass can fail, and a vector growing by doubling
89 // to that size holds its old and new buffer at once.
90 struct alignas(64) Stripe
91 {
92 std::mutex mutex;
93 std::deque<std::pair<uint32_t, Tuple>> list;
94 };
95 constexpr size_t kStripes = 64;
96 const auto stripes = std::make_unique<Stripe[]>(kStripes);
97
98 auto edge_epoch = [&vertex_epoch](const Mesh& m_, const Tuple& t) -> uint64_t {
99 const size_t a = t.vid(m_);
100 const size_t b = t.switch_vertex(m_).vid(m_);
101 const uint64_t ea = a < vertex_epoch.size() ? vertex_epoch[a] : 0;
102 const uint64_t eb = b < vertex_epoch.size() ? vertex_epoch[b] : 0;
103 return std::max(ea, eb);
104 };
105
106 // Wrap the driver-provided renewal: keep its behavior (re-enqueue affected tuples
107 // within the pass) and additionally stamp those tuples' vertices with the current
108 // round so the between-pass filter can find the failures adjacent to them.
109 auto driver_renew = executor.renew_neighbor_tuples;
110 executor.renew_neighbor_tuples =
111 [&, driver_renew](const Mesh& m_, Op op, const std::vector<Tuple>& newts) {
112 auto tups = driver_renew(m_, op, newts);
113 for (const auto& [_, t] : tups) {
114 const size_t a = t.vid(m_);
115 const size_t b = t.switch_vertex(m_).vid(m_);
116 if (a < vertex_epoch.size()) {
117 // this is thread-safe because each vertex is only ever modified by one
118 // operation at a time (the two-ring lock)
119 vertex_epoch[a] = round;
120 }
121 if (b < vertex_epoch.size()) {
122 vertex_epoch[b] = round;
123 }
124 }
125 return tups;
126 };
127 executor.on_fail = [&stripes, &op_names](const Mesh&, Op op, const Tuple& t) {
128 uint32_t k = 0;
129 while (k < op_names.size() && op_names[k] != op) ++k;
130 assert(k < op_names.size());
131 Stripe& s = stripes[detail::thread_stripe() % kStripes];
132 std::lock_guard<std::mutex> lock(s.mutex);
133 s.list.emplace_back(k, t);
134 };
135
136 size_t total_success = 0;
137 std::vector<uint8_t> retry;
138 std::vector<uint32_t> list_id(op_names.size());
139 do {
140 ++round;
141 // Handed over, so the executor frees the list once it has queued it.
142 executor(m, std::move(ops));
143 total_success += static_cast<size_t>(executor.get_cnt_success());
144 ops.clear();
145 for (size_t k = 0; k < op_names.size(); ++k) list_id[k] = ops.name_id(op_names[k]);
146 for (size_t i = 0; i < kStripes; ++i) {
147 auto& failures = stripes[i].list;
148 // Retry only a failure whose neighborhood a success of this round modified. Reads
149 // only, so in parallel; queued below in failure order.
150 retry.assign(failures.size(), 0);
151 threading::dynamic_parallel_for(
152 failures.size(),
153 executor.num_threads,
154 4096,
155 [&](size_t b, size_t e) {
156 for (size_t j = b; j < e; ++j) {
157 const Tuple& t = failures[j].second;
158 retry[j] = t.is_valid(m) && edge_epoch(m, t) == round;
159 }
160 });
161 for (size_t j = 0; j < failures.size(); ++j) {
162 if (retry[j]) ops.emplace_back_id(list_id[failures[j].first], failures[j].second);
163 }
164 std::deque<std::pair<uint32_t, Tuple>>().swap(failures);
165 }
166 } while (executor.get_cnt_success() > 0 && !ops.empty() &&
167 (max_passes == 0 || round < max_passes));
168 return total_success;
169}
