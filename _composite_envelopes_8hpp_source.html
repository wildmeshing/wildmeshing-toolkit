172 const std::shared_ptr<Envelope>& member(size_t i) const { return m_members[i]; }
173
174 std::shared_ptr<Envelope> containment(const InputSet set) const
175 {
176 check(set);
177 if ((set & (set - 1)) == 0) return m_members[lowest_bit(set)];
178 // Read-mostly: a set's intersection is built once, then every check only looks it up,
179 // concurrently. An exclusive lock on every lookup serialized the operations' threads.
180 {
181 std::shared_lock<std::shared_mutex> lock(m_mutex);
182 const auto it = m_intersections.find(set);
183 if (it != m_intersections.end()) return it->second;
184 }
185 std::unique_lock<std::shared_mutex> lock(m_mutex);
186 std::shared_ptr<Envelope>& e = m_intersections[set];
187 if (!e) {
188 std::vector<std::shared_ptr<Envelope>> members;
189 for (size_t i = 0; i < m_members.size(); ++i) {
190 if (set & (InputSet(1) << i)) members.push_back(m_members[i]);
191 }
192 e = std::make_shared<IntersectionEnvelope>(std::move(members));
193 }
194 return e;
195 }

198 template <typename Point>
199 std::shared_ptr<Envelope> pull(const InputSet set, const Point& p) const
200 {
201 check(set);
202 if ((set & (set - 1)) == 0) return m_members[lowest_bit(set)];
203 std::shared_ptr<Envelope> best;
204 double worst = -1.;
205 for (size_t i = 0; i < m_members.size(); ++i) {
206 if (!(set & (InputSet(1) << i))) continue;
207 const double d2 = m_members[i]->squared_distance(p);
208 if (d2 > worst) {
209 worst = d2;
210 best = m_members[i];
211 }
212 }
213 return best;
214 }
215
216private:
217 void check(const InputSet set) const
218 {
219 if (set == 0) {
220 log_and_throw_error(
221 "a tracked simplex has an empty input set: the insertion or an operation did "
222 "not carry it");
223 }
224 if (m_members.size() < size_t(SurfaceTagAttributes::max_inputs) &&
225 (set >> m_members.size())) {
226 log_and_throw_error(
227 "input set {:#x} names an input beyond the {} that have envelopes",
228 set,
229 m_members.size());
230 }
231 }
232 static size_t lowest_bit(const InputSet set)
233 {
234 size_t i = 0;
235 while (!(set & (InputSet(1) << i))) ++i;
236 return i;
237 }
