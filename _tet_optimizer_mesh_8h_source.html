669 // ---- SWAP REJECTION INSTRUMENTATION (debug; see swap_reject_report()) ----
670 // Every place a proposed swap can be turned down, counted. Added to find out why an offset
671 // surface flip is never accepted; the counters cost one relaxed atomic increment on a path
672 // that is already refusing the operation, and nothing reads them unless a caller asks.
673 //
674 // The refusals TetMesh makes itself, below every hook (the valence re-test, an open ring, a
675 // face or edge that already exists, no allowed 4-4 / 5-6 case beating the current cells, out
676 // of slots, a duplicate tet), were once visible here only as before_pass - after_enter. They
677 // are now reported one by one through TetMesh::op_event(); see op_accounting_report(). The
678 // per-turn line of this block still prints that difference as `connectivity`.
679 //
680 // Face swaps count their own reasons (face_*, and base_before / interior_hook / base_after /
681 // after_cells on their path) in the per-kind table alone -- swap_reject_kind_only() -- so
682 // swap_reject_report(), which has only ever described edge swaps, keeps meaning what it meant.
683 enum class SwapReject : int {
684 base_before, // TetMesh::swap_*_before said no
685 base_after, // TetMesh::swap_*_after said no
686 valence, // wrong incident-tet count for this swap kind
687 bbox, // edge on the bounding box
688 surface_not_allowed, // has surface faces and allow_surface_swap() is false
689 prepare_flip, // prepare_surface_flip() refused (see the flip_* reasons)
690 interior_hook, // swap_before_interior() refused
691 flip_open_boundary, // flipping would change the surface's boundary loops
692 flip_not_two_surf, // not exactly two incident surface faces
693 flip_nonmanifold_edge, // more than two incident surface faces
694 flip_cd_nonmanifold, // a surface face already sits on the new edge (c,d)
695 flip_new_face_surface, // (a,c,d) or (b,c,d) is already tagged surface
696 flip_app_refused, // swap_before_surface() refused (see the app_* reasons)
697 flip_wrong_case, // 4-4 / 5-6: the retetrahedralization is not the one that makes
698 // the surface diagonal (c,d), so this case is not the flip
699 app_capture_label, // application: the swap ring spans two construction labels
700 app_capture_tag, // application: the swap ring spans two cell tags
701 app_fid_missing, // application: a face of the pair could not be found
702 app_class_mismatch, // application: the two faces are different tracked surfaces
703 app_mask_mismatch, // application: the two faces carry different boundary masks
704 app_sag_raised, // application: the flip would raise the local sag
705 after_inverted, // a created cell is inverted
706 after_quality, // swap_quality_allowed() said no
707 after_cells, // swap_after_cells() refused (see the app_after_* reasons)
708 app_after_side_conflict, // application: two ring vertices of a new cell sit on two sides
709 app_after_no_side, // application: a new cell holds no ring vertex to take a side from
710 after_envelope, // a new surface triangle left the envelope
711 face_tracked_surface, // face swap: the face is a tracked surface face
712 face_tracked_bbox, // face swap: the face is on the bounding box
713 face_inverted, // face swap: one of the three new cells would be inverted
714 face_not_better, // face swap: a new cell is not strictly better than the two old ones
715 COUNT
716 };

721 attempt,
722 surface_attempt,
723 before_pass,
724 after_enter,
725 accepted,
726 COUNT

756 // ---- COLLAPSE REJECTION INSTRUMENTATION (see op_accounting_report()) ----
757 // Every place a collapse is turned down in a hook, counted once, at the place that decided.
758 // Before this, 85% of the collapse attempts on the deliverable cube (target_distance_rel 1e-2
759 // / front_conv_rel 1e-4, turn 5: 213k of 232k) were refused with nothing recorded but the
760 // ops guard's 367. What TetMesh refuses below the hooks (the link condition, a duplicate tet)
761 // arrives through op_event(). Before-hook reasons come first, after-hook reasons from
762 // after_base on.
763 enum class CollapseReject : int {
764 coarsen_length, // coarsening pass only: the edge is at or above the collapse target
765 bbox, // v1 sits on a bounding-box face that v2 does not
766 surface_leaves_envelope, // v1 on the surface, v2 not, and v2 not inside the envelope
767 inverted, // a cell the collapse reshapes would invert
768 quality, // collapse_quality_allowed() said no
769 substructure_link, // preserve_topology, both ends on the surface: substructure link
770 app_both_surfaces, // application: an end on the input complex and an end on the offset
771 app_front_unlimited, // application: an unlimited-length pass may not remove a front vertex
772 app_leaves_input, // application: v1 on the input complex, v2 not
773 app_leaves_offset, // application: v1 on the offset surface, v2 not
774 app_leaves_region, // application: v1 on a region boundary, v2 not
775 app_order2, // application: an order-2 vertex would merge into a lower-order one
776 app_substructure_link, // application: the substructure link condition, unconditional
777 app_ops_guard, // application: the ops divergence guard
778 after_base, // TetMesh::collapse_edge_after said no
779 after_connectivity, // collapse_after_connectivity() said no
780 after_envelope, // a surface face the survivor keeps left the envelope
781 after_face_attribute, // a face attribute found no face left to land on
782 after_coarsen_region, // coarsening only: the region's worst quality rose
783 COUNT
784 };

793 // ---- OPERATION ACCOUNTING (counting only) ----
794 // TetMesh reports every operation's attempt and its one outcome through op_event(); this
795 // counts them per kind. op_accounting_report() then checks, per kind, that the outcomes add
796 // up to the attempts, and that the before- and after-hook refusals add up to the reasons the
797 // hooks counted (SwapReject / CollapseReject). A nonzero remainder is a refusal nobody named.

800 if (e == OpEvent::attempt) current_op_kind() = k;
801 m_op_events[size_t(k)][size_t(e)].fetch_add(1, std::memory_order_relaxed);
802 }

808 static thread_local OpKind k = OpKind::COUNT;
