142 // The open boundaries are held by the base: tagged at init (tag_order2_edges()), every
143 // tagged edge a collapse keeps, or a smoothing move touches, must stay in
144 // m_order2_edge_envelope, no swap removes one, and a collapse that leaves one with other

146 // per-vertex flags, their tube m_order2_envelope (the same tube as the base's), the
147 // collapse's point test (a flagged vertex collapses only onto a flagged vertex or into the
148 // tube, implied by the base's segment test whenever the vertex has a tagged edge) and
149 // is_open_boundary_edge(), which classified edges by the flags and the tube.

172 // only used with unit tests
173 void create_mesh_attributes(
174 const std::vector<VertexAttributes>& _vertex_attribute,
175 const std::vector<TetAttributes>& _tet_attribute)
176 {
177 const size_t n_tet = _tet_attribute.size();
178 m_vertex_attribute.resize(_vertex_attribute.size());
179 m_feature_point_attribute.resize(_vertex_attribute.size());
180 m_face_attribute.resize(4 * n_tet);
181 m_tet_attribute.resize(n_tet);
182
183 // new for edge
184 // m_edge_attribute.resize(6 * n_tet);
185
186 for (size_t i = 0; i < _vertex_attribute.size(); i++)
187 m_vertex_attribute[i] = _vertex_attribute[i];
188
189 // Keep whatever init() reserved. AttributeCollection::resize only ever grows, so the
190 // calls above cannot shrink a collection -- but assigning m_attributes directly does,
191 // and it used to drop the tet attributes to exactly n_tet. The attributes have to stay
192 // at least as large as the connectivity: an operation that creates a new element
193 // indexes them by its id, and a 5->6 swap creates one. That left m_tet_attribute one
194 // short of the tet the swap adds, and AttributeCollection::operator[] read past the end
195 // of it -- ASan reports a heap-buffer-overflow reached from swap_edge_56_after, and
196 // libstdc++ then aborts on the corrupted heap a few allocations later, while libc++
197 // carries on and the tests pass.
198 const size_t tcap = std::max(n_tet, m_tet_attribute.size());
199 m_tet_attribute.m_attributes = std::vector<TetAttributes>(tcap);
200 for (size_t i = 0; i < n_tet; i++) m_tet_attribute[i] = _tet_attribute[i];
201 for (size_t i = 0; i < n_tet; i++)
202 m_tet_attribute[i].m_quality = get_quality(tuple_from_tet(i));
203 }
