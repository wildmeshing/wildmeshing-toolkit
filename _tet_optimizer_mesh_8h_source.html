810 // ---- SWAP REJECTION INSTRUMENTATION (debug; see swap_reject_report()) ----
811 // Every place a proposed swap can be turned down, counted. Added to find out why an offset
812 // surface flip is never accepted; the counters cost one relaxed atomic increment on a path
813 // that is already refusing the operation, and nothing reads them unless a caller asks.
814 //
815 // The refusals TetMesh makes itself, below every hook (the valence re-test, an open ring, a
816 // face or edge that already exists, no allowed 4-4 / 5-6 case beating the current cells, out
817 // of slots, a duplicate tet), were once visible here only as before_pass - after_enter. They
818 // are now reported one by one through TetMesh::op_event(); see op_accounting_report(). The
819 // per-turn line of this block still prints that difference as `connectivity`.
820 //
821 // Face swaps count their own reasons (face_*, and base_before / interior_hook / base_after /
822 // after_cells on their path) in the per-kind table alone -- swap_reject_kind_only() -- so
823 // swap_reject_report(), which has only ever described edge swaps, keeps meaning what it meant.
824 enum class SwapReject : int {
825 base_before, // TetMesh::swap_*_before said no
826 base_after, // TetMesh::swap_*_after said no
827 valence, // wrong incident-tet count for this swap kind
828 bbox, // edge on the bounding box
829 surface_not_allowed, // has surface faces and allow_surface_swap() is false
830 feature_edge, // the swap would delete a tagged feature edge (it is the operated edge)
831 prepare_flip, // prepare_surface_flip() refused (see the flip_* reasons)
832 interior_hook, // swap_before_interior() refused
833 flip_open_boundary, // flipping would change the surface's boundary loops
834 flip_not_two_surf, // not exactly two incident surface faces
835 flip_nonmanifold_edge, // more than two incident surface faces
836 flip_cd_nonmanifold, // a surface face already sits on the new edge (c,d)
837 flip_new_face_surface, // (a,c,d) or (b,c,d) is already tagged surface
838 flip_app_refused, // swap_before_surface() refused (see the app_* reasons)
839 flip_wrong_case, // 4-4 / 5-6: the retetrahedralization is not the one that makes
840 // the surface diagonal (c,d), so this case is not the flip
841 app_capture_label, // application: the swap ring spans two construction labels
842 app_capture_tag, // application: the swap ring spans two cell tags
843 app_fid_missing, // application: a face of the pair could not be found
844 app_class_mismatch, // application: the two faces are different tracked surfaces
845 app_mask_mismatch, // application: the two faces carry different boundary masks
846 app_sag_raised, // application: the flip would raise the local sag
847 after_inverted, // a created cell is inverted
848 after_quality, // swap_quality_allowed() said no
849 after_cells, // swap_after_cells() refused (see the app_after_* reasons)
850 app_after_side_conflict, // application: two ring vertices of a new cell sit on two sides
851 app_after_no_side, // application: a new cell holds no ring vertex to take a side from
852 after_envelope, // a new surface triangle left the envelope
853 face_tracked_surface, // face swap: the face is a tracked surface face
854 face_tracked_bbox, // face swap: the face is on the bounding box
855 face_inverted, // face swap: one of the three new cells would be inverted
856 face_not_better, // face swap: a new cell is not strictly better than the two old ones
857 COUNT
858 };

903 // ---- COLLAPSE REJECTION INSTRUMENTATION (see op_accounting_report()) ----
904 // Every place a collapse is turned down in a hook, counted once, at the place that decided.
905 // Before this, 85% of the collapse attempts on the deliverable cube (target_distance_rel 1e-2
906 // / front_conv_rel 1e-4, turn 5: 213k of 232k) were refused with nothing recorded but the
907 // ops guard's 367. What TetMesh refuses below the hooks (the link condition, a duplicate tet)
908 // arrives through op_event(). Before-hook reasons come first, after-hook reasons from
909 // after_base on.
910 enum class CollapseReject : int {
911 coarsen_length, // coarsening pass only: the edge is at or above the collapse target
912 bbox, // v1 sits on a bounding-box face that v2 does not
913 surface_leaves_envelope, // v1 on the surface, v2 not, and v2 not inside the envelope
914 inverted, // a cell the collapse reshapes would invert
915 quality, // collapse_quality_allowed() said no
916 substructure_link, // preserve_topology, both ends on the surface: substructure link
917 feature_leaves_tube, // a tagged feature edge the collapse moves would leave its tube
918 app_both_surfaces, // application: an end on the input complex and an end on the offset
919 app_front_unlimited, // application: an unlimited-length pass may not remove a front vertex
920 app_leaves_input, // application: v1 on the input complex, v2 not
921 app_leaves_offset, // application: v1 on the offset surface, v2 not
922 app_leaves_region, // application: v1 on a region boundary, v2 not
923 app_order2, // application: an order-2 vertex would merge into a lower-order one
924 app_substructure_link, // application: the substructure link condition, unconditional
925 app_ops_guard, // application: the ops divergence guard
926 after_base, // TetMesh::collapse_edge_after said no
927 after_connectivity, // collapse_after_connectivity() said no
928 after_envelope, // a surface face the survivor keeps left the envelope
929 after_face_attribute, // a face attribute found no face left to land on
930 after_order2_chord, // an order-2 edge the survivor keeps is no longer order-2 substructure
931 after_feature_edge, // a tagged feature edge found no edge left to land on
932 after_coarsen_region, // coarsening only: the region's worst quality rose
933 COUNT
934 };

943 // ---- OPERATION ACCOUNTING (counting only) ----
944 // TetMesh reports every operation's attempt and its one outcome through op_event(); this
945 // counts them per kind. op_accounting_report() then checks, per kind, that the outcomes add
946 // up to the attempts, and that the before- and after-hook refusals add up to the reasons the
947 // hooks counted (SwapReject / CollapseReject). A nonzero remainder is a refusal nobody named.
948 void op_event(OpKind k, OpEvent e) const override
949 {
950 if (e == OpEvent::attempt) current_op_kind() = k;
951 m_op_events[size_t(k)][size_t(e)].fetch_add(1, std::memory_order_relaxed);
952 // The edge swaps whose hooks count SwapStage; see swap_reject_report().
953 if (e == OpEvent::screened &&
954 (k == OpKind::swap_32 || k == OpKind::swap_44 || k == OpKind::swap_56)) {
955 swap_stage(SwapStage::screened);
956 }
957 }

963 static thread_local OpKind k = OpKind::COUNT;

1075 // ---- Feature-edge tag propagation (m_track_feature_edges) ----------------------------
1076 //
1077 // An edge attribute's slot is (lowest incident tet id) * 6 + local index, so it moves
1078 // whenever that tet dies. Any edge whose slot can move is an edge of an affected tet
1079 // (deleted tets are affected; created tets only affect their own six edges), so caching
1080 // every edge of every affected tet before the operation and rewriting afterwards covers
1081 // every slot that can move -- the same argument the face tags rely on.
1082 //
1083 // Restore comes in two forms because the operations differ in what they create:
1084 // * split and the swaps CREATE tets, whose slots may be reused and hold stale values;
1085 // restore_region walks every edge of every created tet and writes the cached value or
1086 // the default. The created region provably contains every surviving cached edge.
1087 // * collapse creates NO tets, and its link edges (la,lb) -- which contain neither
1088 // endpoint, a case a tet FACE cannot exhibit -- can live on in tets outside any
1089 // enumerable region. restore_remap walks the CACHE instead: remaps v1 -> v2, OR-merges
1090 // colliding pairs, and writes each surviving edge wherever its slot now is. Returns
1091 // false if a TAGGED cached edge no longer exists (the caller aborts, mirroring the
1092 // face path's try_tuple_from_face bailout).
