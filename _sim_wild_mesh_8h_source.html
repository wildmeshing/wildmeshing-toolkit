127 void create_mesh_attributes(
128 const std::vector<VertexAttributes>& _vertex_attribute,
129 const std::vector<TetAttributes>& _tet_attribute)
130 {
131 const size_t n_tet = _tet_attribute.size();
132 m_vertex_attribute.resize(_vertex_attribute.size());
133 m_face_attribute.resize(4 * n_tet);
134 m_tet_attribute.resize(n_tet);
135
136 for (size_t i = 0; i < _vertex_attribute.size(); i++)
137 m_vertex_attribute[i] = _vertex_attribute[i];
138
139 // Keep whatever init() reserved. AttributeCollection::resize only ever grows, so the
140 // calls above cannot shrink a collection -- but assigning m_attributes directly does,
141 // and it used to drop the tet attributes to exactly n_tet. The attributes have to stay
142 // at least as large as the connectivity: an operation that creates a new element
143 // indexes them by its id, and a 5->6 swap creates one. tetwild hit the same thing (see
144 // the note on its copy); there the overrun wrote a double past the end and libc++ let
145 // it pass, here it assigns a std::set and segfaults outright.
146 const size_t tcap = std::max(n_tet, m_tet_attribute.size());
147 m_tet_attribute.m_attributes = std::vector<TetAttributes>(tcap);
148 for (size_t i = 0; i < n_tet; i++) m_tet_attribute[i] = _tet_attribute[i];
149 for (size_t i = 0; i < n_tet; i++)
150 m_tet_attribute[i].m_quality = get_quality(tuple_from_tet(i));
151 }
