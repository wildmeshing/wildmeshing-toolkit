20inline Json
21background_diagnosis(const Mesh& m, const std::vector<uint8_t>& movable, double threshold)
22{
23 size_t bad = 0, blocked = 0;
24 double maximum = 0;
25 std::map<size_t, size_t> histogram;
26 Json worst;
27 for (size_t i = 0; i < m.cells.size(); ++i) {
28 const auto& c = m.cells[i];
29 if (c.region != 3 || (c.type != 4 && c.type != 7)) continue;
30 size_t free = 0;
31 for (size_t v : c.v) free += movable[v];
32 if (c.energy > threshold) {
33 ++bad;
34 ++histogram[free];
35 blocked += free == 0;
36 }
37 if (c.energy > maximum) {
38 maximum = c.energy;
39 Json vertices = Json::array();
40 for (size_t v : c.v)
41 vertices.push_back(
42 {{"row", v},
43 {"movable", bool(movable[v])},
44 {"shell", bool(m.shell[v])},
45 {"input", bool(m.input[v])},
46 {"outer", bool(m.outer[v])},
47 {"rod_reserved", m.fixed(v)}});
48 worst = {
49 {"cell_row", i},
50 {"type", c.type},
51 {"AMIPS", c.energy},
52 {"vertices", vertices}};
53 }
54 }
55 Json counts = Json::object();
56 for (auto [k, v] : histogram) counts[std::to_string(k)] = v;
57 return {
58 {"bad_cells", bad},
59 {"maximum_AMIPS", maximum},
60 {"bad_cells_with_no_movable_vertices", blocked},
61 {"bad_cell_movable_vertex_histogram", counts},
62 {"worst", worst}};
63}

66inline bool block_step(
67 native::Mesh& m,
68 size_t ci,
69 const std::vector<uint8_t>& movable,
70 double length,
71 Json& diagnosis,
72 bool expand,
73 double target)
74{
75 std::vector<size_t> group;
76 std::set<size_t> patch;
77 for (size_t v : m.cells[ci].v)
78 if (movable[v]) {
79 group.push_back(v);
80 patch.insert(m.star[v].begin(), m.star[v].end());
81 }
82 if (group.empty()) return false;
83 if (expand) {
84 const auto seeds = group;
85 std::set<size_t> added(group.begin(), group.end());
86 std::vector<std::pair<double, size_t>> neighbors;
87 for (size_t v : seeds)
88 for (size_t c : m.star[v])
89 for (size_t u : m.cells[c].v)
90 if (movable[u] && added.insert(u).second) {
91 double priority = 0;
92 for (size_t j : m.star[u]) priority = std::max(priority, m.cells[j].energy);
93 neighbors.push_back({priority, u});
94 }
95 std::sort(neighbors.begin(), neighbors.end(), [](auto a, auto b) {
96 return a.first != b.first ? a.first > b.first : a.second < b.second;
97 });
98 for (auto [priority, u] : neighbors)
99 if (group.size() < 48) group.push_back(u);
100 patch.clear();
101 for (size_t v : group) patch.insert(m.star[v].begin(), m.star[v].end());
102 }
103 std::map<size_t, size_t> index;
104 for (size_t j = 0; j < group.size(); ++j) index[group[j]] = j;
105 std::vector<native::V> gradient(group.size(), native::V::Zero()), old;
106 long double before = 0;
107 for (size_t j : patch) {
108 double derivative;
109 before += mixed_loss(m.cells[j].energy, target, &derivative);
110 auto g = cell_gradient(m, m.cells[j]);
111 for (size_t k = 0; k < m.cells[j].v.size(); ++k)
112 if (auto it = index.find(m.cells[j].v[k]); it != index.end())
113 gradient[it->second] += derivative * g[k];
114 }
115 double scale = 0;
116 for (auto g : gradient) {
117 if (!g.allFinite()) {
118 diagnosis["nonfinite_gradient"] = true;
119 return false;
120 }
121 scale = std::max(scale, g.cwiseAbs().maxCoeff());
122 }
123 if (!(scale > 0)) return false;
124 for (auto& g : gradient) g /= scale;
125 const auto objective_gradient = gradient;
126 const Eigen::Index dimension = 3 * group.size();
127 Eigen::VectorXd direction(dimension);
128 for (size_t j = 0; j < group.size(); ++j) direction.segment<3>(3 * j) = -gradient[j];
129 std::vector<Eigen::VectorXd> constraints;
130 for (size_t j : patch) {
131 const auto& c = m.cells[j];
132 if (c.type != 4 || c.energy < 1e6) continue;
133 native::V a = m.p[c.v[1]] - m.p[c.v[0]], b = m.p[c.v[2]] - m.p[c.v[0]],
134 d = m.p[c.v[3]] - m.p[c.v[0]];
135 std::array<native::V, 4> dg;
136 dg[1] = b.cross(d);
137 dg[2] = d.cross(a);
138 dg[3] = a.cross(b);
139 dg[0] = -dg[1] - dg[2] - dg[3];
140 Eigen::VectorXd normal = Eigen::VectorXd::Zero(dimension);
141 for (size_t k = 0; k < 4; ++k)
142 if (auto it = index.find(c.v[k]); it != index.end())
143 normal.segment<3>(3 * it->second) = dg[k];
144 if (normal.norm() > 0) constraints.push_back(normal.normalized());
145 }
146 std::vector<double> correction(constraints.size());
147 for (int iteration = 0; iteration < 300; ++iteration) {
148 for (size_t j = 0; j < constraints.size(); ++j) {
149 direction += correction[j] * constraints[j];
150 const double amount = std::max(0., -direction.dot(constraints[j]));
151 direction += amount * constraints[j];
152 correction[j] = -amount;
153 }
154 }
155 double derivative = 0;
156 for (size_t j = 0; j < group.size(); ++j)
157 derivative += objective_gradient[j].dot(direction.segment<3>(3 * j));
158 if (!(direction.norm() > 1e-15) || !(derivative < 0)) {
159 diagnosis = {
160 {"cone_constraints", constraints.size()},
161 {"projected_norm", direction.norm()},
162 {"derivative", derivative}};
163 return false;
164 }
165 direction *= .2 * length / direction.norm();
166 for (size_t j = 0; j < group.size(); ++j) gradient[j] = direction.segment<3>(3 * j);
167 for (size_t v : group) old.push_back(m.p[v]);
168 size_t invalid = 0, no_descent = 0, not_lower = 0;
169 for (int bt = 0; bt < 48; ++bt) {
170 long double slope = 0;
171 for (size_t j = 0; j < group.size(); ++j) {
172 m.p[group[j]] = old[j] + std::ldexp(1., -bt) * gradient[j];
173 slope += (m.p[group[j]] - old[j]).dot(objective_gradient[j]);
174 }
175 if (!(slope < 0)) {
176 ++no_descent;
177 continue;
178 }
179 bool valid = true;
180 long double after = 0;
181 std::vector<native::Cell> proposed;
182 for (size_t j : patch) {
183 auto c = m.cells[j];
184 if (!std::isfinite(native::quality(m.p, c))) {
185 valid = false;
186 break;
187 }
188 after += mixed_loss(c.energy, target);
189 proposed.push_back(std::move(c));
190 }
191 if (!valid)
192 ++invalid;
193 else if (after < before) {
194 size_t k = 0;
195 for (size_t j : patch) m.cells[j] = std::move(proposed[k++]);
196 diagnosis = {
197 {"backtracks", bt},
198 {"free_vertices", group.size()},
199 {"before", double(before)},
200 {"after", double(after)}};
201 return true;
202 } else
203 ++not_lower;
204 }
205 for (size_t j = 0; j < group.size(); ++j) m.p[group[j]] = old[j];
206 diagnosis = {
207 {"free_vertices", group.size()},
208 {"invalid_trials", invalid},
209 {"unrepresentable_or_no_descent", no_descent},
210 {"nondecreasing_trials", not_lower}};
211 return false;
212}

215inline uint64_t connectivity_signature(const Mesh& m)
216{
217 uint64_t value = 1469598103934665603ULL;
218 const auto mix = [&](uint64_t x) { value = (value ^ x) * 1099511628211ULL; };
219 for (const auto& c : m.cells) {
220 mix(c.type);
221 mix(c.region);
222 mix(c.v.size());
223 for (size_t v : c.v) mix(v);
224 }
225 return value;
226}

272 if (maximum < overall_maximum || (maximum == overall_maximum && bad < overall_bad)) {
273 overall_maximum = maximum;
274 overall_bad = bad;
275 overall_positions = m.p;
276 selected_phase = name;
277 }

284 if (c.region == 3 && (c.type == 4 || c.type == 7) && c.energy > options.threshold)
285 for (size_t v : c.v)
286 if (movable[v]) priority[v] = std::max(priority[v], c.energy);
287 std::vector<size_t> active;
288 for (size_t v = 0; v < priority.size(); ++v)
289 if (priority[v] > 0) active.push_back(v);
290 std::sort(active.begin(), active.end(), [&](size_t a, size_t b) {
291 return priority[a] != priority[b] ? priority[a] > priority[b] : a < b;
292 });
293 size_t accepted = 0;
294 for (size_t v : active) {
295 if (deadline.expired()) break;
296 accepted +=
297 mixed_vertex_step(m, rules, reference, freedom, v, target, thickness_weight);
298 }
299 vertex_total += accepted;
300 const auto state = background_diagnosis(m, movable, options.threshold);
301 history.push_back({{"sweep", pass + 1}, {"accepted", accepted}, {"background", state}});
302 if (!accepted || state["bad_cells"] == 0) break;
303 }
304 auto state = background_diagnosis(m, movable, options.threshold);
305 phases.push_back(
306 {{"method", "vertex_gradient"},
307 {"accepted", vertex_total},
308 {"history", std::move(history)},
309 {"final", state}});
310 total += vertex_total;
311 keep_phase("vertex_gradient", state);
312 const auto joint_phase = [&](const char* name, int sweeps, bool expand, bool keep_best) {
313 Json history = Json::array();
314 size_t accepted_total = 0;
315 auto best_positions = m.p;
316 double best_maximum = background_diagnosis(m, movable, options.threshold)["maximum_AMIPS"];
317 int selected_pass = 0;
318 for (int pass = 0; pass < sweeps && !deadline.expired(); ++pass) {
319 std::vector<size_t> bad;
320 for (size_t ci = 0; ci < m.cells.size(); ++ci) {
321 const auto& c = m.cells[ci];
322 if (c.region == 3 && (c.type == 4 || c.type == 7) && c.energy > options.threshold)
323 bad.push_back(ci);
324 }
325 if (bad.empty()) break;
326 std::sort(bad.begin(), bad.end(), [&](size_t a, size_t b) {
327 return m.cells[a].energy > m.cells[b].energy;
328 });
329 size_t accepted = 0;
330 for (size_t ci : bad) {
331 if (deadline.expired()) break;
332 Json detail;
333 accepted +=
334 block_step(m, ci, movable, reference.base_length, detail, expand, target);
335 }
336 accepted_total += accepted;
337 const auto state = background_diagnosis(m, movable, options.threshold);
338 if (state["maximum_AMIPS"].get<double>() < best_maximum) {
339 best_maximum = state["maximum_AMIPS"];

341 selected_pass = pass + 1;
342 }
343 history.push_back({{"sweep", pass + 1}, {"accepted", accepted}, {"background", state}});
344 if (!accepted || state["bad_cells"] == 0) break;
345 }
346 if (keep_best) {
347 m.p = std::move(best_positions);
348 for (auto& c : m.cells) native::quality(m.p, c);
349 }
350 const auto state = background_diagnosis(m, movable, options.threshold);
351 phases.push_back(
352 {{"method", name},
353 {"accepted", accepted_total},
354 {"selected_pass", selected_pass},
355 {"history", std::move(history)},
356 {"final", state}});
357 total += accepted_total;
358 keep_phase(name, state);
359 };
360 joint_phase("volume_feasible_joint_gradient", options.joint_sweeps, false, false);
361 joint_phase("one_ring_joint_gradient", options.neighborhood_sweeps, true, true);
362 m.p = std::move(overall_positions);
363 for (size_t v = 0; v < m.p.size(); ++v)
364 if (!movable[v] && m.p[v] != frozen_positions[v])
365 throw std::runtime_error("background_relocation_moved_fixed_vertex");
366 if (connectivity_signature(m) != signature)
367 throw std::runtime_error("background_relocation_changed_connectivity");
368 for (auto& c : m.cells)
369 if (!std::isfinite(native::quality(m.p, c)))
370 throw std::runtime_error("background_relocation_inverted_cell");
371 if (topology(m, faces_of(m.cells)) != topology_before)
372 throw std::runtime_error("background_relocation_changed_topology");
373 if (std::abs(native::volume(m) - volume0) > 1e-9 * std::abs(volume0))
374 throw std::runtime_error("background_relocation_changed_volume");
375 const auto final = background_diagnosis(m, movable, options.threshold);
376 logger().info(
377 "Background quality relocation: {} -> {} bad native cells; maximum {} -> {}",
378 initial["bad_cells"].get<size_t>(),
379 final["bad_cells"].get<size_t>(),
380 initial["maximum_AMIPS"].get<double>(),
381 final["maximum_AMIPS"].get<double>());
382 return {
383 {"status",
384 final["bad_cells"] == 0 ? "complete_below_threshold" : "complete_with_residuals"},
385 {"configuration",
386 {{"vertex_sweeps", options.vertex_sweeps},
387 {"joint_sweeps", options.joint_sweeps},
388 {"neighborhood_sweeps", options.neighborhood_sweeps},
389 {"threshold", options.threshold},
390 {"time_limit", options.time_limit}}},
391 {"initial", initial},
392 {"final", final},
393 {"before_quality", before_quality},
394 {"after_quality", quality(m)},
395 {"accepted", total},
396 {"selected_phase", selected_phase},
397 {"phases", std::move(phases)},
398 {"timed_out", deadline.expired()},
399 {"seconds", deadline.seconds()},
400 {"shell_input_exterior_and_reserved_vertices_fixed", true},
401 {"connectivity_preserved", true}};
402}
