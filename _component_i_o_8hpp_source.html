98 void raw_triangles(const std::string& name, MatrixXd& V, MatrixXi& F) const
99 {
100 const auto& data = mesh(name);

102 for (const auto& [type, c] : data.cells) {

126 // Like read_triangle_mesh(paths): concatenate the raw inputs, then clean them once, so
127 // vertex merging sees the same vertices (isolated ones included) in the same order.
128 // `read`, as there: the inputs as given, concatenated, and where their vertices went.
129 void triangles(
130 const std::vector<std::string>& names,
131 MatrixXd& V,
132 MatrixXi& F,
133 double rel,
134 double abs,
135 std::vector<int>* provenance = nullptr,
136 ReadProvenance* read = nullptr) const
137 {
138 if (!memory()) {
139 read_triangle_mesh(names, V, F, rel, abs, provenance, read);
140 return;
141 }
142 V.resize(0, 3);
143 F.resize(0, 3);
144 if (provenance) provenance->clear();
145 for (size_t i = 0; i < names.size(); ++i) {
146 MatrixXd v;
147 MatrixXi f;
148 raw_triangles(names[i], v, f);
149 const auto nv = V.rows(), nf = F.rows();
150 f.array() += nv;
151 V.conservativeResize(nv + v.rows(), 3);
152 V.bottomRows(v.rows()) = v;
153 F.conservativeResize(nf + f.rows(), 3);
154 F.bottomRows(f.rows()) = f;
155 if (provenance) provenance->resize(F.rows(), int(i));
156 }
157 if (read) {
158 read->V = V;
159 read->F = F;
160 }
161 clean_triangle_mesh(V, F, rel, abs, provenance, read ? &read->vertex_map : nullptr);
162 if (!F.rows()) throw std::runtime_error("No faces after cleaning");
163 }

204template <typename Mesh, typename Position>
205MeshData triangle_data(const Mesh& m, Position position)
206{
207 MeshData data;
208 const auto vs = m.get_vertices();
209 const auto fs = m.get_faces();
210 data.V.resize(vs.size(), 3);
211 std::map<size_t, int64_t> rows;
212 for (size_t i = 0; i < vs.size(); ++i) {
213 rows[vs[i].vid(m)] = i;
214 data.V.row(i) = position(vs[i].vid(m));
215 }
216 auto& c = data.cells[CellType::Triangle].connectivity;
217 c.resize(fs.size(), 3);
218 for (size_t i = 0; i < fs.size(); ++i) {
219 auto v = m.oriented_tri_vertices(fs[i]);
220 for (int j = 0; j < 3; ++j) c(i, j) = rows.at(v[j].vid(m));
221 }
222 data.validate();
223 return data;
224}
