585 // ---- SWAP REJECTION INSTRUMENTATION (debug; see swap_reject_report()) ----
586 // Every place a proposed swap can be turned down, counted. Added to find out why an offset
587 // surface flip is never accepted; the counters cost one relaxed atomic increment on a path
588 // that is already refusing the operation, and nothing reads them unless a caller asks.
589 //
590 // The refusals TetMesh makes itself, below every hook (the valence re-test, an open ring, a
591 // face or edge that already exists, no allowed 4-4 / 5-6 case beating the current cells, out
592 // of slots, a duplicate tet), were once visible here only as before_pass - after_enter. They
593 // are now reported one by one through TetMesh::op_event(); see op_accounting_report(). The
594 // per-turn line of this block still prints that difference as `connectivity`.
595 //
596 // Face swaps count their own reasons (face_*, and base_before / interior_hook / base_after /
597 // after_cells on their path) in the per-kind table alone -- swap_reject_kind_only() -- so
598 // swap_reject_report(), which has only ever described edge swaps, keeps meaning what it meant.
599 enum class SwapReject : int {
600 base_before, // TetMesh::swap_*_before said no
601 base_after, // TetMesh::swap_*_after said no
602 valence, // wrong incident-tet count for this swap kind
603 bbox, // edge on the bounding box
604 surface_not_allowed, // has surface faces and allow_surface_swap() is false
605 prepare_flip, // prepare_surface_flip() refused (see the flip_* reasons)
606 interior_hook, // swap_before_interior() refused
607 flip_open_boundary, // flipping would change the surface's boundary loops
608 flip_not_two_surf, // not exactly two incident surface faces
609 flip_nonmanifold_edge, // more than two incident surface faces
610 flip_cd_nonmanifold, // a surface face already sits on the new edge (c,d)
611 flip_new_face_surface, // (a,c,d) or (b,c,d) is already tagged surface
612 flip_app_refused, // swap_before_surface() refused (see the app_* reasons)
613 flip_wrong_case, // 4-4 / 5-6: the retetrahedralization is not the one that makes
614 // the surface diagonal (c,d), so this case is not the flip
615 app_capture_label, // application: the swap ring spans two construction labels
616 app_capture_tag, // application: the swap ring spans two cell tags
617 app_fid_missing, // application: a face of the pair could not be found
618 app_class_mismatch, // application: the two faces are different tracked surfaces
619 app_mask_mismatch, // application: the two faces carry different boundary masks
620 app_sag_raised, // application: the flip would raise the local sag
621 after_inverted, // a created cell is inverted
622 after_quality, // swap_quality_allowed() said no
623 after_cells, // swap_after_cells() refused (see the app_after_* reasons)
624 app_after_side_conflict, // application: two ring vertices of a new cell sit on two sides
625 app_after_no_side, // application: a new cell holds no ring vertex to take a side from
626 after_envelope, // a new surface triangle left the envelope
627 face_tracked_surface, // face swap: the face is a tracked surface face
628 face_tracked_bbox, // face swap: the face is on the bounding box
629 face_inverted, // face swap: one of the three new cells would be inverted
630 face_not_better, // face swap: a new cell is not strictly better than the two old ones
631 COUNT
632 };

637 attempt,
638 surface_attempt,
639 before_pass,
640 after_enter,
641 accepted,
642 COUNT
643 };

672 // ---- COLLAPSE REJECTION INSTRUMENTATION (see op_accounting_report()) ----
673 // Every place a collapse is turned down in a hook, counted once, at the place that decided.
674 // Before this, 85% of the collapse attempts on the deliverable cube (target_distance_rel 1e-2
675 // / front_conv_rel 1e-4, turn 5: 213k of 232k) were refused with nothing recorded but the
676 // ops guard's 367. What TetMesh refuses below the hooks (the link condition, a duplicate tet)
677 // arrives through op_event(). Before-hook reasons come first, after-hook reasons from
678 // after_base on.
679 enum class CollapseReject : int {
680 coarsen_length, // coarsening pass only: the edge is at or above the collapse target
681 bbox, // v1 sits on a bounding-box face that v2 does not
682 surface_leaves_envelope, // v1 on the surface, v2 not, and v2 not inside the envelope
683 inverted, // a cell the collapse reshapes would invert
684 quality, // collapse_quality_allowed() said no
685 substructure_link, // preserve_topology, both ends on the surface: substructure link
686 app_both_surfaces, // application: an end on the input complex and an end on the offset
687 app_front_unlimited, // application: an unlimited-length pass may not remove a front vertex
688 app_leaves_input, // application: v1 on the input complex, v2 not
689 app_leaves_offset, // application: v1 on the offset surface, v2 not
690 app_leaves_region, // application: v1 on a region boundary, v2 not
691 app_order2, // application: an order-2 vertex would merge into a lower-order one
692 app_substructure_link, // application: the substructure link condition, unconditional
693 app_ops_guard, // application: the ops divergence guard
694 after_base, // TetMesh::collapse_edge_after said no
695 after_connectivity, // collapse_after_connectivity() said no
696 after_envelope, // a surface face the survivor keeps left the envelope
697 after_face_attribute, // a face attribute found no face left to land on
698 after_coarsen_region, // coarsening only: the region's worst quality rose
699 COUNT
700 };

709 // ---- OPERATION ACCOUNTING (counting only) ----
710 // TetMesh reports every operation's attempt and its one outcome through op_event(); this
