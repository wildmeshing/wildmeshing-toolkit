61 void raw_triangles(const std::string& name, MatrixXd& V, MatrixXi& F) const
62 {
63 const auto& data = mesh(name);
64 attributes(data.vertex_attributes, reader_vertex_attributes);
65 for (const auto& [type, c] : data.cells) {
66 if (type != CellType::Triangle)
67 throw std::runtime_error("Expected triangle input: " + name);
68 attributes(c.attributes, reader_corner_attributes);
69 }
70 V = data.V;
71 F = indices(data.cell(CellType::Triangle).connectivity);
72 if (!F.rows()) throw std::runtime_error("Empty triangle input: " + name);
73 }

89 // Like read_triangle_mesh(paths): concatenate the raw inputs, then clean them once, so
90 // vertex merging sees the same vertices (isolated ones included) in the same order.
91 void triangles(
92 const std::vector<std::string>& names,
93 MatrixXd& V,
94 MatrixXi& F,
95 double rel,
96 double abs,
97 std::vector<int>* provenance = nullptr) const
98 {
99 if (!memory()) {
100 read_triangle_mesh(names, V, F, rel, abs, provenance);
101 return;
102 }
103 V.resize(0, 3);
104 F.resize(0, 3);
105 if (provenance) provenance->clear();
106 for (size_t i = 0; i < names.size(); ++i) {
107 MatrixXd v;
108 MatrixXi f;
109 raw_triangles(names[i], v, f);
110 const auto nv = V.rows(), nf = F.rows();
111 f.array() += nv;
112 V.conservativeResize(nv + v.rows(), 3);
113 V.bottomRows(v.rows()) = v;
114 F.conservativeResize(nf + f.rows(), 3);
115 F.bottomRows(f.rows()) = f;
116 if (provenance) provenance->resize(F.rows(), int(i));
117 }
118 clean_triangle_mesh(V, F, rel, abs, provenance);
119 if (!F.rows()) throw std::runtime_error("No faces after cleaning");
120 }

161template <typename Mesh, typename Position>
162MeshData triangle_data(const Mesh& m, Position position)
163{
164 MeshData data;
165 const auto vs = m.get_vertices();
166 const auto fs = m.get_faces();
167 data.V.resize(vs.size(), 3);
168 std::map<size_t, int64_t> rows;
169 for (size_t i = 0; i < vs.size(); ++i) {
170 rows[vs[i].vid(m)] = i;
171 data.V.row(i) = position(vs[i].vid(m));
172 }
173 auto& c = data.cells[CellType::Triangle].connectivity;
174 c.resize(fs.size(), 3);
175 for (size_t i = 0; i < fs.size(); ++i) {
176 auto v = m.oriented_tri_vertices(fs[i]);
177 for (int j = 0; j < 3; ++j) c(i, j) = rows.at(v[j].vid(m));
178 }
179 data.validate();
180 return data;
181}
