479 {
480 const size_t floor = 64;
481 // A mesh that grows on demand only needs room for the passes not to grow at once.
482 const double factor = storage_grows_on_demand() ? 1.25 : m_preallocation_factor;
483 double c = std::ceil(factor * static_cast<double>(live_count));

485 if (capacity < live_count) capacity = live_count;
486 return capacity < floor ? floor : capacity;
487 }
