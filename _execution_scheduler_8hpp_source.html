324 bool operator()(AppMesh& m, const std::vector<std::pair<Op, Tuple>>& operation_tuples)

326 // The queue holds an operation's INDEX rather than its name. edit_operation_maps is a
327 // std::map, so iterating it yields the names in lexicographic order and an index is
328 // exactly a name's rank among them -- comparing indices is therefore identical to
329 // comparing the strings, which is what keeps the queue order, and so the output,
330 // bit-for-bit unchanged. What it buys is that a queue element is now trivially
331 // copyable: a heap sift moves 8 bytes instead of a std::string, and a tie on the
332 // priority is an integer compare instead of a string compare. It also turns the
333 // per-operation dispatch from a string-keyed std::map lookup into an index.
334 using OpId = uint32_t;
335 using Elem = std::tuple<double, OpId, Tuple, size_t>; // priority, op index, tuple, #retries
336 // Each task owns its queue outright -- it is seeded before any thread starts, and the
337 // task both pops from it and pushes its renewed operations back into it -- so those need
338 // no lock. `final_queue` is the one that genuinely crosses threads: tasks push retry
339 // overflow into it while running, and it is drained after the barrier. Both are
340 // std::priority_queue with the same comparator underneath, so pop order is unchanged.
341 using LocalQueue = wmtk::threading::serial_priority_queue<Elem>;
342 using SharedQueue = wmtk::threading::concurrent_priority_queue<Elem>;
343
344 std::vector<const Op*> op_name;
345 std::vector<std::function<std::optional<std::vector<Tuple>>(AppMesh&, const Tuple&)>*>
346 op_fn;
347 std::map<Op, OpId> op_id;
348 op_name.reserve(edit_operation_maps.size());
349 op_fn.reserve(edit_operation_maps.size());
350 for (auto& kv : edit_operation_maps) {
351 op_id.emplace(kv.first, OpId(op_name.size()));
352 op_name.push_back(&kv.first);
353 op_fn.push_back(&kv.second);
354 }
355 // An operation with no entry here was previously default-constructed into the map by
356 // operator[] and then called, which throws bad_function_call -- so it cannot occur in
357 // any working configuration. Say so plainly rather than ordering it arbitrarily.
358 const auto id_of = [&op_id](const Op& name) {
359 const auto it = op_id.find(name);
360 if (it == op_id.end()) {
361 log_and_throw_error("No operation registered under the name '{}'.", name);
362 }
363 return it->second;
364 };
365
366 std::atomic<bool> stop(false);
367 cnt_success = 0;
368 cnt_fail = 0;
369
370 // Whether anything actually watches the success count *while the pass runs*. When no
371 // stopping criterion is configured -- the case for every tetwild/triwild/simwild pass --
372 // nobody does, and the counters can be accumulated per task and folded in at the end
373 // instead of hammering one shared cache line from every thread on every operation.
374 const bool track_live_success =
375 stopping_criterion_checking_frequency != std::numeric_limits<size_t>::max();
376 std::atomic<size_t> live_success(0);
377
378 std::vector<LocalQueue> queues(num_threads);
379 SharedQueue final_queue;
380
381 // Contention accounting. Everything here is either a per-task local folded in once or a
382 // write to the task's own slot, so it adds nothing to the inner loop. It answers the
383 // two questions the pass could not previously be asked: how often ring acquisition
384 // loses a race, and how much of a "parallel" pass is really the serial drain.
385 m_stats = PassStats{};
386 std::atomic<size_t> lock_failures(0);
387 std::atomic<size_t> overflowed(0);
388 std::vector<double> task_seconds(queues.size(), 0.);
389
390 // Per-task tallies folded into the shared counters once, on the way out. The guard is
391 // RAII rather than a line at the bottom because run_single_queue has early returns.
392 //
393 // DECLARED HERE, NOT INSIDE THE LAMBDA, to work around an Apple clang 17 codegen bug.
394 // A local class declared inside a GENERIC lambda (one with an `auto` parameter, so its
395 // operator() is a template) gets its destructor emitted as an undefined reference and
396 // never defined, at every optimization level including -O0. The link then fails with
397 // Undefined symbols: ... ::'lambda'(auto&, int)::operator()<...>
398 // ::CountFlusher::~CountFlusher()
399 // referenced from every translation unit that instantiates a pass -- 26 references
400 // across 6 archives, defined nowhere. Homebrew clang 22 compiles the same code
401 // correctly, which is why CI does not see this.
402 //
403 // Hoisting the class one scope out is enough: it is still local to operator(), which is
404 // itself a member of a class template, and that instantiates fine. Nothing else changes
405 // -- `counts` is still constructed per task inside the lambda, so the RAII flush and its
406 // ordering are identical.
407 struct CountFlusher
408 {
409 std::atomic_int& success_total;
410 std::atomic_int& fail_total;
411 std::atomic<size_t>& lock_failure_total;
412 std::atomic<size_t>& overflow_total;
413 int success = 0;
414 int fail = 0;
415 size_t lock_failure = 0;
416 size_t overflow = 0;
417 ~CountFlusher()
418 {
419 success_total.fetch_add(success, std::memory_order_relaxed);
420 fail_total.fetch_add(fail, std::memory_order_relaxed);
421 lock_failure_total.fetch_add(lock_failure, std::memory_order_relaxed);
422 overflow_total.fetch_add(overflow, std::memory_order_relaxed);
423 }
424 };

426 auto run_single_queue = [&](auto& Q, int task_id) {
427 CountFlusher counts{cnt_success, cnt_fail, lock_failures, overflowed};
428
429 Elem ele_in_queue;
430 // Operations that lost a race for their ring wait here rather than going straight
431 // back into the live queue. Pushing them back immediately is a spin: the element was
432 // just popped as the queue's maximum, `retry` is the last tie-break key in Elem, so
433 // the requeued copy compares strictly greater than what was popped and nothing else
434 // was added -- it comes right back off the top and is retried against a conflict that
435 // has had no time to clear. Deferring lets every other operation this task owns run
436 // first, which is both useful work and the delay the conflict needs.
437 std::vector<Elem> second_chance;
438 // Operations popped since the deferred list was last given back to the queue.
439 // Waiting for the queue to drain completely can mean thousands of operations, by
440 // which time the mesh around a deferred operation has moved and its work has to be
441 // rediscovered. A bounded window gives the conflict time to clear without letting
442 // the operation go stale. 0 = only when the queue empties.
443 size_t since_refill = 0;
444 const auto refill = [&] {
445 for (auto& e : second_chance) {
446 Q.emplace(std::move(e));
447 }
448 second_chance.clear();
449 since_refill = 0;
450 };
451 for (;;) {
452 if (!second_chance.empty() && deferral_window > 0 &&
453 since_refill >= deferral_window) {
454 refill();
455 }
456 if (!Q.try_pop(ele_in_queue)) {
457 if (second_chance.empty()) {
458 break;
459 }
460 // Queue exhausted: the deferred operations have now had everything else
461 // run ahead of them, so give them another go. `retry` still increments on
462 // each attempt and still overflows to final_queue at max_retry_limit, so
463 // this terminates after at most that many rounds.
464 refill();
465 std::this_thread::yield();
466 continue;
467 }
468 ++since_refill;
469 auto& [weight, op, tup, retry] = ele_in_queue;
470 if (!tup.is_valid(m)) {
471 continue;
472 }
473
474 std::vector<Elem> renewed_elements;
475 {
476 auto locked_vid = lock_vertices(
477 m,
478 tup,
479 task_id); // Note that returning `Tuples` would be invalid.
480 if (!locked_vid) {
481 counts.lock_failure++;
482 retry++;
483 if (retry < max_retry_limit) {
484 second_chance.push_back(ele_in_queue);
485 } else {
486 retry = 0;
487 counts.overflow++;
488 final_queue.emplace(ele_in_queue);
489 }
490 continue;
491 }
492 if (tup.is_valid(m)) {
493 const Op& op_str = *op_name[op];
494 if (!is_weight_up_to_date(
495 m,
496 std::tuple<double, Op, Tuple>(weight, op_str, tup))) {
497 operation_cleanup(m);
498 continue;
499 } // this can encode, in qslim, recompute(energy) == weight.
500 auto newtup = (*op_fn[op])(m, tup);
501 std::vector<std::pair<Op, Tuple>> renewed_tuples;
502 if (newtup) {
503 renewed_tuples = renew_neighbor_tuples(m, op_str, newtup.value());
504 counts.success++;
505 if (track_live_success) {
506 live_success.fetch_add(1, std::memory_order_relaxed);
507 }
508 } else {
509 on_fail(m, op_str, tup);
510 counts.fail++;
511 }
512 for (const auto& [o, e] : renewed_tuples) {
513 auto val = priority(m, o, e);
514 if (should_renew(val)) {
515 renewed_elements.emplace_back(val, id_of(o), e, 0);
516 }
517 }
518 }
519 operation_cleanup(m); // Maybe use RAII
520 }
521 for (auto& e : renewed_elements) {
522 Q.emplace(e);
523 }
524
525 if (stop.load(std::memory_order_acquire)) {
526 return;
527 }
528 if (track_live_success && live_success.load(std::memory_order_relaxed) >
529 stopping_criterion_checking_frequency) {
530 if (stopping_criterion(m)) {
531 stop.store(true);
532 return;
533 }
534 }
535 }
536 };
537
538 if (policy == ExecutionPolicy::kSeq) {
539 for (const auto& [op, e] : operation_tuples) {
540 if (!e.is_valid(m)) {
541 continue;
542 }
543 final_queue.emplace(priority(m, op, e), id_of(op), e, 0);
544 }
545 run_single_queue(final_queue, 0);
546 } else {
547 for (const auto& [op, e] : operation_tuples) {
548 if (!e.is_valid(m)) {
549 continue;
550 }
551 queues[get_partition_id(m, e)].emplace(priority(m, op, e), id_of(op), e, 0);
552 }
553 // Comment out parallel: work on serial first.
554 using clock = std::chrono::steady_clock;
555 const auto t_parallel = clock::now();
556 wmtk::threading::task_group tg;
557 for (int task_id = 0; task_id < queues.size(); task_id++) {
558 tg.run([&run_single_queue, &queues, &task_seconds, task_id] {
559 const auto t0 = clock::now();
560 run_single_queue(queues[task_id], task_id);
561 // Each task writes only its own slot.
562 task_seconds[task_id] =
563 std::chrono::duration<double>(clock::now() - t0).count();
564 });
565 }
566 tg.wait();
567 m_stats.parallel_seconds =
568 std::chrono::duration<double>(clock::now() - t_parallel).count();
569 m_stats.final_queue_size = final_queue.size();
570
571 logger().debug("Parallel Complete, remains element {}", final_queue.size());
572
573 const auto t_tail = clock::now();
574 run_single_queue(final_queue, 0);
575 m_stats.serial_tail_seconds =
576 std::chrono::duration<double>(clock::now() - t_tail).count();
577 }
578
579 m_stats.lock_failures = lock_failures.load(std::memory_order_relaxed);
580 m_stats.overflowed = overflowed.load(std::memory_order_relaxed);
581 if (!task_seconds.empty()) {
582 const auto mm = std::minmax_element(task_seconds.begin(), task_seconds.end());
583 m_stats.idlest_task_seconds = *mm.first;
584 m_stats.busiest_task_seconds = *mm.second;
585 }
586
587 logger().info(
588 "executed: {} | success / fail: {} / {}",
589 (int)cnt_success + (int)cnt_fail,
590 (int)cnt_success,
591 (int)cnt_fail);
592 log_contention();
593 return true;
594 }

628 void log_contention() const
629 {
630 if (policy == ExecutionPolicy::kSeq || !logger().should_log(spdlog::level::debug)) {
631 return;
632 }
633 const int executed = (int)cnt_success + (int)cnt_fail;
634 const double total = m_stats.parallel_seconds + m_stats.serial_tail_seconds;
635 logger().debug(
636 " contention: {} ring-acquisition failures over {} executed ops ({:.2f} per op); "
637 "{} overflowed to the serial queue ({} queued at the barrier)",
638 m_stats.lock_failures,
639 executed,
640 executed > 0 ? double(m_stats.lock_failures) / executed : 0.,
641 m_stats.overflowed,
642 m_stats.final_queue_size);
643 logger().debug(
644 " time: {:.4}s parallel + {:.4}s serial tail ({:.1f}% of the pass); busiest task "
645 "{:.4}s, idlest {:.4}s",
646 m_stats.parallel_seconds,
647 m_stats.serial_tail_seconds,
648 total > 0. ? 100. * m_stats.serial_tail_seconds / total : 0.,
649 m_stats.busiest_task_seconds,
650 m_stats.idlest_task_seconds);
651 }

654 std::atomic_int cnt_success = 0;
655 std::atomic_int cnt_fail = 0;
