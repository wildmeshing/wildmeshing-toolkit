5inline double mixed_loss(double q, double target, double* derivative = nullptr)
6{
7 const double z = std::max(0., q / target - 1);
8 if (derivative) *derivative = 2 * z / target + 1e-8 / q;
9 return z * z + 1e-8 * std::log(q);
10}
11inline Json mixed_score(const Mesh& m, double target)
12{
13 size_t bad = 0;
14 double maximum = 0;
15 long double loss = 0;
16 for (const auto& c : m.cells) {
17 bad += c.energy > target;
18 maximum = std::max(maximum, c.energy);
19 loss += mixed_loss(c.energy, target);
20 }

22 {"maximum_AMIPS", maximum},

64inline std::vector<V> cell_gradient(const Mesh& m, const Cell& c)
65{
66 std::vector<V> grad(c.v.size(), V::Zero());
67 if (c.type == 4) {
68 std::array<V, 4> x, dg;
69 for (int j = 0; j < 4; ++j) x[j] = m.p[c.v[j]] - m.p[c.v[0]];
70 const double det = native::determinant(m.p, c.v);
71 double s = 0;
72 for (int i = 0; i < 4; ++i)
73 for (int j = 0; j < i; ++j) s += (x[i] - x[j]).squaredNorm();
74 dg[1] = x[2].cross(x[3]);
75 dg[2] = x[3].cross(x[1]);
76 dg[3] = x[1].cross(x[2]);
77 dg[0] = -dg[1] - dg[2] - dg[3];
78 const V sum = x[0] + x[1] + x[2] + x[3];
79 for (int j = 0; j < 4; ++j)
80 grad[j] = c.energy * ((8 * x[j] - 2 * sum) / s - (2. / 3) * dg[j] / det);
81 return grad;
82 }
83 for (const auto& q : native::quadrature(c.type, 4)) {
84 const M a = native::matrix(m.p, c, q);
85 const double det = a.determinant(), s = a.squaredNorm();
86 M cof;
87 cof.col(0) = a.col(1).cross(a.col(2));
88 cof.col(1) = a.col(2).cross(a.col(0));
89 cof.col(2) = a.col(0).cross(a.col(1));
90 const M g = q.w / std::pow(det, 2. / 3) * (2 * a - (2. / 3) * (s / det) * cof);
91 for (size_t j = 0; j < c.v.size(); ++j) grad[j] += g * q.g[j];
92 }
93 return grad;
94}
95struct InputFreedom
96{
97 std::vector<M> tangent;

100 explicit InputFreedom(const Mesh& m, const Rules& rules)
101 : tangent(m.p.size(), M::Identity())
102 , faces(m.p.size())
103 , children(m.p.size())
104 {
105 for (const auto& [k, f] : rules.kept)
106 for (size_t v : f.v) faces[v].push_back(f);
107 for (size_t v = 0; v < m.p.size(); ++v) {
108 if (m.outer[v] || m.fixed(v)) tangent[v].setZero();
109 if (!m.input[v]) {
110 if (m.correspondence[v] >= 0 && m.input.at(m.correspondence[v]))
111 children[m.correspondence[v]].push_back(v);
112 continue;
113 }
114 if (faces[v].empty()) {
115 tangent[v].setZero();
116 continue;
117 }
118 Eigen::MatrixXd normals(faces[v].size(), 3);
119 for (size_t j = 0; j < faces[v].size(); ++j)
120 normals.row(j) = rules.patch_normal[faces[v][j].patch].transpose();
121 Eigen::JacobiSVD<Eigen::MatrixXd> svd(normals, Eigen::ComputeFullV);
122 M projection = M::Identity();
123 for (int j = 0; j < svd.singularValues().size(); ++j)
124 if (svd.singularValues()[j] > 1e-10)
125 projection -= svd.matrixV().col(j) * svd.matrixV().col(j).transpose();
126 tangent[v] = projection;
127 }
128 }

252 valid = false;

257 }

262 after += thickness_weight * e * e;
263 }

265 size_t j = 0;

272 }
273 }

276 {"vertex", v},

278 {"before", before},
279 {"after", after},
280 {"backtracks", bt}};

282 }
283 for (size_t j = 0; j < group.size(); ++j) m.p[group[j]] = old[j];
284 }
285 for (size_t j = 0; j < group.size(); ++j) m.p[group[j]] = old[j];
286 return false;
