943 std::shared_ptr<SampleEnvelope> surface_envelope_for_face(
944 const std::array<size_t, 3>& vids) const override
945 {
946 uint64_t mask = face_mask(vids);
947 bool all_offset = true;
948 for (const size_t v : vids) {
949 all_offset = all_offset && m_vertex_extra[v].m_is_on_offset;
950 }
951 // The ambiguous case: all corners can be on region boundaries and on the offset surface
952 // at once. The corner-mask AND is then necessary but not sufficient for the face lying
953 // on a shared boundary; ask the face's own class, the only record that distinguishes a
954 // chord from a boundary. Reading the slot is safe here: a split child never reaches
955 // this branch, and the m_is_surface_fs guard leaves both constraints standing when a
956 // slot is illegible.
957 if (mask != 0 && all_offset) {
958 if (const auto found = try_tuple_from_face(vids)) {
959 const size_t fid = std::get<1>(*found);
960 if (m_face_attribute[fid].m_is_surface_fs) {
961 if (face_is_offset(fid)) {
962 mask = 0; // an offset face lies on no region boundary
963 } else {
964 all_offset = false; // a region face is not the offset surface
965 }
966 }
967 }
968 }
969 const std::shared_ptr<SampleEnvelope> base = containment_for(mask, all_offset);
970 if (base || m_deform_tags.empty()) return base;
971 // deform_others' ops-only tube: a released boundary is held by no mask -- its vertices
972 // were freed so smoothing can carry the object -- which would leave the operations free
973 // to decimate and reposition it. A face the masks and the offset class do not claim, but
974 // which lies on a released boundary by its incident tets' current tags, is held to the
975 // tube around the boundary's current shape.
976 if (const auto found = try_tuple_from_face(vids)) {
977 if (face_borders_released_boundary(std::get<0>(*found))) return released_envelope();
978 }
979 return nullptr;
980 }

994 std::shared_ptr<SampleEnvelope> smoothing_energy_envelope(const size_t vid) const override
995 {
996 if (m_vertex_extra[vid].m_is_on_offset && !vertex_is_on_region(vid)) {
997 return nullptr;
998 }
999 const uint64_t mask = vertex_boundary_mask(vid);
1000 if (mask == 0) {
1001 return nullptr;
1002 }
1003 std::shared_ptr<SampleEnvelope> best;
1004 double worst_d2 = -1.;
1005 for (const auto& [tag, env] : m_tag_envelopes) {
1006 const auto it = m_tag_bit.find(tag);
1007 if (it == m_tag_bit.end() || !(mask & (uint64_t(1) << it->second))) continue;
1008 if (!best) {
1009 best = env;
1010 if ((mask & (mask - 1)) == 0) break; // single bit: no violation contest to run
1011 worst_d2 = env->squared_distance(m_vertex_attribute[vid].m_posf);
1012 continue;
1013 }
1014 const double d2 = env->squared_distance(m_vertex_attribute[vid].m_posf);
1015 if (d2 > worst_d2) {
1016 worst_d2 = d2;
1017 best = env;
1018 }
1019 }
1020 return best;
1021 }
