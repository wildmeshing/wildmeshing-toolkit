1029 {

1080 double band_cell_term(const std::array<size_t, 4>& vids) const
1081 {
1082 if (m_n_regions > 1 && !m_offset_params.allow_mult_comp) {
1083 log_and_throw_error(
1084 "the energy's band term reads one field, and this input has {} regions",
1085 m_n_regions);
1086 }
1087 if (m_offset_params.band_volume_rule == "exact" ||
1088 m_offset_params.band_volume_rule == "front_flux" || swept_rule()) {
1089 // EXPERIMENTAL_band_volume_rule "exact" (and "front_flux", whose per-cell quantity
1090 // it is, see above): J_t / delta, J_t = int_t (d - delta) dV in closed form
1091 // (ExactBandIntegral), in place of the corner bound
1092 const std::array<Vector3d, 4> p = {
1093 {m_vertex_attribute[vids[0]].m_posf,
1094 m_vertex_attribute[vids[1]].m_posf,
1095 m_vertex_attribute[vids[2]].m_posf,
1096 m_vertex_attribute[vids[3]].m_posf}};
1097 const double delta = m_offset_params.target_distance;
1098 return m_band_exact->value(p, delta) / delta;
1099 }
1100 int64_t best = -1;
1101 return band_cell_corner_bound(vids, best);
1102 }
1104 double band_cell_corner_bound(const std::array<size_t, 4>& vids, int64_t& best) const
1105 {
1106 const double delta = m_offset_params.target_distance;
1107 std::array<Vector3d, 4> p;
1108 for (int j = 0; j < 4; ++j) p[size_t(j)] = m_vertex_attribute[vids[size_t(j)]].m_posf;
1109 const double vol = std::abs((p[1] - p[0]).dot((p[2] - p[0]).cross(p[3] - p[0]))) / 6.;
1110 const double md =
1111 min_mean_distance(*m_input_complex_envelope, m_input_complex_pad, p.data(), 4, best);
1112 return vol * ((md - delta) / delta);

2397 // raw sum of AMIPS over the one-ring (AMIPSEnergy3D), as TetWild's does.
2398 if (tetwild_rules()) return nullptr;
2399 auto sum = std::make_shared<optimization::EnergySum>();
2400 sum->add_energy(amips3_energy(vid));
2401 // Before the march (repulsion_smoothing()) there is no front.
2402 if (is_repulsion_vertex(vid)) {
2403 sum->add_energy(repulsion_energy(vid));
2404 return sum;
2405 }
2406 if (vertex_carries_offset_term(vid)) {
2407 sum->add_energy(front_energy(vid));
2408 } else if (!vertex_is_held(vid)) {
2409 // E's band term moves with this vertex too -- except a vertex held on the input.
2410 // There d = 0 at the vertex and d has a kink: the term's slope along the input's
2411 // normal jumps across the surface, and its curvature there is 0 (or ~1e10 near an
2412 // input edge), so the Newton step along the normal is set by AMIPS alone and lands
2413 // far off; the projected step then discards the normal motion anyway. MEASURED
2414 // 2026-10-09, cube, serial, 3 turns: with the term these solves took 24.6 Newton
2415 // iterations and ~20 line-search trials per iteration (72% of the smoother's value
2416 // calls), without it 3.9 and 1; 19.3 s -> 11.3 s. E still guards the move:
2417 // smooth_nonfront_vertex() vetoes on the ring's sum of tet_energy().
2418 // Under "exact", not where it cancels (BAND TERM THAT CANCELS, tet_energy()): a ring
2419 // of band cells clear of the input's bisectors tiles the star of vid, so the term is a
2420 // constant, and its cells' exact integrals would add only work and their rounding to
2421 // every Newton step. The veto's own test (cancels_before in smooth_nonfront_vertex()).
2422 // Constant wherever the move can end, not only here: vid is not held, so not on the
2423 // domain wall (face_is_held()), and its ring closes around it; the engine keeps no
2424 // position that inverts a ring cell, so vid stays inside the convex hull of its link;
2425 // and the test reads the points only through their box, their least side of each
2426 // bisector polytope's planes, and min_P max d_P (d_P convex), none of which a point
